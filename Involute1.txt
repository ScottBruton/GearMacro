' True while the gear form is open; main() loops with DoEvents so SW stays interactive (modeless pick)
Public gbGearFormOpen As Boolean

Public Sub CloseGearForm()
    gbGearFormOpen = False
    On Error Resume Next
    Unload UserForm1
    On Error GoTo 0
End Sub

Sub main()
    On Error GoTo ErrorHandler
    GearTrainRole = GEAR_ROLE_DRIVING
    GearTrainMeshAngleDeg = 0#
    GearTrainDrivingPartTitle = ""
    GearTrainDrivingBodyLabel = ""
    GearTrainDrivingAxisName = ""
    GearTrainDrivingIsInternal = False
    GearTrainLayoutCenterDistanceMM = 0#
    GearGeom.IsInternal = False
    Load UserForm1
    gbGearFormOpen = True
    UserForm1.Show vbModeless
    ' Keep macro alive until form closes; otherwise the form blocks graphics or unloads
    Do While gbGearFormOpen
        DoEvents
    Loop
    Exit Sub
ErrorHandler:
    MsgBox "Error loading UserForm1:" & vbCrLf & Err.Number & " - " & Err.Description, vbCritical, "Form Error"
End Sub

' Orchestrator: root cylinder -> tooth sketch -> extrude -> pattern -> combine -> validate
Sub GenerateGear(PressureAngle As Double, Module As Double, NumTeeth As Integer, Width As Double)
    On Error GoTo GenFail

    Set swApp = Application.SldWorks
    If swApp Is Nothing Then
        MsgBox "SolidWorks is not running.", vbCritical
        Exit Sub
    End If

    GearGeom.PressureAngleDeg = PressureAngle
    GearGeom.ModuleMM = Module
    GearGeom.NumTeeth = NumTeeth
    GearGeom.WidthMM = Width

    If Not ValidateGearInputsPublic(GearGeom.PressureAngleDeg, GearGeom.ModuleMM, GearGeom.NumTeeth, GearGeom.WidthMM) Then
        Exit Sub
    End If

    CalculateGearGeometry

    If GearTrainRole = GEAR_ROLE_DRIVEN Then
        If GearTrainDrivingPartTitle = "" Then
            MsgBox "Select a driving gear from the list (Refresh list after opening the part).", vbExclamation
            Exit Sub
        End If
        If Not ApplyDrivenGearInputsFromDrivingGear() Then
            MsgBox "Driven gear mode requires a valid driving gear (body name ""teethT pitchDia"" e.g. 100T 125, or InvoluteGear custom properties).", vbExclamation
            Exit Sub
        End If
        CalculateGearGeometry
        Set swDoc = FindOpenPartDocumentByTitle(GearTrainDrivingPartTitle)
        SyncDrivingGearAxisReference swDoc
        If swDoc Is Nothing Then
            MsgBox "Could not open the driving gear part: " & GearTrainDrivingPartTitle, vbCritical
            Exit Sub
        End If
    Else
        Set swDoc = swApp.NewPart
        If swDoc Is Nothing Then
            MsgBox "Could not create a new part document.", vbCritical
            Exit Sub
        End If
    End If

    ' Ensure target part is active (SelectByID2 / sketch ops fail on inactive docs)
    Dim activateErr As Long
    Dim docTitle As String
    docTitle = swDoc.GetTitle
    On Error Resume Next
    swApp.ActivateDoc3 docTitle, True, 0, activateErr
    If activateErr <> 0 Then swApp.ActivateDoc2 docTitle, False, 0
    swDoc.Visible = True
    On Error GoTo GenFail

    Set swPart = swDoc
    Set swModelDocExt = swDoc.Extension
    Set swSketchManager = swPart.SketchManager
    Set swFeatMgr = swDoc.FeatureManager
    Set swSelMgr = swDoc.SelectionManager
    Pi = 3.14159265358979

    swModelDocExt.SetUserPreferenceInteger swUserPreferenceIntegerValue_e.swUnitsLinear, _
        swUserPreferenceOption_e.swDetailingNoOptionSpecified, swMM
    swApp.SetUserPreferenceToggle swUserPreferenceToggle_e.swInputDimValOnCreate, False

    DebugLog = ""
    DebugPrint "=== GenerateGear start (blank+cut+pattern) z=" & GearGeom.NumTeeth & " m=" & GearGeom.ModuleMM & " ==="

    SyncGearTrainLayoutCenterFromForm
    ResetGearBuildFeatureRefs
    ResetGearSketchOffset
    SetGearTrainSketchOffsetFromLayout
    CaptureSolidBodyCountBeforeGearBuild

    ' Addendum disk boss, cut one tooth-gap, circular pattern the cut
    If Not CreateOrResetPart() Then
        DebugPrint "GenerateGear stopped at CreateOrResetPart"
        GoTo GenFail
    End If
    If Not CreateAddendumBlank() Then
        DebugPrint "GenerateGear stopped at CreateAddendumBlank"
        GoTo GenFail
    End If
    If Not CreateToothGapSketch() Then
        DebugPrint "GenerateGear stopped at CreateToothGapSketch"
        GoTo GenFail
    End If
    If Not CutExtrudeToothGap() Then
        DebugPrint "GenerateGear stopped at CutExtrudeToothGap"
        GoTo GenFail
    End If
    If Not CircularPatternTooth() Then
        DebugPrint "GenerateGear stopped at CircularPatternTooth"
        GoTo GenFail
    End If
    If Not RebuildAndValidate() Then
        DebugPrint "GenerateGear stopped at RebuildAndValidate"
        GoTo GenFail
    End If

    NameGearSolidBodyAfterBuild
    NameGearAxisForCurrentBuild
    StoreGearMetadataOnPart
    If GearTrainRole = GEAR_ROLE_DRIVEN Then
        DebugPrint "Gear train driven layout: C=" & Format(GetGearTrainLayoutCenterDistanceMM(), "0.###") & _
            " mm angle=" & Format(GearTrainMeshAngleDeg, "0.###") & " deg from driving " & GearTrainDrivingPartTitle & _
            " axis=" & GearTrainDrivingAxisName
    End If

    swDoc.ViewZoomtofit2
    swDoc.ClearSelection2 True
    DebugPrint "=== GenerateGear completed successfully ==="
    Exit Sub

GenFail:
    DebugPrint "=== GenerateGear FAILED: " & Err.Description & " ==="
    MsgBox "Gear generation failed. Open Immediate Window (Ctrl+G) and read the last ""stopped at"" line.", vbCritical, "Gear Macro"
End Sub
