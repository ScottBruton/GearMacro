Private Sub cbGenerate_Click()
    Dim PressureAngle As Double
    Dim Module As Double
    Dim NumTeeth As Integer
    Dim Width As Double
    Dim ErrorMsg As String
    
    ' Validate and get Pressure Angle
    If Trim(tbPressureAngle.Text) = "" Then
        ErrorMsg = ErrorMsg & "Pressure Angle is required." & vbCrLf
    Else
        On Error Resume Next
        PressureAngle = CDbl(tbPressureAngle.Text)
        If Err.Number <> 0 Or PressureAngle <= 0 Then
            ErrorMsg = ErrorMsg & "Pressure Angle must be a positive number." & vbCrLf
        End If
        On Error GoTo 0
    End If
    
    ' Validate and get Module
    If Trim(tbModule.Text) = "" Then
        ErrorMsg = ErrorMsg & "Module is required." & vbCrLf
    Else
        On Error Resume Next
        Module = CDbl(tbModule.Text)
        If Err.Number <> 0 Or Module <= 0 Then
            ErrorMsg = ErrorMsg & "Module must be a positive number." & vbCrLf
        End If
        On Error GoTo 0
    End If
    
    ' Validate and get Number of Teeth
    If Trim(tbTeeth.Text) = "" Then
        ErrorMsg = ErrorMsg & "Number of Teeth is required." & vbCrLf
    Else
        On Error Resume Next
        NumTeeth = CInt(tbTeeth.Text)
        If Err.Number <> 0 Or NumTeeth <= 0 Then
            ErrorMsg = ErrorMsg & "Number of Teeth must be a positive integer." & vbCrLf
        Else
            ' Check undercutting constraint
            If Trim(tbPressureAngle.Text) <> "" Then
                Dim MinTeeth As Double
                Dim PressureAngleRad As Double
                Dim Pi As Double
                Pi = 3.14159265358979
                PressureAngle = CDbl(tbPressureAngle.Text)
                PressureAngleRad = PressureAngle * Pi / 180
                MinTeeth = 2 / (Sin(PressureAngleRad) ^ 2)
                
                If NumTeeth < MinTeeth Then
                    ErrorMsg = ErrorMsg & "WARNING: Undercutting will occur! " & _
                               "Minimum teeth for " & Format(PressureAngle, "0.0") & "° pressure angle: " & _
                               Format(Int(MinTeeth) + 1, "0") & ". Continue anyway?" & vbCrLf
                End If
            End If
        End If
        On Error GoTo 0
    End If
    
    ' Validate and get Width
    If Trim(tbWidth.Text) = "" Then
        ErrorMsg = ErrorMsg & "Width is required." & vbCrLf
    Else
        On Error Resume Next
        Width = CDbl(tbWidth.Text)
        If Err.Number <> 0 Or Width <= 0 Then
            ErrorMsg = ErrorMsg & "Width must be a positive number." & vbCrLf
        End If
        On Error GoTo 0
    End If
    
    ' Check if there are any errors
    If ErrorMsg <> "" Then
        MsgBox "Please correct the following errors:" & vbCrLf & vbCrLf & ErrorMsg, vbExclamation, "Validation Error"
        Exit Sub
    End If
    
    ' Hide the form and generate the gear
    Me.Hide
    Call GenerateGear(PressureAngle, Module, NumTeeth, Width)
    Unload Me
    
End Sub

Private Sub ValidateInputs()
    ' This function validates all inputs and provides feedback
    On Error Resume Next
    Dim PressureAngle As Double
    Dim Module As Double
    Dim NumTeeth As Integer
    Dim Width As Double
    Dim MinTeeth As Double
    Dim Pi As Double
    Dim PressureAngleRad As Double
    Dim WarningMsg As String
    Dim PitchDiameter As Double
    
    Pi = 3.14159265358979
    
    ' Reset all text box backgrounds to white (only if controls exist)
    If Not tbPressureAngle Is Nothing Then tbPressureAngle.BackColor = vbWhite
    If Not tbModule Is Nothing Then tbModule.BackColor = vbWhite
    If Not tbTeeth Is Nothing Then tbTeeth.BackColor = vbWhite
    If Not tbWidth Is Nothing Then tbWidth.BackColor = vbWhite
    
    ' Validate Pressure Angle
    If Not tbPressureAngle Is Nothing And Trim(tbPressureAngle.Text) <> "" Then
        On Error Resume Next
        PressureAngle = CDbl(tbPressureAngle.Text)
        If Err.Number = 0 And PressureAngle > 0 Then
            ' Check if it's a standard value (14.5, 20, or 25 degrees)
            If PressureAngle <> 14.5 And PressureAngle <> 20 And PressureAngle <> 25 Then
                WarningMsg = WarningMsg & "Note: Standard pressure angles are 14.5°, 20°, or 25°." & vbCrLf
            End If
            If PressureAngle < 10 Or PressureAngle > 30 Then
                tbPressureAngle.BackColor = RGB(255, 200, 200) ' Light red
                WarningMsg = WarningMsg & "Warning: Pressure angle should typically be between 10° and 30°." & vbCrLf
            End If
        End If
        On Error GoTo 0
    End If
    
    ' Validate Module
    If Not tbModule Is Nothing And Trim(tbModule.Text) <> "" Then
        On Error Resume Next
        Module = CDbl(tbModule.Text)
        If Err.Number = 0 And Module > 0 Then
            If Module < 0.5 Or Module > 25 Then
                tbModule.BackColor = RGB(255, 200, 200) ' Light red
                WarningMsg = WarningMsg & "Warning: Module should typically be between 0.5 and 25 mm." & vbCrLf
            End If
        End If
        On Error GoTo 0
    End If
    
    ' Validate Number of Teeth and check undercutting constraint
    If Not tbTeeth Is Nothing And Not tbPressureAngle Is Nothing And Trim(tbTeeth.Text) <> "" And Trim(tbPressureAngle.Text) <> "" Then
        On Error Resume Next
        NumTeeth = CInt(tbTeeth.Text)
        PressureAngle = CDbl(tbPressureAngle.Text)
        If Err.Number = 0 And NumTeeth > 0 And PressureAngle > 0 Then
            ' Calculate minimum teeth to avoid undercutting: z_min = 2 / sin²(a)
            PressureAngleRad = PressureAngle * Pi / 180
            MinTeeth = 2 / (Sin(PressureAngleRad) ^ 2)
            
            If NumTeeth < MinTeeth Then
                tbTeeth.BackColor = RGB(255, 200, 200) ' Light red
                WarningMsg = WarningMsg & "WARNING: Undercutting will occur! " & _
                             "Minimum teeth for " & Format(PressureAngle, "0.0") & "° pressure angle: " & _
                             Format(Int(MinTeeth) + 1, "0") & vbCrLf
            ElseIf NumTeeth < MinTeeth + 2 Then
                tbTeeth.BackColor = RGB(255, 255, 200) ' Light yellow
                WarningMsg = WarningMsg & "Note: Recommended minimum teeth: " & _
                             Format(Int(MinTeeth) + 3, "0") & " to avoid undercutting." & vbCrLf
            End If
            
            If NumTeeth > 200 Then
                tbTeeth.BackColor = RGB(255, 200, 200) ' Light red
                WarningMsg = WarningMsg & "Warning: Very high number of teeth may cause performance issues." & vbCrLf
            End If
        End If
        On Error GoTo 0
    End If
    
    ' Validate Width relative to Module and Pitch Diameter
    If Not tbWidth Is Nothing And Not tbModule Is Nothing And Not tbTeeth Is Nothing And Trim(tbWidth.Text) <> "" And Trim(tbModule.Text) <> "" And Trim(tbTeeth.Text) <> "" Then
        On Error Resume Next
        Width = CDbl(tbWidth.Text)
        Module = CDbl(tbModule.Text)
        NumTeeth = CInt(tbTeeth.Text)
        If Err.Number = 0 And Width > 0 And Module > 0 And NumTeeth > 0 Then
            PitchDiameter = Module * NumTeeth
            
            ' Width should be reasonable relative to module (typically 0.3 to 2.0 times module)
            If Width < 0.3 * Module Then
                tbWidth.BackColor = RGB(255, 200, 200) ' Light red
                WarningMsg = WarningMsg & "Warning: Width is very small relative to module." & vbCrLf
            ElseIf Width > 5 * Module Then
                tbWidth.BackColor = RGB(255, 255, 200) ' Light yellow
                WarningMsg = WarningMsg & "Note: Width is large relative to module." & vbCrLf
            End If
            
            ' Width should be reasonable relative to pitch diameter (typically < 0.5 * diameter)
            If Width > 0.5 * PitchDiameter Then
                tbWidth.BackColor = RGB(255, 255, 200) ' Light yellow
                WarningMsg = WarningMsg & "Note: Width is large relative to pitch diameter." & vbCrLf
            End If
        End If
        On Error GoTo 0
    End If
    
    ' Update status (you might want to add a label for this)
    ' For now, we'll just use the background colors as visual feedback
    
    ' Update gear calculations display
    UpdateGearCalculations
End Sub

Private Sub UpdateGearCalculations()
    ' Calculate and display pitch diameter and other gear train info
    Dim Module As Double
    Dim NumTeeth As Integer
    Dim PressureAngle As Double
    Dim PitchDiameter As Double
    Dim BaseDiameter As Double
    Dim AddendumDiameter As Double
    Dim DedendumDiameter As Double
    Dim Pi As Double
    Dim CenterDistance As Double
    Dim Gear2Teeth As Integer
    Dim InfoText As String
    Dim ctrl As Object
    Dim ErrNum As Long
    
    Pi = 3.14159265358979
    InfoText = ""
    
    ' Calculate pitch diameter for current gear
    If Not tbModule Is Nothing And Not tbTeeth Is Nothing And Trim(tbModule.Text) <> "" And Trim(tbTeeth.Text) <> "" Then
        On Error Resume Next
        Module = CDbl(tbModule.Text)
        ErrNum = Err.Number
        NumTeeth = CInt(tbTeeth.Text)
        If ErrNum <> 0 Or Err.Number <> 0 Then
            On Error GoTo 0
            Exit Sub
        End If
        On Error GoTo 0
        
        If Module > 0 And NumTeeth > 0 Then
            PitchDiameter = Module * NumTeeth
            InfoText = "Pitch Diameter: " & Format(PitchDiameter, "0.000") & " mm"
            
            ' Update pitch diameter label - try multiple possible names
            UpdateLabel "lblPitchDiameter", Format(PitchDiameter, "0.000") & " mm"
            UpdateLabel "PitchDiameter", Format(PitchDiameter, "0.000") & " mm"
            UpdateTextBox "txtPitchDiameter", Format(PitchDiameter, "0.000") & " mm"
            
            ' Calculate other diameters - these don't require pressure angle except for base diameter
            ' Addendum and Dedendum can be calculated with just Module and Pitch Diameter
            AddendumDiameter = PitchDiameter + 2 * Module
            DedendumDiameter = PitchDiameter - 2.5 * Module
            
            InfoText = InfoText & vbCrLf & "Addendum Diameter: " & Format(AddendumDiameter, "0.000") & " mm"
            InfoText = InfoText & vbCrLf & "Dedendum Diameter: " & Format(DedendumDiameter, "0.000") & " mm"
            
            ' Update Addendum Diameter
            UpdateLabel "lblAddendumDiameter", Format(AddendumDiameter, "0.000") & " mm"
            UpdateLabel "AddendumDiameter", Format(AddendumDiameter, "0.000") & " mm"
            UpdateTextBox "txtAddendumDiameter", Format(AddendumDiameter, "0.000") & " mm"
            
            ' Update Dedendum Diameter
            UpdateLabel "lblDedendumDiameter", Format(DedendumDiameter, "0.000") & " mm"
            UpdateLabel "DedendumDiameter", Format(DedendumDiameter, "0.000") & " mm"
            UpdateTextBox "txtDedendumDiameter", Format(DedendumDiameter, "0.000") & " mm"
            
            ' Base Diameter requires pressure angle
            On Error Resume Next
            If Not tbPressureAngle Is Nothing Then
                If Trim(tbPressureAngle.Text) <> "" Then
                    PressureAngle = CDbl(tbPressureAngle.Text)
                    ErrNum = Err.Number
                    On Error GoTo 0
                    
                    If ErrNum = 0 And PressureAngle > 0 And PitchDiameter > 0 Then
                        ' Convert pressure angle from degrees to radians for Cos function
                        Dim PressureAngleRad As Double
                        PressureAngleRad = PressureAngle * Pi / 180
                        BaseDiameter = PitchDiameter * Cos(PressureAngleRad)
                        InfoText = InfoText & vbCrLf & "Base Diameter: " & Format(BaseDiameter, "0.000") & " mm"
                        
                        ' Update Base Diameter
                        UpdateLabel "lblBaseDiameter", Format(BaseDiameter, "0.000") & " mm"
                        UpdateLabel "BaseDiameter", Format(BaseDiameter, "0.000") & " mm"
                        UpdateTextBox "txtBaseDiameter", Format(BaseDiameter, "0.000") & " mm"
                    End If
                Else
                    On Error GoTo 0
                End If
            Else
                On Error GoTo 0
            End If
            
            ' Calculate center distance if second gear teeth is provided
            On Error Resume Next
            Dim Gear2TeethControl As Object
            Set Gear2TeethControl = Me.Controls("tbGear2Teeth")
            If Err.Number = 0 And Not Gear2TeethControl Is Nothing Then
                If Trim(Gear2TeethControl.Text) <> "" Then
                    Gear2Teeth = CInt(Gear2TeethControl.Text)
                    ErrNum = Err.Number
                    On Error GoTo 0
                    
                    If ErrNum = 0 And Gear2Teeth > 0 Then
                        CenterDistance = (Module * (NumTeeth + Gear2Teeth)) / 2
                        InfoText = InfoText & vbCrLf & "Center Distance: " & Format(CenterDistance, "0.000") & " mm"
                        
                        ' Update Center Distance
                        UpdateLabel "lblCenterDistance", Format(CenterDistance, "0.000") & " mm"
                        UpdateLabel "CenterDistance", Format(CenterDistance, "0.000") & " mm"
                        UpdateTextBox "txtCenterDistance", Format(CenterDistance, "0.000") & " mm"
                    End If
                Else
                    On Error GoTo 0
                End If
            Else
                On Error GoTo 0
            End If
            
            ' Update a general info label or textbox if it exists (for displaying all calculated values)
            UpdateLabel "lblGearInfo", InfoText
            UpdateTextBox "txtGearInfo", InfoText
            UpdateLabel "GearInfo", InfoText
        End If
    End If
End Sub

Private Sub UpdateLabel(ControlName As String, Value As String)
    ' Helper function to update a label control
    On Error Resume Next
    Dim ctrl As Object
    Set ctrl = Me.Controls(ControlName)
    If Err.Number = 0 Then
        ctrl.Caption = Value
    End If
    On Error GoTo 0
End Sub

Private Sub UpdateTextBox(ControlName As String, Value As String)
    ' Helper function to update a textbox control
    On Error Resume Next
    Dim ctrl As Object
    Set ctrl = Me.Controls(ControlName)
    If Err.Number = 0 Then
        ctrl.Text = Value
    End If
    On Error GoTo 0
End Sub

Private Sub cbGearTrain_Click()
    ' Lock/unlock module based on gear train mode
    If cbGearTrain.Value = True Then
        ' Gear train mode: lock module and pressure angle
        tbModule.Enabled = False
        tbPressureAngle.Enabled = False
        tbModule.BackColor = RGB(240, 240, 240) ' Light gray to show it's locked
        tbPressureAngle.BackColor = RGB(240, 240, 240)
        ' Show message about gear train requirements
        If TypeName(Me.Controls("lblGearTrainInfo")) = "Label" Then
            Me.Controls("lblGearTrainInfo").Caption = "Module and Pressure Angle locked for gear train compatibility"
            Me.Controls("lblGearTrainInfo").Visible = True
        End If
    Else
        ' Normal mode: unlock module and pressure angle
        tbModule.Enabled = True
        tbPressureAngle.Enabled = True
        tbModule.BackColor = vbWhite
        tbPressureAngle.BackColor = vbWhite
        If TypeName(Me.Controls("lblGearTrainInfo")) = "Label" Then
            Me.Controls("lblGearTrainInfo").Visible = False
        End If
    End If
    ValidateInputs
End Sub
