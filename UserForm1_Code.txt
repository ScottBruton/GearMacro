' UserForm1 Code Module
' Copy into the UserForm1 code module in your VBA project.
'
' FORM SHOW MODE: main() uses Show vbModeless + DoEvents loop until form closes (required for SW body pick).
' SW UserForms have no ShowModal property — modeless is only vbModeless in code.
'
' ADD THESE CONTROLS IN THE VBA FORM DESIGNER (names must match):
'   optDrivingGear   OptionButton  Caption "Driving gear"   GroupName = "GearRole"
'   optDrivenGear    OptionButton  Caption "Driven gear"    GroupName = "GearRole"
'   fraDrivenGear    Frame         Caption "Driven gear layout"
'   lblDrivingGear   Label         Caption "Driving gear:"
'   btnCancel / cbCancel / btnClose — call CloseGearForm (ends main() DoEvents loop)
'   cboDrivingGearBody ComboBox    Style=2 — macro gears from all open parts
'                                  Bodies named ""teethT pitchDia"" (e.g. 9T 11.25) or InvoluteGear custom properties
'   tbDrivingGearBody TextBox      Optional summary; Locked=True
'   btnRefreshDrivingGear CommandButton Caption "Refresh list" (or keep btnPickDrivingBody name)
'   lblMeshAngle     Label         Caption "Mesh angle (deg, 0=inline):"
'   tbMeshAngle      TextBox       default "0"
'   lblTrainCenter   Label         Caption "Centre distance:"
'   tbTrainCenter    TextBox       Locked=True (read-only, shows C = m(z1+z2)/2)
'   cboMfgMethod     ComboBox      Style=2 (dropdown list) — items filled in code, not RowSource
'   tbCenterClearance, tbBacklash, tbTrainCenterAdjusted
'
' Existing controls: tbPressureAngle, tbModule, tbTeeth, tbWidth, cbGenerate, etc.

Private Sub UserForm_Initialize()
    On Error Resume Next
    tbPressureAngle.Text = "20"
    tbModule.Text = "1.25"
    tbTeeth.Text = "9"
    tbWidth.Text = "10"

    If ControlExists("optDrivingGear") Then optDrivingGear.Value = True
    If ControlExists("tbMeshAngle") Then tbMeshAngle.Text = "0"

    InitMfgMethodCombo

    UpdateGearRoleUI
    On Error GoTo 0

    On Error Resume Next
    ValidateInputs
    UpdateGearCalculations
    On Error GoTo 0
End Sub

Private Function ControlExists(ByVal ctrlName As String) As Boolean
    ControlExists = Not (FindControlByName(Me, ctrlName) Is Nothing)
End Function

' Controls inside a Frame are not on Me.Controls — use FindControlByName for get/set/visible
Private Sub SetFormTextBox(ByVal ctrlName As String, ByVal value As String)
    Dim ctrl As Object
    Set ctrl = FindControlByName(Me, ctrlName)
    If ctrl Is Nothing Then Exit Sub
    On Error Resume Next
    ctrl.Text = value
    On Error GoTo 0
End Sub

Private Function GetFormTextBox(ByVal ctrlName As String) As String
    Dim ctrl As Object
    GetFormTextBox = ""
    Set ctrl = FindControlByName(Me, ctrlName)
    If ctrl Is Nothing Then Exit Function
    On Error Resume Next
    GetFormTextBox = ctrl.Text
    On Error GoTo 0
End Function

Private Sub SetFormControlVisible(ByVal ctrlName As String, ByVal visible As Boolean)
    Dim ctrl As Object
    Set ctrl = FindControlByName(Me, ctrlName)
    If ctrl Is Nothing Then Exit Sub
    On Error Resume Next
    ctrl.Visible = visible
    On Error GoTo 0
End Sub

Private Function FindControlByName(ByVal parent As Object, ByVal ctrlName As String) As Object
    Dim c As Object
    On Error Resume Next
    Set FindControlByName = parent.Controls(ctrlName)
    If Not FindControlByName Is Nothing Then Exit Function
    On Error GoTo 0
    For Each c In parent.Controls
        If TypeName(c) = "Frame" Then
            Set FindControlByName = FindControlByName(c, ctrlName)
            If Not FindControlByName Is Nothing Then Exit Function
        End If
    Next c
End Function

' ComboBox on a SW UserForm: RowSource does not work — use AddItem in Initialize.
' Control (Name) must be cboMfgMethod, or left as ComboBox1 (fallback below).
Private Function GetMfgMethodCombo() As Object
    Dim names As Variant
    Dim i As Long
    Dim c As Object
    names = Array("cboMfgMethod", "ComboBox1", "ComboBox2")
    For i = LBound(names) To UBound(names)
        Set GetMfgMethodCombo = FindControlByName(Me, CStr(names(i)))
        If Not GetMfgMethodCombo Is Nothing Then Exit Function
    Next i
    For Each c In AllComboBoxesOnForm(Me)
        Set GetMfgMethodCombo = c
        Exit Function
    Next c
End Function

Private Function AllComboBoxesOnForm(ByVal parent As Object) As Collection
    Dim result As New Collection
    Dim c As Object
    Dim subC As Object
    For Each c In parent.Controls
        If TypeName(c) = "ComboBox" Then result.Add c
        If TypeName(c) = "Frame" Then
            For Each subC In AllComboBoxesOnForm(c)
                result.Add subC
            Next subC
        End If
    Next c
    Set AllComboBoxesOnForm = result
End Function

Private Sub InitMfgMethodCombo()
    Dim cb As Object
    On Error Resume Next
    Set cb = GetMfgMethodCombo()
    If cb Is Nothing Then Exit Sub
    If cb.ListCount > 0 Then Exit Sub
    With cb
        .Clear
        .Style = 2
        .AddItem "Nominal (CAD)"
        .AddItem "3D print"
        .AddItem "CNC"
        .AddItem "OTS (catalog)"
        .ListIndex = 0
    End With
    ApplyMfgMethodDefaults
    On Error GoTo 0
End Sub

Private Sub UserForm_Activate()
    InitMfgMethodCombo
    On Error Resume Next

    On Error GoTo 0
End Sub

Private Sub ApplyMfgMethodDefaults()
    Dim cb As Object
    On Error Resume Next
    Set cb = GetMfgMethodCombo()
    If cb Is Nothing Then Exit Sub
    Select Case cb.ListIndex
        Case 1
            SetFormTextBox "tbCenterClearance", "0.15"
            SetFormTextBox "tbBacklash", "0.10"
        Case 2
            SetFormTextBox "tbCenterClearance", "0.03"
            SetFormTextBox "tbBacklash", "0"
        Case 0, 3
            SetFormTextBox "tbCenterClearance", "0"
            SetFormTextBox "tbBacklash", "0"
    End Select
    UpdateTrainCenterDistance
    On Error GoTo 0
End Sub

Private Sub UpdateGearRoleUI()
    Dim showDriven As Boolean
    On Error Resume Next

    showDriven = False
    If ControlExists("optDrivenGear") Then showDriven = optDrivenGear.Value

    SetFormControlVisible "fraDrivenGear", showDriven
    SetFormControlVisible "lblDrivingGear", showDriven
    SetFormControlVisible "cboDrivingGearBody", showDriven
    SetFormControlVisible "tbDrivingGearBody", showDriven
    SetFormControlVisible "btnPickDrivingBody", showDriven
    SetFormControlVisible "btnRefreshDrivingGear", showDriven
    SetFormControlVisible "lblMeshAngle", showDriven
    SetFormControlVisible "tbMeshAngle", showDriven
    SetFormControlVisible "lblTrainCenter", showDriven
    SetFormControlVisible "tbTrainCenter", showDriven

    If showDriven Then
        tbModule.Enabled = False
        tbPressureAngle.Enabled = False
        tbModule.BackColor = RGB(240, 240, 240)
        tbPressureAngle.BackColor = RGB(240, 240, 240)
        RefreshDrivingGearBodyCombo
        UpdateDrivingGearDisplay
        UpdateTrainCenterDistance
    Else
        tbModule.Enabled = True
        tbPressureAngle.Enabled = True
        tbModule.BackColor = vbWhite
        tbPressureAngle.BackColor = vbWhite
    End If
    On Error GoTo 0
End Sub

Private Function GetDrivingGearBodyCombo() As Object
    Set GetDrivingGearBodyCombo = FindControlByName(Me, "cboDrivingGearBody")
End Function

Private Sub RefreshDrivingGearBodyCombo()
    Dim cb As Object
    Dim i As Long
    Dim n As Long
    Dim prevTitle As String
    On Error Resume Next
    Set cb = GetDrivingGearBodyCombo()
    If cb Is Nothing Then Exit Sub
    prevTitle = GearTrainDrivingPartTitle
    n = RefreshDrivingGearBodyComboList()
    cb.Clear
    cb.Style = 2
    If n < 1 Then
        cb.AddItem "(no solid bodies — open the driving gear part)"
        cb.ListIndex = 0
        GearTrainDrivingPartTitle = ""
        Exit Sub
    End If
    For i = 0 To n - 1
        cb.AddItem DrivingGearComboItemCaption(i)
    Next i
    cb.ListIndex = -1
    For i = 0 To n - 1
        If GearTrainDrivingComboPartTitle(i) = prevTitle And prevTitle <> "" Then
            cb.ListIndex = i
            Exit For
        End If
    Next i
    If cb.ListIndex < 0 And n = 1 Then
        cb.ListIndex = 0
        ApplyDrivingGearFromComboSelection
    End If
    On Error GoTo 0
End Sub

Private Sub ApplyDrivingGearFromComboSelection()
    Dim cb As Object
    On Error Resume Next
    Set cb = GetDrivingGearBodyCombo()
    If cb Is Nothing Then Exit Sub
    If cb.ListIndex < 0 Then
        GearTrainDrivingPartTitle = ""
        UpdateDrivingGearDisplay
        Exit Sub
    End If
    If ApplyDrivingGearFromComboIndex(cb.ListIndex) Then
        UpdateDrivingGearDisplay
        UpdateTrainCenterDistance
        UpdateGearCalculations
    Else
        GearTrainDrivingPartTitle = ""
        UpdateDrivingGearDisplay
        SetFormTextBox "tbDrivingGearBody", "(choose a macro-built driving gear from the list)"
        UpdateTrainCenterDistance
    End If
    On Error GoTo 0
End Sub

Private Sub UpdateDrivingGearDisplay()
    On Error Resume Next
    If ControlExists("tbDrivingGearBody") Then
        If GearTrainDrivingPartTitle <> "" Then
            tbDrivingGearBody.Text = GearTrainDrivingPartTitle & "  (z=" & GearTrainDrivingTeeth & _
                ", m=" & Format(GearTrainDrivingModuleMM, "0.###") & ", " & GearTrainDrivingBodyLabel & ")"
        Else
            tbDrivingGearBody.Text = "(select driving gear from the list above)"
        End If
    End If
    On Error GoTo 0
End Sub

Private Sub UpdateTrainCenterDistance()
    Dim c As Double
    Dim z2 As Long
    Dim teethText As String
    On Error Resume Next
    If Not ControlExists("tbTrainCenter") Then Exit Sub

    If GearTrainDrivingTeeth <= 0 Or GearTrainDrivingModuleMM <= 0 Then
        SetFormTextBox "tbTrainCenter", "(select a macro-built driving gear)"
        SetFormTextBox "tbTrainCenterAdjusted", ""
        Exit Sub
    End If

    teethText = GetFormTextBox("tbTeeth")
    If teethText = "" Then teethText = Trim$(tbTeeth.Text)
    If teethText = "" Then
        SetFormTextBox "tbTrainCenter", "(enter number of teeth for driven gear)"
        SetFormTextBox "tbTrainCenterAdjusted", ""
        Exit Sub
    End If

    z2 = CLng(teethText)
    c = CenterDistanceMM(GearTrainDrivingModuleMM, GearTrainDrivingTeeth, z2)
    SetFormTextBox "tbTrainCenter", Format(c, "0.###") & " mm"
    GearTrainCenterDistanceMM = c
    UpdateAdjustedCenterDistance c
    On Error GoTo 0
End Sub

Private Sub UpdateAdjustedCenterDistance(ByVal cNominal As Double)
    Dim deltaC As Double
    Dim clrText As String
    On Error Resume Next
    If Not ControlExists("tbTrainCenterAdjusted") Then Exit Sub
    deltaC = 0#
    clrText = GetFormTextBox("tbCenterClearance")
    If clrText <> "" Then deltaC = CDbl(clrText)
    SetFormTextBox "tbTrainCenterAdjusted", Format(cNominal + deltaC, "0.###") & " mm"
    On Error GoTo 0
End Sub

Private Sub cbGenerate_Click()
    Dim PressureAngle As Double
    Dim Module As Double
    Dim NumTeeth As Integer
    Dim Width As Double
    Dim ErrorMsg As String

    If ControlExists("optDrivenGear") And optDrivenGear.Value Then
        GearTrainRole = GEAR_ROLE_DRIVEN
        If GearTrainDrivingPartTitle = "" Then
            ApplyDrivingGearFromComboSelection
        End If
        If GearTrainDrivingPartTitle = "" Then
            MsgBox "Choose the driving gear from the list (open that part first, then Refresh list).", vbExclamation
            Exit Sub
        End If
        On Error Resume Next
        GearTrainMeshAngleDeg = CDbl(tbMeshAngle.Text)
        On Error GoTo 0
    Else
        GearTrainRole = GEAR_ROLE_DRIVING
    End If

    ' --- validation (same as before) ---
    ErrorMsg = ""
    If Trim(tbPressureAngle.Text) = "" Then
        ErrorMsg = ErrorMsg & "Pressure Angle is required." & vbCrLf
    Else
        On Error Resume Next
        PressureAngle = CDbl(tbPressureAngle.Text)
        If Err.Number <> 0 Or PressureAngle <= 0 Then ErrorMsg = ErrorMsg & "Pressure Angle must be a positive number." & vbCrLf
        On Error GoTo 0
    End If

    If Trim(tbModule.Text) = "" Then
        ErrorMsg = ErrorMsg & "Module is required." & vbCrLf
    Else
        On Error Resume Next
        Module = CDbl(tbModule.Text)
        If Err.Number <> 0 Or Module <= 0 Then ErrorMsg = ErrorMsg & "Module must be a positive number." & vbCrLf
        On Error GoTo 0
    End If

    If Trim(tbTeeth.Text) = "" Then
        ErrorMsg = ErrorMsg & "Number of Teeth is required." & vbCrLf
    Else
        On Error Resume Next
        NumTeeth = CInt(tbTeeth.Text)
        If Err.Number <> 0 Or NumTeeth <= 0 Then
            ErrorMsg = ErrorMsg & "Number of Teeth must be a positive integer." & vbCrLf
        End If
        On Error GoTo 0
    End If

    If Trim(tbWidth.Text) = "" Then
        ErrorMsg = ErrorMsg & "Width is required." & vbCrLf
    Else
        On Error Resume Next
        Width = CDbl(tbWidth.Text)
        If Err.Number <> 0 Or Width <= 0 Then ErrorMsg = ErrorMsg & "Width must be a positive number." & vbCrLf
        On Error GoTo 0
    End If

    If ErrorMsg <> "" Then
        MsgBox "Please correct the following errors:" & vbCrLf & vbCrLf & ErrorMsg, vbExclamation, "Validation Error"
        Exit Sub
    End If

    Me.Hide
    Call GenerateGear(PressureAngle, Module, NumTeeth, Width)
    ' Modeless: show form again after generate (do not Unload)
    Me.Show vbModeless
End Sub

Private Sub ValidateInputs()
    On Error Resume Next
    Dim PressureAngle As Double
    Dim Module As Double
    Dim NumTeeth As Integer
    Dim Width As Double
    Dim MinTeeth As Double
    Dim Pi As Double
    Dim PressureAngleRad As Double
    Dim WarningMsg As String
    Dim PitchDiameter As Double

    Pi = 3.14159265358979
    If Not tbPressureAngle Is Nothing Then tbPressureAngle.BackColor = vbWhite
    If Not tbModule Is Nothing Then tbModule.BackColor = vbWhite
    If Not tbTeeth Is Nothing Then tbTeeth.BackColor = vbWhite
    If Not tbWidth Is Nothing Then tbWidth.BackColor = vbWhite

    If Not tbTeeth Is Nothing And Not tbPressureAngle Is Nothing And Trim(tbTeeth.Text) <> "" And Trim(tbPressureAngle.Text) <> "" Then
        NumTeeth = CInt(tbTeeth.Text)
        PressureAngle = CDbl(tbPressureAngle.Text)
        If NumTeeth > 0 And PressureAngle > 0 Then
            PressureAngleRad = PressureAngle * Pi / 180
            MinTeeth = 2 / (Sin(PressureAngleRad) ^ 2)
            If NumTeeth < MinTeeth Then tbTeeth.BackColor = RGB(255, 200, 200)
        End If
    End If

    UpdateGearCalculations
    UpdateTrainCenterDistance
End Sub

Private Sub UpdateGearCalculations()
    Dim Module As Double
    Dim NumTeeth As Integer
    Dim PressureAngle As Double
    Dim PitchDiameter As Double
    Dim BaseDiameter As Double
    Dim AddendumDiameter As Double
    Dim DedendumDiameter As Double
    Dim Pi As Double
    Dim CenterDistance As Double
    Dim Gear2Teeth As Integer
    Dim InfoText As String

    Pi = 3.14159265358979
    InfoText = ""

    If Not tbModule Is Nothing And Not tbTeeth Is Nothing And Trim(tbModule.Text) <> "" And Trim(tbTeeth.Text) <> "" Then
        On Error Resume Next
        Module = CDbl(tbModule.Text)
        NumTeeth = CInt(tbTeeth.Text)
        On Error GoTo 0
        If Module > 0 And NumTeeth > 0 Then
            PitchDiameter = Module * NumTeeth
            InfoText = "Pitch Diameter: " & Format(PitchDiameter, "0.000") & " mm"
            UpdateLabel "lblPitchDiameter", Format(PitchDiameter, "0.000") & " mm"
            AddendumDiameter = PitchDiameter + 2 * Module
            DedendumDiameter = PitchDiameter - 2.5 * Module
            UpdateLabel "lblAddendumDiameter", Format(AddendumDiameter, "0.000") & " mm"
            UpdateLabel "lblDedendumDiameter", Format(DedendumDiameter, "0.000") & " mm"
            If Trim(tbPressureAngle.Text) <> "" Then
                PressureAngle = CDbl(tbPressureAngle.Text)
                BaseDiameter = PitchDiameter * Cos(PressureAngle * Pi / 180)
                UpdateLabel "lblBaseDiameter", Format(BaseDiameter, "0.000") & " mm"
            End If
            If ControlExists("optDrivenGear") And optDrivenGear.Value And GearTrainDrivingTeeth > 0 Then
                CenterDistance = CenterDistanceMM(GearTrainDrivingModuleMM, GearTrainDrivingTeeth, NumTeeth)
                UpdateLabel "lblCenterDistance", Format(CenterDistance, "0.000") & " mm"
                UpdateTextBox "tbTrainCenter", Format(CenterDistance, "0.000") & " mm"
                UpdateAdjustedCenterDistance CenterDistance
            End If
            UpdateLabel "lblGearInfo", InfoText
        End If
    End If
End Sub
