Public Function FormatVBError(Optional ByVal context As String = "") As String
    Dim s As String
    If Err.Number = 0 Then
        FormatVBError = ""
        Exit Function
    End If
    s = "Err " & Err.Number & ": " & Err.Description
    If context <> "" Then s = context & " — " & s
    FormatVBError = s
End Function

Public Function FormatSWFeatureError(ByVal swFeat As SldWorks.Feature, Optional ByVal context As String = "") As String
    Dim errCode As Long
    Dim s As String
    If swFeat Is Nothing Then
        FormatSWFeatureError = context
        Exit Function
    End If
    On Error Resume Next
    errCode = swFeat.GetErrorCode2(True)
    If errCode <> 0 Then
        s = swFeat.name & " feature error code=" & errCode
        If context <> "" Then s = context & " — " & s
        FormatSWFeatureError = s
    ElseIf context <> "" Then
        FormatSWFeatureError = context
    End If
    On Error GoTo 0
End Function

    If errLine = "" Then errLine = gLastGearBuildError
    If errLine = "" Then errLine = FormatVBError()

Public Sub ClearSWSelection()
    Dim i As Long
    Dim n As Long
    Dim selObj As Object

    On Error Resume Next
    If swDoc Is Nothing Then Exit Sub
    Set swSelMgr = swDoc.SelectionManager

    swDoc.ClearSelection2 True
    n = swSelMgr.GetSelectedObjectCount2(-1)
    For i = n To 1 Step -1
        Set selObj = swSelMgr.GetSelectedObject6(i, -1)
        If Not selObj Is Nothing Then selObj.Select2 False, -1
    Next i
    swDoc.ClearSelection2 True
    On Error GoTo 0
End Sub

Public Function Atn2(ByVal y As Double, ByVal x As Double) As Double
    If x > 0# Then
        Atn2 = Atn(y / x)
    ElseIf x < 0# Then
        Atn2 = Atn(y / x) + Pi
    ElseIf y > 0# Then
        Atn2 = Pi / 2#
    ElseIf y < 0# Then
        Atn2 = -Pi / 2#
    Else
        Atn2 = 0#
    End If
End Function

Public Sub CaptureSolidBodyCountBeforeGearBuild()
    Dim vBodies As Variant
    On Error Resume Next
    gGearSolidBodyCountBefore = 0
    If swPart Is Nothing Then Exit Sub
    vBodies = swPart.GetBodies2(swSolidBody, False)
    If IsEmpty(vBodies) Then Exit Sub
    If IsArray(vBodies) Then gGearSolidBodyCountBefore = UBound(vBodies) - LBound(vBodies) + 1
    On Error GoTo 0
End Sub

Private Function HasGearBodyNamePrefix(ByVal bodyName As String) As Boolean
    Dim i As Long
    Dim ch As String
    HasGearBodyNamePrefix = False
    bodyName = Trim$(bodyName)
    If Len(bodyName) < 6 Then Exit Function
    If UCase$(Left$(bodyName, 4)) <> "GEAR" Then Exit Function
    i = 5
    Do While i <= Len(bodyName)
        ch = Mid$(bodyName, i, 1)
        If ch >= "0" And ch <= "9" Then
            i = i + 1
        Else
            Exit Do
        End If
    Loop
    HasGearBodyNamePrefix = (i > 5 And Mid$(bodyName, i, 1) = "_")
End Function

Private Function StripGearBodyNamePrefix(ByVal bodyName As String) As String
    Dim i As Long
    Dim ch As String
    StripGearBodyNamePrefix = Trim$(bodyName)
    If Not HasGearBodyNamePrefix(StripGearBodyNamePrefix) Then Exit Function
    i = 5
    Do While i <= Len(StripGearBodyNamePrefix)
        ch = Mid$(StripGearBodyNamePrefix, i, 1)
        If ch >= "0" And ch <= "9" Then
            i = i + 1
        Else
            Exit Do
        End If
    Loop
    If Mid$(StripGearBodyNamePrefix, i, 1) = "_" Then
        StripGearBodyNamePrefix = Trim$(Mid$(StripGearBodyNamePrefix, i + 1))
    End If
End Function

Private Function CountMacroGearsInPart() As Long
    Dim vBodies As Variant
    Dim swBody As SldWorks.Body2
    Dim i As Long
    Dim pitchDia As Double
    Dim z As Long

    CountMacroGearsInPart = 0
    If swPart Is Nothing Then Exit Function
    vBodies = swPart.GetBodies2(swBodyType_e.swSolidBody, True)
    If Not IsArray(vBodies) Then Exit Function
    For i = LBound(vBodies) To UBound(vBodies)
        Set swBody = vBodies(i)
        If IsMacroGearBody(swBody, swDoc) Then
            CountMacroGearsInPart = CountMacroGearsInPart + 1
        ElseIf HasGearBodyNamePrefix(swBody.name) Then
            CountMacroGearsInPart = CountMacroGearsInPart + 1
        ElseIf TryParseGearBodyName(swBody.name, pitchDia, z) Then
            CountMacroGearsInPart = CountMacroGearsInPart + 1
        End If
    Next i
End Function

' Solid body from the current addendum boss (Face2.GetBody â€” reliable; GetBodies2 order is not creation order)
Public Function GetCurrentGearSolidBody() As SldWorks.Body2
    Dim swFace As SldWorks.Face2
    Dim vBodies As Variant
    Dim idx As Long

    On Error Resume Next
    If Not swAddendumExtrudeFeat Is Nothing Then
        Set swFace = FindAddendumCylindricalFace()
        If Not swFace Is Nothing Then
            Set GetCurrentGearSolidBody = swFace.GetBody
            If Not GetCurrentGearSolidBody Is Nothing Then Exit Function
        End If
    End If

    If swPart Is Nothing Then Exit Function
    vBodies = swPart.GetBodies2(swSolidBody, False)
    If Not IsArray(vBodies) Then Exit Function
    idx = UBound(vBodies)
    Set GetCurrentGearSolidBody = vBodies(idx)
    On Error GoTo 0
End Function

Public Function ReadGearInputsFromForm() As Boolean
    On Error GoTo ReadFail

    GearGeom.pressureAngleDeg = CDbl(UserForm1.tbPressureAngle.Text)
    GearGeom.moduleMM = CDbl(UserForm1.tbModule.Text)
    GearGeom.numTeeth = CLng(UserForm1.tbTeeth.Text)
    GearGeom.widthMM = CDbl(UserForm1.tbWidth.Text)
    GearGeom.HubDiameterMM = GetFormDoubleOrDefault("tbHubDia", 0#)
    GearGeom.ShaftDiameterMM = GetFormDoubleOrDefault("tbShaftDia", 0#)

    ReadGearInputsFromForm = ValidateGearInputsPublic(GearGeom.pressureAngleDeg, GearGeom.moduleMM, GearGeom.numTeeth, GearGeom.widthMM)
    If ReadGearInputsFromForm Then
        CalculateGearGeometry
        DebugPrint "ReadGearInputsFromForm OK"
    End If
    Exit Function

ReadFail:
    MsgBox "Could not read form inputs: " & Err.Description, vbCritical
    ReadGearInputsFromForm = False
End Function

Public Function ValidateGearInputsPublic(ByVal pressureAngleDeg As Double, ByVal moduleMM As Double, ByVal numTeeth As Long, ByVal widthMM As Double) As Boolean
    Dim errMsg As String
    Dim minTeeth As Double
    Dim alphaRad As Double
    errMsg = ""

    If pressureAngleDeg <= 0# Or pressureAngleDeg >= 90# Then errMsg = errMsg & "Invalid pressure angle." & vbCrLf
    If moduleMM <= 0# Then errMsg = errMsg & "Invalid module." & vbCrLf
    If numTeeth < 3 Then errMsg = errMsg & "Tooth count must be >= 3." & vbCrLf
    If widthMM <= 0# Then errMsg = errMsg & "Invalid width." & vbCrLf

    If GearGeom.IsInternal Then
        Dim tipDia As Double
        Dim rootDia As Double
        alphaRad = pressureAngleDeg * 3.14159265358979 / 180#
        minTeeth = 2# / (Sin(alphaRad) ^ 2)
        If numTeeth < minTeeth + 1 Then
            errMsg = errMsg & "Internal gear needs more teeth (min ~" & Format(minTeeth + 1, "0") & " at this pressure angle)." & vbCrLf
        End If
        tipDia = moduleMM * numTeeth - 2# * moduleMM
        rootDia = moduleMM * numTeeth + 2.5 * moduleMM
        If tipDia <= 0# Or rootDia <= tipDia Then
            errMsg = errMsg & "Internal gear: tip diameter must be less than root diameter (increase teeth or module)." & vbCrLf
        End If
    End If

    If errMsg <> "" Then
        DebugPrint "ValidateGearInputsPublic FAIL: " & Replace(Trim$(errMsg), vbCrLf, " | ")
        MsgBox errMsg, vbExclamation, "Invalid Gear Inputs"
        ValidateGearInputsPublic = False
    Else
        ValidateGearInputsPublic = True
    End If
End Function

Public Sub CalculateGearGeometry()
    Dim alphaRad As Double
    Dim z As Double

    Pi = 3.14159265358979
    z = GearGeom.numTeeth
    alphaRad = GearGeom.pressureAngleDeg * Pi / 180#

    GearGeom.pitchDiameterMM = GearGeom.moduleMM * z
    GearGeom.PitchRadiusMM = GearGeom.pitchDiameterMM / 2#
    GearGeom.BaseDiameterMM = GearGeom.pitchDiameterMM * Cos(alphaRad)
    GearGeom.BaseRadiusMM = GearGeom.BaseDiameterMM / 2#

    If GearGeom.IsInternal Then
        GearGeom.AddendumDiameterMM = GearGeom.pitchDiameterMM - 2# * GearGeom.moduleMM
        GearGeom.DedendumDiameterMM = GearGeom.pitchDiameterMM + 2.5 * GearGeom.moduleMM
    Else
        GearGeom.AddendumDiameterMM = GearGeom.pitchDiameterMM + 2# * GearGeom.moduleMM
        GearGeom.DedendumDiameterMM = GearGeom.pitchDiameterMM - 2.5 * GearGeom.moduleMM
    End If
    GearGeom.AddendumRadiusMM = GearGeom.AddendumDiameterMM / 2#
    GearGeom.DedendumRadiusMM = GearGeom.DedendumDiameterMM / 2#

    If GearGeom.HubDiameterMM <= 0# Then GearGeom.HubDiameterMM = GearGeom.pitchDiameterMM * 0.4
    If GearGeom.ShaftDiameterMM <= 0# Then GearGeom.ShaftDiameterMM = 10#
    If GearGeom.HubDiameterMM <= GearGeom.ShaftDiameterMM Then
        GearGeom.HubDiameterMM = GearGeom.ShaftDiameterMM + 2# * GearGeom.moduleMM
    End If
    If GearGeom.HubDiameterMM >= GearGeom.DedendumDiameterMM Then
        GearGeom.HubDiameterMM = GearGeom.DedendumDiameterMM - 2# * GearGeom.moduleMM
    End If

    GearGeom.HubRadiusMM = GearGeom.HubDiameterMM / 2#
    GearGeom.ShaftRadiusMM = GearGeom.ShaftDiameterMM / 2#

    GearGeom.RimWidthMM = GearGeom.widthMM / 2#
    GearGeom.HubWidthMM = GearGeom.widthMM / 2#

    GearGeom.CircularPitchMM = Pi * GearGeom.moduleMM
    GearGeom.AngularPitchDeg = 360# / z

    GearGeom.GapArcLengthMM = Pi * GearGeom.pitchDiameterMM / z
    GearGeom.GapHalfArcLengthMM = GearGeom.GapArcLengthMM / 2#
    GearGeom.GapRootFilletMM = GearGeom.GapArcLengthMM * GEAR_MCM_ROOT_FILLET_RATIO

    GearGeom.PitchRadiusM = GearGeom.PitchRadiusMM / MM_PER_M
    GearGeom.BaseRadiusM = GearGeom.BaseRadiusMM / MM_PER_M
    GearGeom.AddendumRadiusM = GearGeom.AddendumRadiusMM / MM_PER_M
    GearGeom.DedendumRadiusM = GearGeom.DedendumRadiusMM / MM_PER_M
    GearGeom.WidthM = GearGeom.widthMM / MM_PER_M
    GearGeom.HubRadiusM = GearGeom.HubRadiusMM / MM_PER_M
    GearGeom.ShaftRadiusM = GearGeom.ShaftRadiusMM / MM_PER_M
    GearGeom.RimWidthM = GearGeom.RimWidthMM / MM_PER_M
    GearGeom.HubWidthM = GearGeom.HubWidthMM / MM_PER_M

    DebugPrint "CalculateGearGeometry: " & IIf(GearGeom.IsInternal, "INTERNAL", "EXTERNAL") & _
        " Dp=" & Format(GearGeom.pitchDiameterMM, "0.###") & _
        " Da=" & Format(GearGeom.AddendumDiameterMM, "0.###") & _
        " Dd=" & Format(GearGeom.DedendumDiameterMM, "0.###") & _
        " Hub=" & Format(GearGeom.HubDiameterMM, "0.###") & _
        " Shaft=" & Format(GearGeom.ShaftDiameterMM, "0.###") & _
        " gapArc=" & Format(GearGeom.GapArcLengthMM, "0.####") & " mm"
End Sub

Private Function GetRefPlaneByIndex(ByVal planeIndex As Long) As SldWorks.Feature
    Dim ft As SldWorks.Feature
    Dim n As Long
    n = 0
    Set ft = swPart.FirstFeature
    Do While Not ft Is Nothing
        If ft.GetTypeName2 = "RefPlane" Then
            n = n + 1
            If n = planeIndex Then
                Set GetRefPlaneByIndex = ft
                Exit Function
            End If
        End If
        Set ft = ft.GetNextFeature
    Loop
End Function

Private Function SelectSWPlaneByName(ByVal planeName As String, Optional ByVal append As Boolean = False) As Boolean
    SelectSWPlaneByName = swModelDocExt.SelectByID2(planeName, "PLANE", 0, 0, 0, append, 0, Nothing, 0)
End Function

' Resolve front plane from feature tree (1st RefPlane = Front in default templates; same as original macro)
Private Function ResolveAndStoreFrontPlane() As Boolean
    Dim desc As String

    Set swFrontPlaneFeat = GetRefPlaneByIndex(1)
    If swFrontPlaneFeat Is Nothing Then
        DebugPrint "ResolveAndStoreFrontPlane: no RefPlane in part"
        ResolveAndStoreFrontPlane = False
        Exit Function
    End If

    gFrontPlaneName = swFrontPlaneFeat.name
    On Error Resume Next
    desc = swFrontPlaneFeat.Description
    On Error GoTo 0
    If desc <> "" Then gFrontPlaneName = desc

    ResolveAndStoreFrontPlane = True
    DebugPrint "Front plane resolved: " & gFrontPlaneName & " (feature " & swFrontPlaneFeat.name & ")"
End Function

' Call Feature.Select (Sub in VBA - no reliable return value)
Private Sub SelectPlaneFeature(ByVal planeFeat As SldWorks.Feature, Optional ByVal append As Boolean = True)
    If planeFeat Is Nothing Then Exit Sub
    On Error Resume Next
    planeFeat.Select append
    On Error GoTo 0
End Sub

Private Function SelectFrontPlaneFeature(Optional ByVal append As Boolean = False) As Boolean
    If swFrontPlaneFeat Is Nothing Then
        If Not ResolveAndStoreFrontPlane() Then
            SelectFrontPlaneFeature = False
            Exit Function
        End If
    End If
    If Not append Then ClearSWSelection
    SelectPlaneFeature swFrontPlaneFeat, append
    If swSelMgr.GetSelectedObjectCount2(-1) < 1 Then
        SelectFrontPlaneFeature = SelectSWPlaneByName(gFrontPlaneName, append)
        If Not SelectFrontPlaneFeature Then SelectFrontPlaneFeature = SelectSWPlaneByName("Front Plane", append)
    Else
        SelectFrontPlaneFeature = True
    End If
End Function

' Start sketch on front plane - original: RefPlaneFt(1).Select True, InsertSketch True
' NOTE: Feature.Select is a Sub in SW VBA - do NOT test its return value (always False wrongly)
Public Function BeginSketchOnFrontPlane() As Boolean
    Dim selCount As Long

    BeginSketchOnFrontPlane = False
    ExitActiveSketchIfAny
    Set swFrontPlaneFeat = GetRefPlaneByIndex(1)
    If swFrontPlaneFeat Is Nothing Then
        DebugPrint "BeginSketchOnFrontPlane: no front RefPlane"
        Exit Function
    End If
    gFrontPlaneName = swFrontPlaneFeat.name

    On Error Resume Next
    swFrontPlaneFeat.Select True
    selCount = swSelMgr.GetSelectedObjectCount2(-1)
    If selCount < 1 Then
        swModelDocExt.SelectByID2 gFrontPlaneName, "PLANE", 0, 0, 0, False, 0, Nothing, 0
    End If
    If swSelMgr.GetSelectedObjectCount2(-1) < 1 Then
        swModelDocExt.SelectByID2 "Front Plane", "PLANE", 0, 0, 0, False, 0, Nothing, 0
    End If
    On Error GoTo 0

    swSketchManager.InsertSketch True
    If swSketchManager.ActiveSketch Is Nothing Then
        DebugPrint "BeginSketchOnFrontPlane: InsertSketch did not open a sketch"
        Exit Function
    End If

    BeginSketchOnFrontPlane = True

' Anchor a sketch point on the gear axis (Origin for first driving gear; pierce ref axis otherwise)
Public Function EnsureAxisAnchorPointInActiveSketch() As Boolean
    Dim xA As Double
    Dim yA As Double
    Dim swPt As SldWorks.SketchPoint

    EnsureAxisAnchorPointInActiveSketch = False
    On Error Resume Next
    If Not GetGearAxisCenterOnSketchM(xA, yA) Then Exit Function

    Set swPt = FindAxisSketchPointInActiveSketch()
    If swPt Is Nothing Then
        Set swPt = swSketchManager.CreatePoint(xA, yA, 0)
    End If
    If swPt Is Nothing Then Exit Function

    If GearTrainRole = GEAR_ROLE_DRIVING And Abs(xA) < 0.0000000001 And Abs(yA) < 0.0000000001 Then
        ClearSWSelection
        boolstatus = swModelDocExt.SelectByID2("Point1@Origin", "EXTSKETCHPOINT", 0#, 0#, 0#, False, 0, Nothing, 0)
        If boolstatus Then
            swPt.Select4 True, Nothing
            swDoc.SketchAddConstraints "sgCOINCIDENT"
            EnsureAxisAnchorPointInActiveSketch = True
            DebugPrint "EnsureAxisAnchorPointInActiveSketch: coincident to Origin"
            ClearSWSelection
            On Error GoTo 0
            Exit Function
        End If
    End If

    If SelectGearRefAxisForSketchConvert() Then
        swPt.Select4 True, Nothing
        swDoc.SketchAddConstraints "sgPIERCE"
        EnsureAxisAnchorPointInActiveSketch = True
        DebugPrint "EnsureAxisAnchorPointInActiveSketch: pierce to ref axis"
    Else
        EnsureAxisAnchorPointInActiveSketch = True
        DebugPrint "EnsureAxisAnchorPointInActiveSketch: point at boss centre (mm)=" & _
            Format(xA * MM_PER_M, "0.###") & ", " & Format(yA * MM_PER_M, "0.###")
    End If
    ClearSWSelection
    On Error GoTo 0
End Function

Private Function FindAxisSketchPointInActiveSketch() As SldWorks.SketchPoint
    Dim swSk As SldWorks.Sketch
    Dim vPts As Variant
    Dim swPt As SldWorks.SketchPoint
    Dim xA As Double
    Dim yA As Double
    Dim i As Long
    Dim bestDist As Double
    Dim d As Double

    Set FindAxisSketchPointInActiveSketch = Nothing
    If Not GetGearAxisCenterOnSketchM(xA, yA) Then Exit Function
    Set swSk = swSketchManager.ActiveSketch
    If swSk Is Nothing Then Exit Function
    On Error Resume Next
    vPts = swSk.GetSketchPoints2
    If Not IsArray(vPts) Then Exit Function
    bestDist = 1E+30
    For i = LBound(vPts) To UBound(vPts)
        Set swPt = vPts(i)
        d = (swPt.x - xA) ^ 2 + (swPt.y - yA) ^ 2
        If d < bestDist Then
            bestDist = d
            Set FindAxisSketchPointInActiveSketch = swPt
        End If
    Next i
    On Error GoTo 0
End Function

' Lock a sketch point to the part Origin while editing a Front-plane sketch.
Public Function ConstrainSketchPointToOrigin(ByVal swCenterPt As SldWorks.SketchPoint) As Boolean
    ConstrainSketchPointToOrigin = False
    On Error Resume Next
    If swCenterPt Is Nothing Then Exit Function

    ' Method 1: Origin pierce point on the active plane (empty name @ 0,0,0).
    ClearSWSelection
    boolstatus = swModelDocExt.SelectByID2("", "EXTSKETCHPOINT", 0#, 0#, 0#, False, 0, Nothing, 0)
    If boolstatus Then boolstatus = swCenterPt.Select4(True, Nothing)
    If boolstatus Then
        swDoc.SketchAddConstraints "sgCOINCIDENT"
        If Err.Number = 0 Then
            ConstrainSketchPointToOrigin = True
            DebugPrint "ConstrainSketchPointToOrigin: coincident via plane origin @ (0,0,0)"
            ClearSWSelection
            Exit Function
        End If
        Err.Clear
    End If

    ' Method 2: Origin reference point in the tree.
    ClearSWSelection
    swCenterPt.Select4 False, Nothing
    boolstatus = swModelDocExt.SelectByID2("Point1@Origin", "EXTSKETCHPOINT", 0#, 0#, 0#, True, 0, Nothing, 0)
    If boolstatus Then
        swDoc.SketchAddConstraints "sgCOINCIDENT"
        If Err.Number = 0 Then
            ConstrainSketchPointToOrigin = True
            DebugPrint "ConstrainSketchPointToOrigin: coincident to Point1@Origin"
            ClearSWSelection
            Exit Function
        End If
        Err.Clear
    End If

    ' Method 3: snap to exact origin coords then re-try coincident.
    swCenterPt.x = 0#
    swCenterPt.y = 0#
    ClearSWSelection
    boolstatus = swModelDocExt.SelectByID2("", "EXTSKETCHPOINT", 0#, 0#, 0#, False, 0, Nothing, 0)
    If boolstatus Then boolstatus = swCenterPt.Select4(True, Nothing)
    If boolstatus Then
        swDoc.SketchAddConstraints "sgCOINCIDENT"
        ConstrainSketchPointToOrigin = (Err.Number = 0)
        If ConstrainSketchPointToOrigin Then DebugPrint "ConstrainSketchPointToOrigin: SetCoords(0,0) + coincident"
    End If
    ClearSWSelection
    On Error GoTo 0
End Function

' Select one side for ApplyCoincidentRelation. append=False clears first; append=True adds to selection.
Private Function CoincidentSelectSide(ByVal kind As Long, ByVal obj As Object, _
    ByVal xM As Double, ByVal yM As Double, ByVal sketchFeat As String, ByVal segName As String, _
    ByVal epIndex As Long, ByVal append As Boolean) As Boolean
    Dim swPt As SldWorks.SketchPoint
    Dim swSeg As SldWorks.SketchSegment
    Dim swFeat As SldWorks.Feature
    Dim swSketch As SldWorks.Sketch
    Dim vSegs As Variant
    Dim vSeg As Variant
    Dim swSkSeg As SldWorks.SketchSegment
    Dim swActiveSk As SldWorks.Sketch
    Dim vPts As Variant
    Dim vPt As Variant
    Dim xEp As Double
    Dim yEp As Double
    Dim zEp As Double
    Dim docTitle As String
    Dim selId As String

    CoincidentSelectSide = False
    On Error Resume Next

    Select Case kind
        Case COINC_SEL_SKETCH_POINT
            Set swPt = obj
            If swPt Is Nothing Then Exit Function
            CoincidentSelectSide = swPt.Select4(append, Nothing)

        Case COINC_SEL_SKETCH_SEGMENT
            Set swSeg = obj
            If swSeg Is Nothing Then Exit Function
            CoincidentSelectSide = swSeg.Select4(append, Nothing)

        Case COINC_SEL_SKETCH_XY
            If Not append Then ClearSWSelection
            CoincidentSelectSide = swModelDocExt.SelectByID2("", "SKETCHPOINT", xM, yM, 0#, append, 0, Nothing, 0)
            If Not CoincidentSelectSide Then
                CoincidentSelectSide = swModelDocExt.SelectByID2("", "SKETCHSEGMENT", xM, yM, 0#, append, 0, Nothing, 0)
            End If
            If Not CoincidentSelectSide And Not obj Is Nothing Then
                Set swSeg = obj
                CoincidentSelectSide = swSeg.Select4(append, Nothing)
            End If

        Case COINC_SEL_ARC_ENDPOINT
            Set swSeg = obj
            If swSeg Is Nothing Then Exit Function
            If epIndex = 0 Then
                swSeg.GetStartPoint2 xEp, yEp, zEp
            Else
                swSeg.GetEndPoint2 xEp, yEp, zEp
            End If
            If Not append Then ClearSWSelection
            CoincidentSelectSide = swModelDocExt.SelectByID2("", "SKETCHPOINT", xEp, yEp, 0#, append, 0, Nothing, 0)
            If Not CoincidentSelectSide Then
                CoincidentSelectSide = swModelDocExt.SelectByID2("", "SKETCHSEGMENT", xEp, yEp, 0#, append, 0, Nothing, 0)
            End If
            If Not CoincidentSelectSide Then
                CoincidentSelectSide = swSeg.Select4(append, Nothing)
            End If

        Case COINC_SEL_EXT_SEGMENT
            If sketchFeat = "" Or segName = "" Then Exit Function
            If Not append Then ClearSWSelection
            selId = segName & "@" & sketchFeat
            CoincidentSelectSide = swModelDocExt.SelectByID2(selId, "EXTSKETCHSEGMENT", xM, yM, 0#, append, 0, Nothing, 0)
            If Not CoincidentSelectSide Then
                docTitle = swDoc.GetTitle
                If docTitle <> "" Then
                    CoincidentSelectSide = swModelDocExt.SelectByID2(selId & "@" & docTitle, "EXTSKETCHSEGMENT", xM, yM, 0#, append, 0, Nothing, 0)
                End If
            End If

        Case COINC_SEL_SEG_IN_SKETCH
            If sketchFeat = "" Or segName = "" Then Exit Function
            Set swFeat = swPart.FeatureByName(sketchFeat)
            If swFeat Is Nothing Then Exit Function
            Set swSketch = swFeat.GetSpecificFeature2
            If swSketch Is Nothing Then Exit Function
            vSegs = swSketch.GetSketchSegments
            If IsEmpty(vSegs) Then Exit Function
            If Not append Then ClearSWSelection
            For Each vSeg In vSegs
                Set swSkSeg = vSeg
                If swSkSeg.GetName = segName Then
                    CoincidentSelectSide = swSkSeg.Select4(append, Nothing)
                    Exit For
                End If
            Next vSeg

        Case COINC_SEL_LINE_ENDPOINT
            Set swSeg = obj
            If swSeg Is Nothing Then Exit Function
            If epIndex = 0 Then
                swSeg.GetStartPoint2 xEp, yEp, zEp
            Else
                swSeg.GetEndPoint2 xEp, yEp, zEp
            End If
            If Not append Then ClearSWSelection
            Set swActiveSk = swSketchManager.ActiveSketch
            If Not swActiveSk Is Nothing Then
                vPts = swActiveSk.GetSketchPoints2
                If Not IsEmpty(vPts) Then
                    For Each vPt In vPts
                        Set swPt = vPt
                        If Abs(swPt.x - xEp) < 0.000000001 And Abs(swPt.y - yEp) < 0.000000001 Then
                            CoincidentSelectSide = swPt.Select4(append, Nothing)
                            GoTo LineEpDone
                        End If
                    Next vPt
                End If
            End If
            CoincidentSelectSide = swModelDocExt.SelectByID2("", "SKETCHPOINT", xEp, yEp, 0#, append, 0, Nothing, 0)
LineEpDone:

        Case Else
            CoincidentSelectSide = False
    End Select

    On Error GoTo 0
End Function

' One coincident constraint from two typed selections. Side A first, side B appended.
' Examples:
'   Point on line:  ApplyCoincidentRelation COINC_SEL_SKETCH_POINT, pt, , , , , , COINC_SEL_SKETCH_SEGMENT, line
'   Arc start / vertical @ pick: ApplyCoincidentRelation COINC_SEL_ARC_ENDPOINT, arc, , , , , 0, COINC_SEL_SKETCH_XY, mirrorLine, xPick, yPick
'   Arc end / layout pitch: ApplyCoincidentRelation COINC_SEL_ARC_ENDPOINT, arc, , , , , 1, COINC_SEL_EXT_SEGMENT, , xEp, yEp, "Gear1_Layout", "Arc2"
Public Function ApplyCoincidentRelation( _
    ByVal kindA As Long, Optional ByVal objA As Object = Nothing, _
    Optional ByVal xA As Double = 0, Optional ByVal yA As Double = 0, _
    Optional ByVal sketchA As String = "", Optional ByVal segNameA As String = "", _
    Optional ByVal epIndexA As Long = 0, _
    Optional ByVal kindB As Long = 0, Optional ByVal objB As Object = Nothing, _
    Optional ByVal xB As Double = 0, Optional ByVal yB As Double = 0, _
    Optional ByVal sketchB As String = "", Optional ByVal segNameB As String = "", _
    Optional ByVal epIndexB As Long = 0) As Boolean

    ApplyCoincidentRelation = False
    If kindB = 0 Then Exit Function
    On Error Resume Next

    ClearSWSelection
    If Not CoincidentSelectSide(kindA, objA, xA, yA, sketchA, segNameA, epIndexA, False) Then Exit Function
    If Not CoincidentSelectSide(kindB, objB, xB, yB, sketchB, segNameB, epIndexB, True) Then Exit Function

    swDoc.SketchAddConstraints "sgCOINCIDENT"
    ApplyCoincidentRelation = (Err.Number = 0)
    If Not ApplyCoincidentRelation Then DebugPrint "ApplyCoincidentRelation WARN kindA=" & kindA & " kindB=" & kindB
    ClearSWSelection
    On Error GoTo 0
End Function

' SketchPoint coincident to a named segment inside another sketch feature (your working select pattern).
Public Function ConstrainSketchPointCoincidentToSegmentInSketch( _
    ByVal swPt As SldWorks.SketchPoint, _
    ByVal sketchFeatName As String, _
    ByVal segName As String) As Boolean

    Dim swFeat As SldWorks.Feature
    Dim swSketch As SldWorks.Sketch
    Dim vSegs As Variant
    Dim vSeg As Variant
    Dim swSeg As SldWorks.SketchSegment

            swDoc.SketchAddConstraints "sgCOINCIDENT"

            ConstrainSketchPointCoincidentToSegmentInSketch = True
            Exit Function
        End If
    Next vSeg

Fail:
    ClearSWSelection
    On Error GoTo 0
End Function

Public Function BuildDrivingGearLayoutCenterSelectId() As String
    Dim drvIdx As Long
    BuildDrivingGearLayoutCenterSelectId = ""

' Position GearN_Center in the layout sketch (Sketch1 / GearN_Layout):
'   Driving  -> coincident to Origin
'   Driven   -> distance dimension from prior gear's Gear(N-1)_Center
Public Sub ConstrainLayoutCenterPointForRole(ByVal swCenterPt As SldWorks.SketchPoint)
    Dim drvPtId As String
    Dim cMM As Double
    Dim dimX As Double
    Dim dimY As Double
    Dim ok As Boolean

    On Error Resume Next
    If swCenterPt Is Nothing Then Exit Sub

    If GearTrainRole = GEAR_ROLE_DRIVING Then
        ok = ConstrainSketchPointToOrigin(swCenterPt)
        If Not ok Then DebugPrint "ConstrainLayoutCenterPointForRole WARN: driving centre not locked to Origin"
        Exit Sub
    End If

    If GearTrainRole = GEAR_ROLE_DRIVEN Then
        drvPtId = BuildDrivingGearLayoutCenterSelectId()
        If drvPtId = "" Then
            DebugPrint "ConstrainLayoutCenterPointForRole: driven centre left at layout coords (no driving point id)"
            Exit Sub
        End If
        ClearSWSelection
        boolstatus = swModelDocExt.SelectByID2(drvPtId, "EXTSKETCHPOINT", 0, 0, 0, False, 0, Nothing, 0)
        If Not boolstatus Then
            DebugPrint "ConstrainLayoutCenterPointForRole: could not select driving layout centre " & drvPtId
            ClearSWSelection
            Exit Sub
        End If
        boolstatus = swCenterPt.Select4(True, Nothing)
        If Not boolstatus Then
            DebugPrint "ConstrainLayoutCenterPointForRole: could not select driven layout centre point"
            ClearSWSelection
            Exit Sub
        End If
        cMM = GetGearTrainLayoutCenterDistanceMM()
        dimX = gSketchOffsetXM * 0.5
        dimY = gSketchOffsetYM * 0.5
        If Abs(dimX) < 0.0000001 And Abs(dimY) < 0.0000001 Then dimX = cMM / MM_PER_M * 0.5
        AddDistanceDimensionBetweenSelectedPoints cMM, GEAR_DIM_LAYOUT_CENTER_DIST, dimX, dimY
        DebugPrint "ConstrainLayoutCenterPointForRole: driven centre dim from " & drvPtId & _
            " C=" & Format(cMM, "0.###") & " mm"
        ClearSWSelection
    End If
    On Error GoTo 0
End Sub

Public Function ConvertRefAxisIntoActiveSketch() As SldWorks.SketchSegment
    Dim swSk As SldWorks.Sketch
    Dim vSegsBefore As Variant
    Dim vSegsAfter As Variant
    Dim nBefore As Long
    Dim i As Long
    Dim swSeg As SldWorks.SketchSegment

    Set ConvertRefAxisIntoActiveSketch = Nothing
    On Error Resume Next
    If swSketchManager.ActiveSketch Is Nothing Then Exit Function
    Set swSk = swSketchManager.ActiveSketch
    vSegsBefore = swSk.GetSketchSegments
    If IsArray(vSegsBefore) Then nBefore = UBound(vSegsBefore) - LBound(vSegsBefore) + 1

    ClearSWSelection
    If Not SelectGearRefAxisForSketchConvert() Then Exit Function
    swModelDocExt.SketchConvertEntities

    vSegsAfter = swSk.GetSketchSegments
    If IsArray(vSegsAfter) Then
        For i = LBound(vSegsAfter) To UBound(vSegsAfter)
            If i > LBound(vSegsAfter) + nBefore - 1 Or nBefore = 0 Then
                Set swSeg = vSegsAfter(i)
                If Not swSeg Is Nothing Then
                    swSeg.ConstructionGeometry = True
                    Set ConvertRefAxisIntoActiveSketch = swSeg
                    Exit For
                End If
            End If
        Next i
        If ConvertRefAxisIntoActiveSketch Is Nothing And nBefore = 0 And UBound(vSegsAfter) >= LBound(vSegsAfter) Then
            Set ConvertRefAxisIntoActiveSketch = vSegsAfter(LBound(vSegsAfter))
            ConvertRefAxisIntoActiveSketch.ConstructionGeometry = True
        End If
    End If
    ClearSWSelection
    If Not ConvertRefAxisIntoActiveSketch Is Nothing Then
        DebugPrint "ConvertRefAxisIntoActiveSketch OK: " & gAxisName
    Else
        DebugPrint "ConvertRefAxisIntoActiveSketch WARN: no segment from " & gAxisName
    End If
    On Error GoTo 0
End Function

Private Function GetLayoutCenterSketchPoint() As SldWorks.SketchPoint
    Dim swSketch As SldWorks.Sketch
    Dim vPts As Variant
    Dim swPt As SldWorks.SketchPoint
    Dim cx As Double
    Dim cy As Double
    Dim bestD As Double
    Dim d As Double
    Dim x As Double
    Dim y As Double
    Dim z As Double
    Dim i As Long
    Dim ptCoords As Variant

    Set GetLayoutCenterSketchPoint = Nothing
    Set swLayoutCenterPt = Nothing
    If swLayoutSketchFeat Is Nothing Then Exit Function

    cx = gSketchOffsetXM
    cy = gSketchOffsetYM
    If GearTrainRole = GEAR_ROLE_DRIVING Then
        cx = 0#
        cy = 0#
    End If

    On Error Resume Next
    Set swSketch = swLayoutSketchFeat.GetSpecificFeature2
    If swSketch Is Nothing Then Exit Function
    vPts = swSketch.GetSketchPoints2
    If Not IsArray(vPts) Then Exit Function

    bestD = 1E+30
    For i = LBound(vPts) To UBound(vPts)
        Set swPt = vPts(i)
        If Not swPt Is Nothing Then
            ptCoords = swPt.GetCoords
            If IsArray(ptCoords) Then
                If UBound(ptCoords) >= 1 Then
                    x = ptCoords(0)
                    y = ptCoords(1)
                    If UBound(ptCoords) >= 2 Then z = ptCoords(2)
                End If
            End If
            d = (x - cx) ^ 2 + (y - cy) ^ 2
            If d < bestD Then
                bestD = d
                Set GetLayoutCenterSketchPoint = swPt
            End If
        End If
    Next i
    If Not GetLayoutCenterSketchPoint Is Nothing Then
        Set swLayoutCenterPt = GetLayoutCenterSketchPoint
    End If
    On Error GoTo 0
End Function

Private Sub DeleteAllRefAxisFeaturesExcept(Optional ByVal keepName As String = "")
    Dim ft As SldWorks.Feature
    Dim names() As String
    Dim i As Long
    Dim n As Long

    n = 0
    ReDim names(0 To 0)
    Set ft = swPart.FirstFeature
    Do While Not ft Is Nothing
        If FeatureIsRefAxis(ft) Then
            If keepName = "" Or StrComp(ft.name, keepName, vbTextCompare) <> 0 Then
                ReDim Preserve names(0 To n)
                names(n) = ft.name
                n = n + 1
            End If
        End If
        Set ft = ft.GetNextFeature
    Loop

    For i = 0 To n - 1
        On Error Resume Next
        ClearSWSelection
        If swModelDocExt.SelectByID2(names(i), "AXIS", 0, 0, 0, False, 0, Nothing, 0) Then
            swDoc.EditDelete
        ElseIf swModelDocExt.SelectByID2(names(i), "REFAXIS", 0, 0, 0, False, 0, Nothing, 0) Then
            swDoc.EditDelete
        End If
        ClearSWSelection
        On Error GoTo 0
    Next i
    If n > 0 Then RebuildSWPart
End Sub

' InsertAxis2: Front Plane first, then sketch point (SW API order).
Private Function InsertSingleRefAxisFrontPlaneAndSketchPoint(ByVal swPt As SldWorks.SketchPoint) As Boolean
    Dim nBefore As Long
    Dim nAfter As Long
    Dim selOk As Boolean

    InsertSingleRefAxisFrontPlaneAndSketchPoint = False

    End If

    RebuildSWPart

    Else
        If nAfter > nBefore Then DeleteAllRefAxisFeaturesExcept

    End If
End Function

' =============================================================================
' SOLIDWORKS: GearN_Axis  (Reference Geometry > Axis) — exactly ONE per gear
' Front Plane + GearN_Center from layout sketch (Select4, not SelectByID2 strings).
' =============================================================================
Public Function CreateGearRefAxisFromLayoutSketchPoint() As Boolean
    Dim axisCountBefore As Long
    Dim axisName As String
    Dim existingAxis As SldWorks.Feature
    Dim swPt As SldWorks.SketchPoint

    CreateGearRefAxisFromLayoutSketchPoint = False
    DebugPrint "CreateGearRefAxisFromLayoutSketchPoint: single-axis path (GearLib paste check)"
    axisName = BuildGearRefAxisName()

    Set existingAxis = FindRefAxisFeatureByName(axisName)
    If Not existingAxis Is Nothing Then
        DeleteAllRefAxisFeaturesExcept axisName
        Set swGearAxisFeat = existingAxis
        ApplyGearAxisFeatureName swGearAxisFeat, axisName
        CreateGearRefAxisFromLayoutSketchPoint = True
        DebugPrint "CreateGearRefAxisFromLayoutSketchPoint OK (reuse): " & gAxisName
        Exit Function
    End If

    If CountRefAxisFeatures() > 0 Then
        DebugPrint "CreateGearRefAxisFromLayoutSketchPoint: removing orphan ref axis(es) from prior failed run"
        DeleteAllRefAxisFeaturesExcept
    End If

    If swFrontPlaneFeat Is Nothing Then
        If Not ResolveAndStoreFrontPlane() Then
            DebugPrint "CreateGearRefAxisFromLayoutSketchPoint: Front Plane not found"
            Exit Function
        End If
    End If

    axisCountBefore = CountRefAxisFeatures()

    If GearTrainRole = GEAR_ROLE_DRIVING And Abs(gSketchOffsetXM) < 0.00001 And Abs(gSketchOffsetYM) < 0.00001 Then
        If Not InsertSingleRefAxisFrontPlaneAndOrigin() Then
            DebugPrint "CreateGearRefAxisFromLayoutSketchPoint: could not create ref axis at Origin"
            Exit Function
        End If
    Else
        Set swPt = GetLayoutCenterSketchPoint()
        If swPt Is Nothing Then
            DebugPrint "CreateGearRefAxisFromLayoutSketchPoint: layout centre point not found"
            Exit Function
        End If
        If Not InsertSingleRefAxisFrontPlaneAndSketchPoint(swPt) Then
            DebugPrint "CreateGearRefAxisFromLayoutSketchPoint: could not create ref axis at layout centre"
            Exit Function
        End If
    End If

    If CommitRefAxisAfterInsert(axisCountBefore, axisName) Then
        CreateGearRefAxisFromLayoutSketchPoint = True
        DebugPrint "CreateGearRefAxisFromLayoutSketchPoint OK: " & gAxisName & " (count=" & CountRefAxisFeatures() & ")"
    Else
        DeleteAllRefAxisFeaturesExcept
        DebugPrint "CreateGearRefAxisFromLayoutSketchPoint: ref axis not committed (count=" & CountRefAxisFeatures() & ")"
    End If
End Function

Public Sub AddDiameterDimensionToCircle(ByVal swCircle As SldWorks.SketchSegment, ByVal dimName As String, ByVal diameterMM As Double)

    Dim radiusM As Double

            If Not CreateGearAxis Then

' True when driven gear centre (layout C) is not the same as the selected gear centre
Private Function DrivenGearLayoutDiffersFromSelectedCenter() As Boolean
    Dim drvXM As Double
    Dim drvYM As Double
    Dim dM As Double

    DrivenGearLayoutDiffersFromSelectedCenter = False
    If Not GetDrivingGearCenterOnFrontPlaneM(drvXM, drvYM) Then
        DrivenGearLayoutDiffersFromSelectedCenter = (Abs(gSketchOffsetXM) > 0.00001 Or Abs(gSketchOffsetYM) > 0.00001)
        Exit Function
    End If
    dM = Sqr((gSketchOffsetXM - drvXM) ^ 2 + (gSketchOffsetYM - drvYM) ^ 2)
    DrivenGearLayoutDiffersFromSelectedCenter = (dM > 0.00001)
End Function

Private Function FindLargestCylindricalFaceOnBody(ByVal swPartDoc As SldWorks.ModelDoc2, ByVal bodyLabel As String) As SldWorks.Face2
    Dim swBody As SldWorks.Body2
    Dim vFaces As Variant
    Dim swFace As SldWorks.Face2
    Dim swSurf As SldWorks.Surface
    Dim cylParams As Variant
    Dim bestR As Double
    Dim r As Double
    Dim i As Long

    Set FindLargestCylindricalFaceOnBody = Nothing
    If swPartDoc Is Nothing Or bodyLabel = "" Then Exit Function
    Set swBody = FindSolidBodyByName(swPartDoc, bodyLabel)
    If swBody Is Nothing Then Exit Function

    On Error Resume Next
    vFaces = swBody.GetFaces
    On Error GoTo 0
    If Not IsArray(vFaces) Then Exit Function

    bestR = 0#
    For i = LBound(vFaces) To UBound(vFaces)
        Set swFace = vFaces(i)
        If swFace Is Nothing Then GoTo NextCylFace
        Set swSurf = swFace.GetSurface
        If swSurf Is Nothing Then GoTo NextCylFace
        If swSurf.IsCylinder() Then
            cylParams = swSurf.CylinderParams
            If IsArray(cylParams) Then
                If UBound(cylParams) >= 6 Then
                    r = cylParams(6)
                    If r > bestR Then
                        bestR = r
                        Set FindLargestCylindricalFaceOnBody = swFace
                    End If
                End If
            End If
        End If
NextCylFace:
    Next i
End Function

' Driven / stacked: ref axis from cylindrical face (radius) of the selected gear body

    On Error GoTo 0

Public Function TryCreateDrivenRefAxisFromBossFace() As Boolean
    Dim swFace As SldWorks.Face2
    Dim axisCountBefore As Long
    Dim axisCountAfter As Long

    TryCreateDrivenRefAxisFromBossFace = False
    ' Stacked gear reusing the base ref axis â€” do not replace with a new boss-face axis
    If GearTrainRole = GEAR_ROLE_STACKED And Not swGearAxisFeat Is Nothing And GearTrainDrivingAxisName <> "" Then Exit Function

    Set swFace = FindAddendumCylindricalFace()
    If swFace Is Nothing Then
        DebugPrint "TryCreateDrivenRefAxisFromBossFace: no cylindrical face on blank boss"
        Exit Function
    End If

    axisCountBefore = CountRefAxisFeatures()
    ExitActiveSketchIfAny
    ClearSWSelection
    RebuildSWPart

    On Error Resume Next
    If Not SelectForCirPatternMark(swFace, 0, False) Then swFace.Select2 False, 0
    boolstatus = swDoc.InsertAxis2(True)
    If Err.Number <> 0 Then
        DebugPrint "TryCreateDrivenRefAxisFromBossFace InsertAxis2 err: " & Err.Number & " " & Err.Description
        Err.Clear
        boolstatus = False
    End If
    On Error GoTo 0

    ClearSWSelection
    RebuildSWPart
    axisCountAfter = CountRefAxisFeatures()

    If axisCountAfter > axisCountBefore Then
        Set swGearAxisFeat = FindLastRefAxisFeature()
        If Not swGearAxisFeat Is Nothing Then
            NameGearAxisForCurrentBuild
            gDrivenPatternUsesBossFace = False
            TryCreateDrivenRefAxisFromBossFace = True
            DebugPrint "TryCreateDrivenRefAxisFromBossFace OK: " & gAxisName
            Exit Function
        End If
    End If

    DebugPrint "TryCreateDrivenRefAxisFromBossFace: using cylindrical face directly (before=" & axisCountBefore & " after=" & axisCountAfter & ")"
    gDrivenPatternUsesBossFace = True
    TryCreateDrivenRefAxisFromBossFace = True
End Function

Private Function FindSecondRefAxisFeature() As SldWorks.Feature

    Dim n As Long

    n = 0

' ---------------------------------------------------------------------------
' 3. CreateOrResetPart
' ---------------------------------------------------------------------------
Public Function CreateOrResetPart() As Boolean
    Dim axisOk As Boolean

    DebugPrint "CreateOrResetPart: begin role=" & GearTrainRole
    gFrontPlaneName = ""
    gAxisName = ""

    If Not GearTrainRoleIsDrivenOrStacked() Then
        On Error Resume Next
        swDoc.ViewZoomtofit2
        swDoc.GraphicsRedraw2
        Err.Clear
        On Error GoTo 0
    End If

    ExitActiveSketchIfAny

    If Not ResolveAndStoreFrontPlane() Then
        NoteGearBuildError "ResolveAndStoreFrontPlane failed — no Front RefPlane in part"
        DebugPrint "CreateOrResetPart: " & gLastGearBuildError
        CreateOrResetPart = False
        Exit Function
    End If
    DebugPrint "CreateOrResetPart: front plane=" & gFrontPlaneName

    If GearTrainRole = GEAR_ROLE_DRIVING Then
        gAxisName = "Z Axis"
        gUseSketchCenterlineAsAxis = False
        ExitActiveSketchIfAny
        ClearSWSelection
        On Error Resume Next
        swSketchManager.AddToDB = True
        swSketchManager.AutoSolve = False
        Set swSelData = swSelMgr.CreateSelectData
        Err.Clear
        On Error GoTo 0
        DebugPrint "CreateOrResetPart OK (McMaster driving, axis=Z Axis)"
        CreateOrResetPart = True
        Exit Function
    End If

    On Error Resume Next
    axisOk = CreateGearAxis()
    If Err.Number <> 0 Then
        DebugPrint "CreateOrResetPart: CreateGearAxis error " & Err.Number & " " & Err.Description
        Err.Clear
        axisOk = False
    End If
    On Error GoTo 0
    If Not axisOk Then
        NoteGearBuildError "CreateGearAxis failed (add-to-part=" & GearTrainRoleIsDrivenOrStacked() & _
            " centerline=" & gUseSketchCenterlineAsAxis & ")"
        DebugPrint "CreateOrResetPart: " & gLastGearBuildError
        CreateOrResetPart = False
        Exit Function
    End If

    ExitActiveSketchIfAny
    ClearSWSelection

    On Error Resume Next
    swSketchManager.AddToDB = True
    swSketchManager.AutoSolve = False
    Set swSelData = swSelMgr.CreateSelectData
    Err.Clear
    On Error GoTo 0

    DebugPrint "CreateOrResetPart OK (axis=" & gAxisName & ")"
    CreateOrResetPart = True
End Function

Private Function TryParseGearBodyName(ByVal bodyName As String, ByRef pitchDiameterMM As Double, ByRef numTeeth As Long) As Boolean
    Dim s As String
    Dim rest As String
    Dim zPart As String
    Dim teethPart As String
    Dim diaPart As String
    Dim i As Long
    Dim ch As String
    Dim tPos As Long

    TryParseGearBodyName = False
    pitchDiameterMM = 0#
    numTeeth = 0
    s = StripGearBodyNamePrefix(bodyName)
    If Len(s) < 3 Then Exit Function

    ' Current format: {teeth}T {pitch dia}  e.g. 100T 125  (also Gear1_100T 125 after strip)
    i = 1
    Do While i <= Len(s)
        ch = Mid$(s, i, 1)
        If ch >= "0" And ch <= "9" Then
            i = i + 1
        Else
            Exit Do
        End If
    Loop
    If i >= 2 Then
        teethPart = Left$(s, i - 1)
        tPos = i
        If UCase$(Mid$(s, tPos, 1)) = "T" Then
            rest = Trim$(Mid$(s, tPos + 1))
            If rest <> "" Then
                i = 1
                Do While i <= Len(rest)
                    ch = Mid$(rest, i, 1)
                    If (ch >= "0" And ch <= "9") Or ch = "." Then
                        i = i + 1
                    Else
                        Exit Do
                    End If
                Loop
                If i >= 2 Then
                    diaPart = Left$(rest, i - 1)
                    numTeeth = CLng(teethPart)
                    pitchDiameterMM = CDbl(diaPart)
                    If numTeeth >= 3 And pitchDiameterMM > 0# Then
                        TryParseGearBodyName = True
                        Exit Function
                    End If
                End If
            End If
        End If
    End If

    ' Legacy: XT {pitch} z{teeth}  e.g. XT 125 z100
    If Len(s) < 4 Then Exit Function
    If UCase$(Left$(s, 2)) <> "XT" Then Exit Function
    If Mid$(s, 3, 1) <> " " And Mid$(s, 3, 1) <> vbTab Then Exit Function
    rest = Trim$(Mid$(s, 4))
    If rest = "" Then Exit Function

    i = 1
    Do While i <= Len(rest)
        ch = Mid$(rest, i, 1)
        If (ch >= "0" And ch <= "9") Or ch = "." Then
            i = i + 1
        Else
            Exit Do
        End If
    Loop
    If i < 2 Then Exit Function
    pitchDiameterMM = CDbl(Left$(rest, i - 1))

    zPart = Trim$(Mid$(rest, i))
    If zPart = "" Then Exit Function
    If UCase$(Left$(zPart, 1)) = "Z" Then zPart = Trim$(Mid$(zPart, 2))
    If zPart = "" Then Exit Function
    numTeeth = CLng(zPart)
    If numTeeth < 3 Then Exit Function
    TryParseGearBodyName = True
End Function

Public Function ReadGearMetadataFromBody(ByVal swBody As SldWorks.Body2, ByVal swPartDoc As SldWorks.ModelDoc2, _
    ByRef moduleMM As Double, ByRef numTeeth As Long, ByRef pressureDeg As Double, ByRef widthMM As Double) As Boolean
    Dim pitchDia As Double
    Dim swExt As SldWorks.ModelDocExtension
    Dim gearIdx As Long

    ReadGearMetadataFromBody = False
    pressureDeg = 20#
    widthMM = GearGeom.widthMM
    If widthMM <= 0# Then widthMM = 10#

    If Not swBody Is Nothing Then
        If TryParseGearBodyName(swBody.name, pitchDia, numTeeth) Then
            moduleMM = pitchDia / CDbl(numTeeth)
            If Not swPartDoc Is Nothing Then
                Set swExt = swPartDoc.Extension
                gearIdx = ParseGearIndexFromBodyName(swBody.name)
                If gearIdx > 0 Then
                    If ReadGearGlobalsFromEquations(swPartDoc, gearIdx, moduleMM, numTeeth, pressureDeg, widthMM) Then
                        GoTo BodyMetaOk
                    End If
                End If
                If Not ReadPartCustomPropertyDouble(swExt, GEAR_PROP_PRESSURE, pressureDeg) Then pressureDeg = 20#
                If Not ReadPartCustomPropertyDouble(swExt, GEAR_PROP_WIDTH, widthMM) Then widthMM = 10#
            End If
BodyMetaOk:
            ReadGearMetadataFromBody = True
            Exit Function
        End If
    End If

    If Not swPartDoc Is Nothing Then
        ReadGearMetadataFromBody = ReadGearMetadataFromDoc(swPartDoc, moduleMM, numTeeth, pressureDeg, widthMM)
    End If
End Function

Private Function GearGlobalVarName(ByVal suffix As String) As String
    GearGlobalVarName = "gear" & GetCurrentGearIndex() & "_" & suffix

Private Function EqQuotedName(ByVal name As String) As String
    EqQuotedName = Chr(34) & name & Chr(34)
End Function

Private Function FindEquationIndexByVarName(ByVal swEqnMgr As SldWorks.EquationMgr, ByVal varName As String) As Long
    Dim i As Long
    Dim eqLine As String
    Dim lhs As String

    FindEquationIndexByVarName = -1
    If swEqnMgr Is Nothing Or varName = "" Then Exit Function
    On Error Resume Next
    For i = 0 To swEqnMgr.GetCount - 1
        eqLine = swEqnMgr.Equation(i)
        lhs = Trim$(Split(eqLine, "=")(0))
        If Left$(lhs, 1) = Chr(34) And Right$(lhs, 1) = Chr(34) Then
            lhs = Mid$(lhs, 2, Len(lhs) - 2)
        End If
        If StrComp(lhs, varName, vbTextCompare) = 0 Then
            FindEquationIndexByVarName = i
            Exit Function
        End If
    Next i
    On Error GoTo 0
End Function

Private Function AddOrUpdateGlobalEquation(ByVal varName As String, ByVal valueText As String) As Boolean
    Dim swEqnMgr As SldWorks.EquationMgr
    Dim eqLine As String
    Dim idx As Long

    AddOrUpdateGlobalEquation = False
    If swPart Is Nothing Or varName = "" Then Exit Function
    Set swEqnMgr = swPart.GetEquationMgr
    If swEqnMgr Is Nothing Then Exit Function

    eqLine = EqQuotedName(varName) & " = " & valueText
    idx = FindEquationIndexByVarName(swEqnMgr, varName)
    On Error Resume Next
    If idx >= 0 Then
        swEqnMgr.Equation(idx) = eqLine
    Else
        idx = swEqnMgr.Add2(-1, eqLine, True)
        If idx < 0 Then Exit Function
    End If
    AddOrUpdateGlobalEquation = True
    DebugPrint "Equation global: " & eqLine
    On Error GoTo 0
End Function

Private Function FindEquationIndexByDimRef(ByVal swEqnMgr As SldWorks.EquationMgr, ByVal dimRef As String) As Long
    Dim i As Long
    Dim eqLine As String
    Dim lhs As String

    FindEquationIndexByDimRef = -1
    If swEqnMgr Is Nothing Or dimRef = "" Then Exit Function
    On Error Resume Next
    For i = 0 To swEqnMgr.GetCount - 1
        eqLine = swEqnMgr.Equation(i)
        lhs = Trim$(Split(eqLine, "=")(0))
        If Left$(lhs, 1) = Chr(34) And Right$(lhs, 1) = Chr(34) Then
            lhs = Mid$(lhs, 2, Len(lhs) - 2)
        End If
        If StrComp(lhs, dimRef, vbTextCompare) = 0 Then
            FindEquationIndexByDimRef = i
            Exit Function
        End If
    Next i
    On Error GoTo 0
End Function

Private Function AddOrUpdateDimensionEquation(ByVal dimRef As String, ByVal rhsExpr As String) As Boolean
    Dim swEqnMgr As SldWorks.EquationMgr
    Dim eqLine As String
    Dim idx As Long

    AddOrUpdateDimensionEquation = False
    If swPart Is Nothing Or dimRef = "" Then Exit Function
    Set swEqnMgr = swPart.GetEquationMgr
    If swEqnMgr Is Nothing Then Exit Function

    eqLine = EqQuotedName(dimRef) & " = " & rhsExpr
    idx = FindEquationIndexByDimRef(swEqnMgr, dimRef)
    On Error Resume Next
    If idx >= 0 Then
        swEqnMgr.Equation(idx) = eqLine
    Else
        idx = swEqnMgr.Add2(-1, eqLine, True)
        If idx < 0 Then Exit Function
    End If
    AddOrUpdateDimensionEquation = True
    DebugPrint "Equation dim: " & eqLine
    On Error GoTo 0
End Function

Private Function ReadGearGlobalsFromEquations(ByVal swPartDoc As SldWorks.ModelDoc2, ByVal gearIndex As Long, _
    ByRef moduleMM As Double, ByRef numTeeth As Long, ByRef pressureDeg As Double, ByRef widthMM As Double) As Boolean
    Dim swEqnMgr As SldWorks.EquationMgr
    Dim prefix As String
    Dim teethVal As Double

    ReadGearGlobalsFromEquations = False
    If swPartDoc Is Nothing Or gearIndex < 1 Then Exit Function
    Set swEqnMgr = swPartDoc.GetEquationMgr
    If swEqnMgr Is Nothing Then Exit Function
    prefix = "gear" & gearIndex & "_"

Private Function ReadEquationGlobalDouble(ByVal swEqnMgr As SldWorks.EquationMgr, ByVal varName As String, ByRef outVal As Double) As Boolean
    Dim idx As Long
    Dim eqLine As String
    Dim rhs As String

    ReadEquationGlobalDouble = False
    idx = FindEquationIndexByVarName(swEqnMgr, varName)
    If idx < 0 Then Exit Function
    eqLine = swEqnMgr.Equation(idx)
    rhs = Trim$(Split(eqLine, "=")(1))
    If rhs = "" Then Exit Function
    On Error Resume Next
    outVal = CDbl(rhs)
    ReadEquationGlobalDouble = (Err.Number = 0)
    On Error GoTo 0
End Function

Public Function CombineGearBodies() As Boolean
    On Error GoTo CombineDone

    Const swCombineAdd As Long = 0   ' swCombineBodiesOperationType_e.swCombineOperationType_Add

    Dim vBodies As Variant
    Dim swBody As SldWorks.Body2
    Dim i As Long
    Dim bodyCount As Long
    Dim appendSel As Boolean

    CombineGearBodies = True
    RebuildSWPart
    vBodies = swPart.GetBodies2(swSolidBody, False)
    If IsEmpty(vBodies) Then
        DebugPrint "CombineGearBodies: no solid bodies"
        CombineGearBodies = False
        Exit Function
    End If

    If IsArray(vBodies) Then
        bodyCount = UBound(vBodies) - LBound(vBodies) + 1
    Else
        bodyCount = 1
    End If

    DebugPrint "CombineGearBodies: body count=" & bodyCount
    If bodyCount < 2 Then
        DebugPrint "CombineGearBodies: skipped (single solid already merged)"
        Exit Function
    End If

    ClearSWSelection
    For i = LBound(vBodies) To UBound(vBodies)
        Set swBody = vBodies(i)
        appendSel = (i > LBound(vBodies))
        If Not SelectSolidBodyForCombine(swBody, appendSel) Then
            DebugPrint "CombineGearBodies: could not select body index " & i
            GoTo CombineDone
        End If
    Next i

    Set swCombineFeat = swFeatMgr.InsertCombineFeature(swCombineAdd, Nothing, False)
    If swCombineFeat Is Nothing Then
        DebugPrint "CombineGearBodies: InsertCombineFeature returned Nothing"
        GoTo CombineDone
    End If

    ClearSWSelection
    RebuildSWPart
    DebugPrint "CombineGearBodies OK (" & swCombineFeat.name & ")"
    CombineGearBodies = True
    Exit Function

CombineDone:
    DebugPrint "CombineGearBodies skipped or failed: " & Err.Number & " " & Err.Description
    ClearSWSelection
    CombineGearBodies = False
End Function

' ---------------------------------------------------------------------------
' 11. RebuildAndValidate
' ---------------------------------------------------------------------------
Public Function RebuildAndValidate() As Boolean
    On Error GoTo ValFail
    Dim vBodies As Variant
    Dim bodyCount As Long
    Dim feat As SldWorks.Feature
    Dim failed As Boolean
    Dim errCount As Long

    DebugPrint "RebuildAndValidate: begin"
    RebuildSWPart
    failed = False
    errCount = 0

    Set feat = swPart.FirstFeature
    Do While Not feat Is Nothing
        If feat.IsSuppressed = False Then
            If feat.GetErrorCode2(True) <> 0 Then
                errCount = errCount + 1
                DebugPrint "Feature error: " & feat.name & " code=" & feat.GetErrorCode2(True)
                failed = True
            End If
        End If
        Set feat = feat.GetNextFeature
    Loop

    vBodies = swPart.GetBodies2(swSolidBody, False)
    If IsEmpty(vBodies) Then
        bodyCount = 0
        failed = True
    ElseIf IsArray(vBodies) Then
        bodyCount = UBound(vBodies) - LBound(vBodies) + 1
    Else
        bodyCount = 1
    End If

    DebugPrint "RebuildAndValidate: bodies=" & bodyCount & ", feature errors=" & errCount

    If bodyCount < 1 Then failed = True
    If failed Then
        NoteGearBuildError "Rebuild validation failed: bodies=" & bodyCount & ", feature errors=" & errCount
        RebuildAndValidate = False
        MsgBox "Gear rebuild reported problems. Bodies=" & bodyCount & ", feature errors=" & errCount & vbCrLf & _
            "See Immediate Window (Ctrl+G) for debug log.", vbExclamation, "Gear Validation"
    Else
        RebuildAndValidate = True
    End If
    Exit Function

ValFail:
    NoteGearBuildError FormatVBError("RebuildAndValidate")
    DebugPrint "RebuildAndValidate FAIL: " & gLastGearBuildError
    RebuildAndValidate = False
End Function

Public Function GetGearBodyCenterOnFrontPlaneM(ByVal swPartDoc As SldWorks.ModelDoc2, ByVal bodyLabel As String, _
    ByRef xM As Double, ByRef yM As Double) As Boolean
    Dim swBody As SldWorks.Body2
    Dim swAxisFeat As SldWorks.Feature
    Dim axisName As String

    GetGearBodyCenterOnFrontPlaneM = False
    xM = 0#: yM = 0#
    If swPartDoc Is Nothing Or bodyLabel = "" Then Exit Function

    axisName = BuildGearAxisFeatureNameFromBodyLabel(bodyLabel)
    If axisName <> "" Then
        Set swAxisFeat = FindRefAxisFeatureByName(axisName, swPartDoc)
        If GetRefAxisPointOnFrontPlaneM(swAxisFeat, xM, yM) Then
            GetGearBodyCenterOnFrontPlaneM = True
            Exit Function
        End If
    End If

    Set swBody = FindSolidBodyByName(swPartDoc, bodyLabel)
    If swBody Is Nothing Then Exit Function

    If GetGearBodyCenterFromBodyBoxM(swBody, xM, yM) Then
        GetGearBodyCenterOnFrontPlaneM = True
        Exit Function
    End If

    GetGearBodyCenterOnFrontPlaneM = GetGearBodyCylindricalCenterOnFrontPlaneM(swPartDoc, bodyLabel, xM, yM)
End Function

' Cylindrical face centre of a macro gear body on the Front sketch plane (metres)
Private Function GetGearBodyCylindricalCenterOnFrontPlaneM(ByVal swPartDoc As SldWorks.ModelDoc2, ByVal bodyLabel As String, _
    ByRef xM As Double, ByRef yM As Double) As Boolean
    Dim swBody As SldWorks.Body2
    Dim vFaces As Variant
    Dim swFace As SldWorks.Face2
    Dim swSurf As SldWorks.Surface
    Dim cylParams As Variant
    Dim bestR As Double
    Dim r As Double
    Dim i As Long

    GetGearBodyCylindricalCenterOnFrontPlaneM = False
    xM = 0#: yM = 0#
    If swPartDoc Is Nothing Or bodyLabel = "" Then Exit Function

    Set swBody = FindSolidBodyByName(swPartDoc, bodyLabel)
    If swBody Is Nothing Then Exit Function

    On Error Resume Next
    vFaces = swBody.GetFaces
    On Error GoTo 0
    If Not IsArray(vFaces) Then Exit Function

    bestR = 0#
    For i = LBound(vFaces) To UBound(vFaces)
        Set swFace = vFaces(i)
        If swFace Is Nothing Then GoTo NextFace
        Set swSurf = swFace.GetSurface
        If swSurf Is Nothing Then GoTo NextFace
        If swSurf.IsCylinder() Then
            cylParams = swSurf.CylinderParams
            If IsArray(cylParams) Then
                If UBound(cylParams) >= 6 Then
                    r = cylParams(6)
                    If r > bestR Then
                        bestR = r
                        xM = cylParams(0)
                        yM = cylParams(1)
                        GetGearBodyCylindricalCenterOnFrontPlaneM = True
                    End If
                End If
            End If
        End If
NextFace:
    Next i
End Function

' Bearing (rad) of ray from part origin through the driving gear centre; mesh angle 0 = along this ray
Private Function GetDrivingGearInlineBearingRad(ByVal drvXM As Double, ByVal drvYM As Double) As Double
    If Abs(drvXM) < 0.0000000001 And Abs(drvYM) < 0.0000000001 Then
        GetDrivingGearInlineBearingRad = 0#
    Else
        GetDrivingGearInlineBearingRad = Atn2(drvYM, drvXM)
    End If
End Function

' Point on ref axis at Front sketch plane (Z ~ 0); gear axes are normal to Front
Private Function GetRefAxisPointOnFrontPlaneM(ByVal swAxisFeat As SldWorks.Feature, ByRef xM As Double, ByRef yM As Double) As Boolean
    Dim swAxis As Object
    Dim params As Variant
    Dim dirZ As Double
    On Error Resume Next
    GetRefAxisPointOnFrontPlaneM = False
    xM = 0#: yM = 0#
    If swAxisFeat Is Nothing Then Exit Function
    Set swAxis = swAxisFeat.GetSpecificFeature2
    If swAxis Is Nothing Then Exit Function
    params = swAxis.GetRefAxisParams
    If IsArray(params) Then
        If UBound(params) >= 5 Then
            dirZ = Abs(params(5))
            If dirZ > 0.85 Then
                xM = params(0)
                yM = params(1)
                GetRefAxisPointOnFrontPlaneM = True
            End If
        ElseIf UBound(params) >= 2 Then
            xM = params(0)
            yM = params(1)
            GetRefAxisPointOnFrontPlaneM = True
        End If
    End If
    On Error GoTo 0
End Function

Public Sub SetGearTrainSketchOffsetFromLayout()
    Dim cM As Double
    Dim aRad As Double
    Dim cMM As Double
    Dim drvXM As Double
    Dim drvYM As Double
    Dim vecXM As Double
    Dim vecYM As Double
    Dim inlineRad As Double
    Dim layoutRad As Double

    If GearTrainRole = GEAR_ROLE_STACKED Then
        If GetDrivingGearCenterOnFrontPlaneM(drvXM, drvYM) Then
            gSketchOffsetXM = drvXM
            gSketchOffsetYM = drvYM
        Else
            gSketchOffsetXM = 0#
            gSketchOffsetYM = 0#
        End If
        gBossStartOffsetZM = GetSelectedBaseGearBodyTopZM()
        DebugPrint "Stacked layout: base """ & GearTrainDrivingBodyLabel & """ centre (mm)=" & _
            Format(gSketchOffsetXM * MM_PER_M, "0.###") & ", " & Format(gSketchOffsetYM * MM_PER_M, "0.###") & _
            " boss start Z (mm)=" & Format(gBossStartOffsetZM * MM_PER_M, "0.###")
        Exit Sub
    End If
    If GearTrainRole <> GEAR_ROLE_DRIVEN Then
        ResetGearSketchOffset
        Exit Sub
    End If
    cMM = GetGearTrainLayoutCenterDistanceMM()
    cM = cMM / MM_PER_M
    aRad = GearTrainMeshAngleDeg * Pi / 180#

    If GetDrivingGearCenterOnFrontPlaneM(drvXM, drvYM) Then
        ' Mesh angle 0 = inline: continue outward along line origin -> driving gear centre
        inlineRad = GetDrivingGearInlineBearingRad(drvXM, drvYM)
        layoutRad = inlineRad + aRad
        vecXM = cM * Cos(layoutRad)
        vecYM = cM * Sin(layoutRad)
        gSketchOffsetXM = drvXM + vecXM
        gSketchOffsetYM = drvYM + vecYM
        DebugPrint "Driven layout: driving """ & GearTrainDrivingBodyLabel & """ centre (mm)=" & Format(drvXM * MM_PER_M, "0.###") & ", " & Format(drvYM * MM_PER_M, "0.###") & _
            " inline=" & Format(inlineRad * 180# / Pi, "0.###") & " deg + mesh=" & Format(GearTrainMeshAngleDeg, "0.###") & _
            " deg C=" & Format(cMM, "0.###") & " -> (" & Format(gSketchOffsetXM * MM_PER_M, "0.###") & ", " & Format(gSketchOffsetYM * MM_PER_M, "0.###") & ") mm"
    Else
        layoutRad = aRad
        vecXM = cM * Cos(layoutRad)
        vecYM = cM * Sin(layoutRad)
        gSketchOffsetXM = vecXM
        gSketchOffsetYM = vecYM
        DebugPrint "Driven layout: driving centre unknown â€” offset from origin only (mm)=" & Format(gSketchOffsetXM * MM_PER_M, "0.###") & ", " & Format(gSketchOffsetYM * MM_PER_M, "0.###") & _
            " (C=" & Format(cMM, "0.###") & " mm, body=" & GearTrainDrivingBodyLabel & ")"
    End If
End Sub

Public Function ParseGearIndexFromBodyName(ByVal bodyName As String) As Long
    Dim i As Long
    Dim ch As String
    Dim digits As String
    ParseGearIndexFromBodyName = 0
    bodyName = Trim$(bodyName)
    If Len(bodyName) < 6 Then Exit Function
    If UCase$(Left$(bodyName, 4)) <> "GEAR" Then Exit Function
    i = 5
    digits = ""
    Do While i <= Len(bodyName)
        ch = Mid$(bodyName, i, 1)
        If ch >= "0" And ch <= "9" Then
            digits = digits & ch
            i = i + 1
        Else
            Exit Do
        End If
    Loop
    If digits <> "" And Mid$(bodyName, i, 1) = "_" Then ParseGearIndexFromBodyName = CLng(digits)
End Function

Public Function CenterDistanceMM(ByVal moduleMM As Double, ByVal z1 As Long, ByVal z2 As Long, _
    Optional ByVal gear1Internal As Boolean = False, Optional ByVal gear2Internal As Boolean = False) As Double
    If gear1Internal And Not gear2Internal Then
        CenterDistanceMM = moduleMM * (CDbl(z1) - CDbl(z2)) / 2#
    ElseIf gear2Internal And Not gear1Internal Then
        CenterDistanceMM = moduleMM * (CDbl(z2) - CDbl(z1)) / 2#
    Else
        CenterDistanceMM = moduleMM * (CDbl(z1) + CDbl(z2)) / 2#
    End If
End Function

Private Sub ClearDrivingGearComboList()
    GearTrainDrivingComboCount = 0
    Erase GearTrainDrivingComboPartTitle
    Erase GearTrainDrivingComboBodyName
End Sub

Private Sub AppendDrivingGearComboEntry(ByVal partTitle As String, ByVal bodyName As String)
    Dim n As Long
    n = GearTrainDrivingComboCount
    ReDim Preserve GearTrainDrivingComboPartTitle(0 To n)
    ReDim Preserve GearTrainDrivingComboBodyName(0 To n)
    GearTrainDrivingComboPartTitle(n) = partTitle
    GearTrainDrivingComboBodyName(n) = bodyName
    GearTrainDrivingComboCount = n + 1
End Sub

Private Function BodyDisplayName(ByVal swBody As SldWorks.Body2, ByVal index1 As Long) As String
    On Error Resume Next
    BodyDisplayName = Trim$(swBody.name)
    If BodyDisplayName = "" Then BodyDisplayName = "Solid Body " & CStr(index1)
    On Error GoTo 0
End Function

Public Function FindOpenPartDocumentByTitle(ByVal partTitle As String) As SldWorks.ModelDoc2
    Dim swDocWalk As SldWorks.ModelDoc2
    On Error Resume Next
    Set FindOpenPartDocumentByTitle = Nothing
    If partTitle = "" Then Exit Function
    Set FindOpenPartDocumentByTitle = swApp.GetOpenDocumentByName(partTitle)
    If Not FindOpenPartDocumentByTitle Is Nothing Then Exit Function
    Set swDocWalk = swApp.GetFirstDocument
    Do While Not swDocWalk Is Nothing
        If swDocWalk.GetTitle = partTitle And swDocWalk.GetType = swDocumentTypes_e.swDocPART Then
            Set FindOpenPartDocumentByTitle = swDocWalk
            Exit Function
        End If
        Set swDocWalk = swDocWalk.GetNext
    Loop
    On Error GoTo 0
End Function

Private Function DrivingGearComboDisplayText(ByVal partTitle As String, ByVal bodyName As String, _
    ByVal hasGearMeta As Boolean, ByVal z1 As Long, ByVal modMM As Double) As String
    If hasGearMeta Then
        DrivingGearComboDisplayText = bodyName & "  (" & partTitle & ", z=" & z1 & ", m=" & Format(modMM, "0.###") & ")"
    Else
        DrivingGearComboDisplayText = bodyName & "  (" & partTitle & " â€” not a macro gear)"
    End If
End Function

Private Sub AddMacroGearBodiesFromPartToCombo(ByVal swPartDoc As SldWorks.ModelDoc2)
    Dim swPart As SldWorks.PartDoc
    Dim vBodies As Variant
    Dim swBody As SldWorks.Body2
    Dim i As Long
    Dim partTitle As String
    Dim bodyName As String

    If swPartDoc Is Nothing Then Exit Sub
    If swPartDoc.GetType <> swDocumentTypes_e.swDocPART Then Exit Sub
    Set swPart = swPartDoc
    partTitle = swPartDoc.GetTitle

    vBodies = swPart.GetBodies2(swBodyType_e.swSolidBody, True)
    If IsEmpty(vBodies) Then Exit Sub

    For i = LBound(vBodies) To UBound(vBodies)
        Set swBody = vBodies(i)
        bodyName = BodyDisplayName(swBody, i - LBound(vBodies) + 1)
        If IsMacroGearBody(swBody, swPartDoc) Then
            AppendDrivingGearComboEntry partTitle, bodyName
        End If
    Next i
End Sub

Private Sub AddMacroGearsFromAssemblyToCombo(ByVal swAssyDoc As SldWorks.ModelDoc2)
    Dim swAssy As SldWorks.AssemblyDoc
    Dim vComps As Variant
    Dim swComp As SldWorks.Component2
    Dim swPartDoc As SldWorks.ModelDoc2
    Dim i As Long
    Dim compName As String
    Dim partTitle As String
    Dim modMM As Double
    Dim z1 As Long
    Dim pa As Double
    Dim w As Double

    Set swAssy = swAssyDoc
    vComps = swAssy.GetComponents(False)
    If IsEmpty(vComps) Then
        AppendDrivingGearComboEntry swAssyDoc.GetTitle, "(no components)"
        Exit Sub
    End If

    For i = LBound(vComps) To UBound(vComps)
        Set swComp = vComps(i)
        If swComp Is Nothing Then GoTo NextComp
        Set swPartDoc = swComp.GetModelDoc2
        If swPartDoc Is Nothing Then GoTo NextComp
        If swPartDoc.GetType <> swDocumentTypes_e.swDocPART Then GoTo NextComp
        If Not ReadGearMetadataFromDoc(swPartDoc, modMM, z1, pa, w) Then GoTo NextComp
        compName = swComp.Name2
        partTitle = swPartDoc.GetTitle
        AppendDrivingGearComboEntry partTitle, compName
NextComp:
    Next i
End Sub

' Rebuild list from all open parts (and assembly macro gear components)
Public Function RefreshDrivingGearBodyComboList() As Long
    Dim swWalk As SldWorks.ModelDoc2

    ClearDrivingGearComboList
    Set swApp = Application.SldWorks
    RefreshDrivingGearBodyComboList = 0

    If swApp Is Nothing Then
        AppendDrivingGearComboEntry "", "(SolidWorks not running)"
        RefreshDrivingGearBodyComboList = GearTrainDrivingComboCount
        Exit Function
    End If

    Set swWalk = swApp.GetFirstDocument
    Do While Not swWalk Is Nothing
        Select Case swWalk.GetType
            Case swDocumentTypes_e.swDocPART
                AddMacroGearBodiesFromPartToCombo swWalk
            Case swDocumentTypes_e.swDocASSEMBLY
                AddMacroGearsFromAssemblyToCombo swWalk
        End Select
        Set swWalk = swWalk.GetNext
    Loop

    If GearTrainDrivingComboCount = 0 Then
        AppendDrivingGearComboEntry "", "(no macro gears â€” name bodies ""teethT pitchDia"" e.g. 100T 125, or generate a driving gear)"
    End If

    RefreshDrivingGearBodyComboList = GearTrainDrivingComboCount
End Function

Public Function DrivingGearComboItemCaption(ByVal index As Long) As String
    Dim swPartDoc As SldWorks.ModelDoc2
    Dim swBody As SldWorks.Body2
    Dim modMM As Double
    Dim z1 As Long
    Dim pa As Double
    Dim w As Double
    Dim hasMeta As Boolean
    Dim partTitle As String
    Dim bodyName As String

    DrivingGearComboItemCaption = ""
    If index < 0 Or index >= GearTrainDrivingComboCount Then Exit Function
    partTitle = GearTrainDrivingComboPartTitle(index)
    bodyName = GearTrainDrivingComboBodyName(index)
    If partTitle = "" Or Left$(bodyName, 1) = "(" Then
        DrivingGearComboItemCaption = bodyName
        Exit Function
    End If
    Set swPartDoc = FindOpenPartDocumentByTitle(partTitle)
    Set swBody = FindSolidBodyByName(swPartDoc, bodyName)
    hasMeta = ReadGearMetadataFromBody(swBody, swPartDoc, modMM, z1, pa, w)
    DrivingGearComboItemCaption = DrivingGearComboDisplayText(partTitle, bodyName, hasMeta, z1, modMM)
End Function

Public Function ApplyDrivingGearFromComboIndex(ByVal index As Long) As Boolean
    Dim swPartDoc As SldWorks.ModelDoc2
    Dim swBody As SldWorks.Body2
    Dim partTitle As String
    Dim bodyName As String
    Dim modMM As Double
    Dim z1 As Long
    Dim pa As Double
    Dim w As Double

    ApplyDrivingGearFromComboIndex = False
    If index < 0 Or index >= GearTrainDrivingComboCount Then Exit Function

    partTitle = GearTrainDrivingComboPartTitle(index)
    bodyName = GearTrainDrivingComboBodyName(index)
    If partTitle = "" Or Left$(bodyName, 1) = "(" Then Exit Function

    Set swPartDoc = FindOpenPartDocumentByTitle(partTitle)
    If swPartDoc Is Nothing Then Exit Function
    Set swBody = FindSolidBodyByName(swPartDoc, bodyName)
    If Not ReadGearMetadataFromBody(swBody, swPartDoc, modMM, z1, pa, w) Then Exit Function

    GearTrainDrivingPartTitle = partTitle
    GearTrainDrivingBodyLabel = bodyName
    GearTrainDrivingModuleMM = modMM
    GearTrainDrivingTeeth = z1
    GearTrainDrivingPressureDeg = pa
    GearTrainDrivingWidthMM = w
    GearTrainDrivingIsInternal = ReadGearIsInternalFromDoc(swPartDoc)
    SyncDrivingGearAxisReference swPartDoc
    ApplyDrivingGearFromComboIndex = True
    DebugPrint "Driving gear from list: " & GearTrainDrivingPartTitle & " / " & GearTrainDrivingBodyLabel & _
        " z=" & z1 & IIf(GearTrainDrivingIsInternal, " INTERNAL", " EXTERNAL") & " axis=" & GearTrainDrivingAxisName
End Function

Public Function PickDrivingGearBodyFromSelection() As Boolean
    Dim swBody As SldWorks.Body2
    Dim swFace As SldWorks.Face2
    Dim swComp As SldWorks.Component2
    Dim swPartDoc As SldWorks.ModelDoc2
    Dim modMM As Double
    Dim z1 As Long
    Dim pa As Double
    Dim w As Double
    Dim selCount As Long
    Dim selType As Long

    PickDrivingGearBodyFromSelection = False
    Set swApp = Application.SldWorks
    Set swDoc = swApp.ActiveDoc
    If swDoc Is Nothing Then Exit Function
    Set swSelMgr = swDoc.SelectionManager
    selCount = swSelMgr.GetSelectedObjectCount2(-1)
    If selCount < 1 Then Exit Function

    selType = swSelMgr.GetSelectedObjectType3(1, -1)
    Set swPartDoc = Nothing
    Set swBody = Nothing

    Select Case selType
        Case swSelectType_e.swSelSOLIDBODIES
            Set swBody = swSelMgr.GetSelectedObject6(1, -1)
            If Not swBody Is Nothing Then Set swPartDoc = swBody.GetParent
        Case swSelectType_e.swSelFACES
            Set swFace = swSelMgr.GetSelectedObject6(1, -1)
            If Not swFace Is Nothing Then
                Set swBody = swFace.GetBody
                If Not swBody Is Nothing Then Set swPartDoc = swBody.GetParent
            End If
        Case swSelectType_e.swSelCOMPONENTS
            Set swComp = swSelMgr.GetSelectedObject6(1, -1)
            If Not swComp Is Nothing Then Set swPartDoc = swComp.GetModelDoc2
        Case Else
            Exit Function
    End Select

    If swPartDoc Is Nothing Then Exit Function
    If swPartDoc.GetType <> swDocumentTypes_e.swDocPART Then Exit Function
    If Not ReadGearMetadataFromBody(swBody, swPartDoc, modMM, z1, pa, w) Then Exit Function

    GearTrainDrivingPartTitle = swPartDoc.GetTitle
    If Not swBody Is Nothing Then
        GearTrainDrivingBodyLabel = Trim$(swBody.name)
    Else
        GearTrainDrivingBodyLabel = "Solid Body"
    End If
    GearTrainDrivingModuleMM = modMM
    GearTrainDrivingTeeth = z1
    GearTrainDrivingPressureDeg = pa
    GearTrainDrivingWidthMM = w
    GearTrainDrivingIsInternal = ReadGearIsInternalFromDoc(swPartDoc)
    SyncDrivingGearAxisReference swPartDoc
    PickDrivingGearBodyFromSelection = True
    DebugPrint "Picked driving gear: " & GearTrainDrivingPartTitle & " / " & GearTrainDrivingBodyLabel & _
        " z=" & z1 & " m=" & modMM & " axis=" & GearTrainDrivingAxisName
End Function

Public Function ApplyDrivenGearInputsFromDrivingGear() As Boolean
    ApplyDrivenGearInputsFromDrivingGear = False
    If GearTrainDrivingTeeth <= 0 Or GearTrainDrivingModuleMM <= 0 Then Exit Function

    GearGeom.pressureAngleDeg = GearTrainDrivingPressureDeg
    GearGeom.moduleMM = GearTrainDrivingModuleMM
    If GearTrainDrivingWidthMM > 0 Then GearGeom.widthMM = GearTrainDrivingWidthMM

    GearTrainCenterDistanceMM = CenterDistanceMM(GearTrainDrivingModuleMM, GearTrainDrivingTeeth, GearGeom.numTeeth, _
        GearTrainDrivingIsInternal, GearGeom.IsInternal)
    ApplyDrivenGearInputsFromDrivingGear = True
    DebugPrint "Driven gear inherits m=" & GearGeom.moduleMM & " pa=" & GearGeom.pressureAngleDeg & _
        " C=" & Format(GearTrainCenterDistanceMM, "0.###") & " mm"
End Function

Public Function ApplyStackedGearInputsFromBaseGear() As Boolean
    ApplyStackedGearInputsFromBaseGear = False
    If GearTrainDrivingTeeth <= 0 Or GearTrainDrivingModuleMM <= 0 Then Exit Function

    GearGeom.pressureAngleDeg = GearTrainDrivingPressureDeg
    GearGeom.moduleMM = GearTrainDrivingModuleMM
    ApplyStackedGearInputsFromBaseGear = True
    DebugPrint "Stacked gear inherits m=" & GearGeom.moduleMM & " pa=" & GearGeom.pressureAngleDeg & _
        " width=" & Format(GearGeom.widthMM, "0.###") & " mm from form"
End Function
