Option Explicit

' FEATURE: Tooth — gap sketch (involute-style construction) + blind cut extrude through rim.
' Paste into VBA module: Tooth
'
' Depends on Involute1: GearGeom, Pi, swToothGapSketchFeat, swToothCutFeat, swFeatMgr, sw2DSketch,
'   gLayoutCenterPointName, swLayoutSketchFeat
' Depends on GearLib: constants, sketch helpers, state, cut scope helpers
'
' Gap sketch construction (matches manual tooth layout):
'   1. Reuse GearN_Layout circles (OD, pitch, root) via SketchUseEdge — no duplicate diameters
'   2. Three construction radials from centre to OD: vertical, half-pitch, full-pitch (arc-length driven)
'   3. Pitch point at pitch circle ∩ half-pitch radial
'   4. Pressure-angle construction line from pitch point to OD
'   5. Flank arc tangent to pressure line at OD, ending on base circle (R = rb * tan(alpha))
'   6. Close gap: full-pitch radial, addendum arc, root arc, root fillet

Private Function ArcMidAngleShort(ByVal angStart As Double, ByVal angEnd As Double) As Double
    Dim span As Double
    span = angEnd - angStart
    Do While span > Pi
        span = span - 2# * Pi
    Loop
    Do While span <= -Pi
        span = span + 2# * Pi
    Loop
    ArcMidAngleShort = angStart + span / 2#
End Function

Private Function CountActiveSketchSegments() As Long
    Dim swSk As SldWorks.Sketch
    Dim vSegs As Variant
    On Error Resume Next
    CountActiveSketchSegments = 0
    Set swSk = swSketchManager.ActiveSketch
    If swSk Is Nothing Then Exit Function
    vSegs = swSk.GetSketchSegments
    If IsArray(vSegs) Then CountActiveSketchSegments = UBound(vSegs) - LBound(vSegs) + 1
    On Error GoTo 0
End Function

Private Function CountActiveSketchPoints() As Long
    Dim swSk As SldWorks.Sketch
    Dim vPts As Variant
    On Error Resume Next
    CountActiveSketchPoints = 0
    Set swSk = swSketchManager.ActiveSketch
    If swSk Is Nothing Then Exit Function
    vPts = swSk.GetSketchPoints2
    If IsArray(vPts) Then CountActiveSketchPoints = UBound(vPts) - LBound(vPts) + 1
    On Error GoTo 0
End Function

Private Function TryConvertLayoutExternalSegment(ByVal segId As String) As SldWorks.SketchSegment
    Dim nBefore As Long
    Dim swSk As SldWorks.Sketch
    Dim vSegs As Variant
    Dim i As Long
    Dim swSeg As SldWorks.SketchSegment

    Set TryConvertLayoutExternalSegment = Nothing
    On Error Resume Next
    Set swSk = swSketchManager.ActiveSketch
    If swSk Is Nothing Then Exit Function
    nBefore = CountActiveSketchSegments()
    ClearSWSelection
    If Not swModelDocExt.SelectByID2(segId, "EXTSKETCHSEGMENT", 0#, 0#, 0#, False, 0, Nothing, 0) Then
        DebugPrint "TryConvertLayoutExternalSegment: select failed " & segId
        Exit Function
    End If
    swSketchManager.SketchUseEdge3 False, False
    vSegs = swSk.GetSketchSegments
    If IsArray(vSegs) Then
        For i = LBound(vSegs) To UBound(vSegs)
            If i >= LBound(vSegs) + nBefore Or nBefore = 0 Then
                Set swSeg = vSegs(i)
                If Not swSeg Is Nothing Then
                    swSeg.ConstructionGeometry = True
                    Set TryConvertLayoutExternalSegment = swSeg
                    Exit Function
                End If
            End If
        Next i
    End If
    On Error GoTo 0
End Function

Private Function TryConvertLayoutCenterPoint() As SldWorks.SketchPoint
    Dim skName As String
    Dim ptId As String
    Dim nBefore As Long
    Dim swSk As SldWorks.Sketch
    Dim vPts As Variant
    Dim i As Long
    Dim swPt As SldWorks.SketchPoint

    Set TryConvertLayoutCenterPoint = Nothing
    skName = GetLayoutSketchFeatName()
    If skName = "" Then Exit Function
    If gLayoutCenterPointName = "" Then gLayoutCenterPointName = BuildGearLayoutCenterPointName()
    ptId = gLayoutCenterPointName & "@" & skName

    On Error Resume Next
    Set swSk = swSketchManager.ActiveSketch
    If swSk Is Nothing Then Exit Function
    nBefore = CountActiveSketchPoints()
    ClearSWSelection
    If Not swModelDocExt.SelectByID2(ptId, "EXTSKETCHPOINT", 0#, 0#, 0#, False, 0, Nothing, 0) Then
        DebugPrint "TryConvertLayoutCenterPoint: select failed " & ptId
        Exit Function
    End If
    swSketchManager.SketchUseEdge3 False, False
    vPts = swSk.GetSketchPoints2
    If IsArray(vPts) Then
        For i = LBound(vPts) To UBound(vPts)
            If i >= LBound(vPts) + nBefore Or nBefore = 0 Then
                Set swPt = vPts(i)
                If Not swPt Is Nothing Then
                    Set TryConvertLayoutCenterPoint = swPt
                    Exit Function
                End If
            End If
        Next i
    End If
    On Error GoTo 0
End Function

Private Function ConvertLayoutReferenceCircles() As Boolean
    Dim skName As String

    ConvertLayoutReferenceCircles = False
    skName = GetLayoutSketchFeatName()
    If skName = "" Then
        NoteGearBuildError "ConvertLayoutReferenceCircles: layout sketch name missing"
        Exit Function
    End If

    Set mGapCenterPt = TryConvertLayoutCenterPoint()
    Set mGapOdCircle = TryConvertLayoutExternalSegment("Arc1@" & skName)
    Set mGapPitchCircle = TryConvertLayoutExternalSegment("Arc2@" & skName)
    Set mGapRootCircle = TryConvertLayoutExternalSegment("Arc3@" & skName)

    If mGapCenterPt Is Nothing Or mGapOdCircle Is Nothing Or mGapPitchCircle Is Nothing Or mGapRootCircle Is Nothing Then
        NoteGearBuildError "ConvertLayoutReferenceCircles: could not convert layout centre / circles"
        Exit Function
    End If
    DebugPrint "ConvertLayoutReferenceCircles OK from " & skName
    ConvertLayoutReferenceCircles = True
End Function

Private Function CreateBaseReferenceCircle(ByVal cx As Double, ByVal cy As Double) As SldWorks.SketchSegment
    Dim rb As Double
    Set CreateBaseReferenceCircle = Nothing
    rb = GearGeom.BaseRadiusM
    Set CreateBaseReferenceCircle = swSketchManager.CreateCircle(cx, cy, 0#, cx + rb, cy, 0#)
    If CreateBaseReferenceCircle Is Nothing Then Exit Function
    CreateBaseReferenceCircle.ConstructionGeometry = True
    ConstrainSketchPointConcentricToCircle mGapCenterPt, CreateBaseReferenceCircle
End Function

Private Sub ConstrainSketchPointCoincidentToCircle(ByVal swPt As SldWorks.SketchPoint, ByVal swCircle As SldWorks.SketchSegment)
    On Error Resume Next
    If swPt Is Nothing Or swCircle Is Nothing Then Exit Sub
    ClearSWSelection
    swPt.Select4 False, Nothing
    swCircle.Select4 True, Nothing
    swDoc.SketchAddConstraints "sgCOINCIDENT"
    ClearSWSelection
    On Error GoTo 0
End Sub

Private Function CreateRadialStubToOd(ByVal cx As Double, ByVal cy As Double, ByVal construction As Boolean) As SldWorks.SketchSegment
    Dim ra As Double
    Dim swLine As SldWorks.SketchSegment
    ra = GearGeom.AddendumRadiusM
    Set swLine = swSketchManager.CreateLine(cx, cy, 0#, cx + ra * 0.15, cy + ra * 0.98, 0#)
    If swLine Is Nothing Then Exit Function
    swLine.ConstructionGeometry = construction
    ConstrainSketchPointCoincidentToSegment mGapCenterPt, swLine
    ConstrainSketchSegmentsCoincident swLine, mGapOdCircle
    Set CreateRadialStubToOd = swLine
End Function

Private Function CreateInvoluteFlankArc(ByVal swPALine As SldWorks.SketchSegment, ByVal angHalf As Double, _
    ByVal cx As Double, ByVal cy As Double) As SldWorks.SketchSegment
    Dim rb As Double
    Dim flankRMM As Double
    Dim tipX As Double
    Dim tipY As Double
    Dim baseX As Double
    Dim baseY As Double
    Dim angBase As Double
    Dim swArc As SldWorks.SketchSegment

    Set CreateInvoluteFlankArc = Nothing
    rb = GearGeom.BaseRadiusM
    flankRMM = GearGeom.BaseRadiusMM * Tan(GearGeom.PressureAngleDeg * Pi / 180#)
    If Not GetLineEndCoords(swPALine, tipX, tipY) Then Exit Function

    angBase = Pi / 2# - angHalf + GearGeom.PressureAngleDeg * Pi / 180#
    baseX = cx + rb * Cos(angBase)
    baseY = cy + rb * Sin(angBase)

    ClearSWSelection
    swPALine.Select4 False, Nothing
    On Error Resume Next
    Set swArc = swSketchManager.CreateTangentArc(tipX, tipY, 0#, baseX, baseY, 0#, 1)
    On Error GoTo 0
    If swArc Is Nothing Then
        Set swArc = swSketchManager.Create3PointArc(tipX, tipY, 0#, (tipX + baseX) / 2#, (tipY + baseY) / 2#, 0#, baseX, baseY, 0#)
    End If
    If swArc Is Nothing Then Exit Function

    ConstrainTangentSketchSegments swArc, swPALine
    ConstrainSketchSegmentsCoincident swArc, mGapBaseCircle
    CreateGapRadiusDimension swArc, GEAR_DIM_GAP_FLANK_ARC_RADIUS, flankRMM, (tipX + baseX) / 2#, (tipY + baseY) / 2#
    Set CreateInvoluteFlankArc = swArc
End Function

Private Function CreateRootFilletArc(ByVal cx As Double, ByVal cy As Double, ByVal angHalf As Double) As SldWorks.SketchSegment
    Dim rr As Double
    Dim filletMM As Double
    Dim xRoot As Double
    Dim yRoot As Double
    Dim xMid As Double
    Dim yMid As Double
    Dim swArc As SldWorks.SketchSegment

    Set CreateRootFilletArc = Nothing
    rr = GearGeom.DedendumRadiusM
    filletMM = GearGeom.GapRootFilletMM
    xRoot = cx + rr * Cos(Pi / 2# - angHalf)
    yRoot = cy + rr * Sin(Pi / 2# - angHalf)
    xMid = cx + (rr + filletMM / MM_PER_M) * Cos(Pi / 2# - angHalf + 0.04)
    yMid = cy + (rr + filletMM / MM_PER_M) * Sin(Pi / 2# - angHalf + 0.04)
    Set swArc = swSketchManager.Create3PointArc(xRoot, yRoot, 0#, xMid, yMid, 0#, cx + rr * Cos(Pi / 2# - angHalf + 0.08), cy + rr * Sin(Pi / 2# - angHalf + 0.08), 0#)
    If swArc Is Nothing Then Exit Function
    ConstrainSketchSegmentsCoradial swArc, mGapRootCircle
    CreateGapRadiusDimension swArc, GEAR_DIM_GAP_ROOT_FILLET, filletMM, xMid, yMid
    Set CreateRootFilletArc = swArc
End Function

Public Function CreateToothGapSketch() As Boolean
    On Error GoTo SketchFail

    Dim cx As Double
    Dim cy As Double
    Dim angHalf As Double
    Dim angFull As Double
    Dim angHalfRad As Double
    Dim angFullRad As Double
    Dim angTipHalf As Double
    Dim angTipFull As Double
    Dim xTipHalf As Double
    Dim yTipHalf As Double
    Dim xTipFull As Double
    Dim yTipFull As Double
    Dim xRootHalf As Double
    Dim yRootHalf As Double
    Dim xRootFull As Double
    Dim yRootFull As Double
    Dim xPitch As Double
    Dim yPitch As Double
    Dim angMidTop As Double
    Dim ra As Double
    Dim rr As Double
    Dim rp As Double

    CreateToothGapSketch = False
    DebugPrint "CreateToothGapSketch: --- layout-linked involute gap ---"
    cx = gSketchOffsetXM
    cy = gSketchOffsetYM
    ra = GearGeom.AddendumRadiusM
    rr = GearGeom.DedendumRadiusM
    rp = GearGeom.PitchRadiusM
    angHalf = GearGeom.GapHalfArcLengthMM / GearGeom.AddendumRadiusMM
    angFull = GearGeom.GapArcLengthMM / GearGeom.AddendumRadiusMM
    angHalfRad = angHalf
    angFullRad = angFull
    angTipHalf = Pi / 2# - angHalfRad
    angTipFull = Pi / 2# - angFullRad

    If Not BeginSketchOnFrontPlane() Then GoTo SketchFail
    DebugPrint "CreateToothGapSketch: front-plane sketch open"
    Set sw2DSketch = swSketchManager.ActiveSketch

    If Not ConvertLayoutReferenceCircles() Then GoTo SketchFail
    Set mGapBaseCircle = CreateBaseReferenceCircle(cx, cy)
    If mGapBaseCircle Is Nothing Then GoTo SketchFail

    ' Step 2 — three construction radials from centre to addendum (vertical + half/full pitch)
    Set mGapVertCenter = swSketchManager.CreateCenterLine(cx, cy, 0#, cx, cy + ra * 1.05, 0#)
    If mGapVertCenter Is Nothing Then GoTo SketchFail
    mGapVertCenter.ConstructionGeometry = True
    ConstrainSketchPointCoincidentToSegment mGapCenterPt, mGapVertCenter
    ConstrainSketchSegmentVertical mGapVertCenter

    Set mGapLineHalf = CreateRadialStubToOd(cx, cy, True)
    Set mGapLineFull = CreateRadialStubToOd(cx, cy, True)
    If mGapLineHalf Is Nothing Or mGapLineFull Is Nothing Then GoTo SketchFail

    If Not CreatePitchArcLengthDimensionBetweenRadials(mGapOdCircle, mGapVertCenter, mGapLineHalf, _
        GEAR_DIM_GAP_HALF_ARC, GearGeom.GapHalfArcLengthMM, cx + ra * 0.35 * Sin(angHalfRad / 2#), cy + ra * 0.35 * Cos(angHalfRad / 2#)) Then
        DebugPrint "CreateToothGapSketch: WARN half-pitch arc dim placement failed"
    End If
    If Not CreatePitchArcLengthDimensionBetweenRadials(mGapOdCircle, mGapVertCenter, mGapLineFull, _
        GEAR_DIM_GAP_ARC, GearGeom.GapArcLengthMM, cx + ra * 0.45 * Sin(angFullRad / 2#), cy + ra * 0.45 * Cos(angFullRad / 2#)) Then
        DebugPrint "CreateToothGapSketch: WARN full-pitch arc dim placement failed"
    End If

    ' Step 3 — pitch point on pitch circle ∩ half-pitch radial
    xPitch = cx + rp * Cos(angTipHalf)
    yPitch = cy + rp * Sin(angTipHalf)
    Set mGapPitchPt = swSketchManager.CreatePoint(xPitch, yPitch, 0#)
    If mGapPitchPt Is Nothing Then GoTo SketchFail
    ConstrainSketchPointCoincidentToSegment mGapPitchPt, mGapLineHalf
    ConstrainSketchPointCoincidentToCircle mGapPitchPt, mGapPitchCircle

    ' Step 4 — pressure-angle construction line from pitch point to addendum
    Set mGapPALine = swSketchManager.CreateLine(xPitch, yPitch, 0#, xPitch + ra * 0.08, yPitch + ra * 0.08, 0#)
    If mGapPALine Is Nothing Then GoTo SketchFail
    mGapPALine.ConstructionGeometry = True
    ConstrainSketchPointCoincidentToSegment mGapPitchPt, mGapPALine
    ConstrainSketchSegmentsCoincident mGapPALine, mGapOdCircle
    CreateGapAngleDimension mGapLineHalf, mGapPALine, GEAR_DIM_GAP_FLANK_ANGLE, GearGeom.PressureAngleDeg, _
        cx + rp * 0.55, cy + rp * 0.55

    ' Step 5 — flank arc tangent to pressure line at addendum, toward base circle
    Set mGapFlankArc = CreateInvoluteFlankArc(mGapPALine, angHalfRad, cx, cy)
    If mGapFlankArc Is Nothing Then GoTo SketchFail

    ' Step 6 — close gap profile (right radial, top addendum arc, root arc, root fillet)
    xTipHalf = cx + ra * Cos(angTipHalf)
    yTipHalf = cy + ra * Sin(angTipHalf)
    xTipFull = cx + ra * Cos(angTipFull)
    yTipFull = cy + ra * Sin(angTipFull)
    xRootHalf = cx + rr * Cos(angTipHalf)
    yRootHalf = cy + rr * Sin(angTipHalf)
    xRootFull = cx + rr * Cos(angTipFull)
    yRootFull = cy + rr * Sin(angTipFull)

    Set mGapRightRad = CreateRadialStubToOd(cx, cy, False)
    If mGapRightRad Is Nothing Then GoTo SketchFail
    ConstrainSketchSegmentColinearToSegment mGapRightRad, mGapLineFull

    angMidTop = ArcMidAngleShort(angTipFull, angTipHalf)
    Set mGapTopArc = swSketchManager.Create3PointArc(xTipFull, yTipFull, 0#, xTipHalf, yTipHalf, 0#, _
        cx + ra * Cos(angMidTop), cy + ra * Sin(angMidTop), 0#)
    If mGapTopArc Is Nothing Then GoTo SketchFail
    ConstrainSketchSegmentsCoradial mGapTopArc, mGapOdCircle

    Set mGapRootArc = swSketchManager.Create3PointArc(xRootHalf, yRootHalf, 0#, xRootFull, yRootFull, 0#, _
        cx + rr * Cos(angMidTop), cy + rr * Sin(angMidTop), 0#)
    If mGapRootArc Is Nothing Then GoTo SketchFail
    ConstrainSketchSegmentsCoradial mGapRootArc, mGapRootCircle

    CreateRootFilletArc cx, cy, angHalfRad

    swSketchManager.InsertSketch True
    Set swToothGapSketchFeat = FindLastSketchFeature()
    If swToothGapSketchFeat Is Nothing Then GoTo SketchFail
    mGapSketchFeatName = swToothGapSketchFeat.Name
    gGapSketchFeatName = mGapSketchFeatName

    ClearSWSelection
    RebuildSWPart
    DebugPrint "CreateToothGapSketch OK feat=" & mGapSketchFeatName & _
        " gapArc=" & Format(GearGeom.GapArcLengthMM, "0.####") & " mm"
    CreateToothGapSketch = True
    Exit Function

SketchFail:
    NoteGearBuildError FormatVBError("CreateToothGapSketch")
    If gLastGearBuildError = "" Then NoteGearBuildError "CreateToothGapSketch failed (sketch geometry or feature missing)"
    DebugPrint "CreateToothGapSketch FAIL: " & gLastGearBuildError
    CreateToothGapSketch = False
End Function

' ---------------------------------------------------------------------------
' Cut extrude (blind through rim width)
' ---------------------------------------------------------------------------
Public Function FindLastCutExtrudeFeature() As SldWorks.Feature
    Dim ft As SldWorks.Feature
    Dim lastCut As SldWorks.Feature
    Dim tn As String

    Set ft = swPart.FirstFeature
    Do While Not ft Is Nothing
        tn = ft.GetTypeName2
        If (tn = "CutExtrude") Or (Left$(ft.Name, 4) = "Cut-") Then Set lastCut = ft
        Set ft = ft.GetNextFeature
    Loop
    Set FindLastCutExtrudeFeature = lastCut
End Function

Private Function SetCutFeatureScopeToCurrentGearBody(ByVal swCutFeat As SldWorks.Feature) As Boolean
    Dim swDef As Object
    Dim swBody As SldWorks.Body2
    Dim bodyArr(0) As Object
    Dim vBodies As Variant

    SetCutFeatureScopeToCurrentGearBody = False
    If swCutFeat Is Nothing Then Exit Function
    Set swBody = GetCurrentGearSolidBody()
    If swBody Is Nothing Then Exit Function

    On Error Resume Next
    Set swDef = swCutFeat.GetDefinition
    If swDef Is Nothing Then Exit Function
    If Not swDef.AccessSelections(swPart, Nothing) Then Exit Function
    swDef.FeatureScope = True
    swDef.AutoSelect = False
    bodyArr(0) = swBody
    vBodies = bodyArr
    swDef.FeatureScopeBodies = vBodies
    If Err.Number <> 0 Then
        Err.Clear
        swDef.Bodies = vBodies
    End If
    SetCutFeatureScopeToCurrentGearBody = swCutFeat.ModifyDefinition(swDef, swPart, Nothing)
    swDef.ReleaseSelectionAccess
    If SetCutFeatureScopeToCurrentGearBody Then
        DebugPrint "SetCutFeatureScopeToCurrentGearBody OK"
    Else
        DebugPrint "SetCutFeatureScopeToCurrentGearBody failed: " & Err.Number & " " & Err.Description
        Err.Clear
    End If
    On Error GoTo 0
End Function

Private Function TryCreateGapCutFeature(ByRef methodUsed As String) As SldWorks.Feature
    Dim lastErr As Long
    Dim lastDesc As String
    Dim useScope As Boolean
    Dim useFeatScope As Boolean
    Dim useAutoSelect As Boolean
    Dim startCond As Long
    Dim startOff As Double

    Set TryCreateGapCutFeature = Nothing
    methodUsed = ""
    useScope = PartNeedsCutFeatureScope()
    useFeatScope = useScope
    useAutoSelect = Not useScope
    GetGearExtrudeStartParams startCond, startOff
    On Error Resume Next

    Err.Clear
    Set TryCreateGapCutFeature = swFeatMgr.FeatureCut4( _
        True, False, False, _
        SW_END_BLIND, SW_END_BLIND, GearGeom.RimWidthM, 0#, _
        False, False, False, False, _
        0#, 0#, _
        False, False, False, False, _
        False, _
        useFeatScope, useAutoSelect, _
        False, False, False, _
        startCond, startOff, False, False)
    If Not TryCreateGapCutFeature Is Nothing Then
        methodUsed = "FeatureCut4 Blind rim width"
        Exit Function
    End If
    lastErr = Err.Number: lastDesc = Err.Description: Err.Clear

    Set TryCreateGapCutFeature = swFeatMgr.FeatureCut4( _
        True, False, False, _
        SW_END_THROUGH_ALL, SW_END_BLIND, 0#, 0#, _
        False, False, False, False, _
        0#, 0#, _
        False, False, False, False, _
        False, _
        useFeatScope, useAutoSelect, _
        False, False, False, _
        startCond, startOff, False, False)
    If Not TryCreateGapCutFeature Is Nothing Then
        methodUsed = "FeatureCut4 ThroughAll"
        Exit Function
    End If
    lastErr = Err.Number: lastDesc = Err.Description: Err.Clear

    Set TryCreateGapCutFeature = swFeatMgr.FeatureCut4( _
        True, False, False, _
        SW_END_BLIND, SW_END_BLIND, GearGeom.WidthM, 0#, _
        False, False, False, False, _
        0#, 0#, _
        False, False, False, False, _
        False, _
        useFeatScope, useAutoSelect, _
        False, False, False, _
        startCond, startOff, False, False)
    If Not TryCreateGapCutFeature Is Nothing Then
        methodUsed = "FeatureCut4 Blind depth"
        Exit Function
    End If
    lastErr = Err.Number: lastDesc = Err.Description: Err.Clear

    Set TryCreateGapCutFeature = swFeatMgr.FeatureCut4( _
        True, False, True, _
        SW_END_THROUGH_ALL, SW_END_BLIND, 0#, 0#, _
        False, False, False, False, _
        0#, 0#, _
        False, False, False, False, _
        False, _
        useFeatScope, useAutoSelect, _
        False, False, False, _
        startCond, startOff, False, False)
    If Not TryCreateGapCutFeature Is Nothing Then
        methodUsed = "FeatureCut4 ThroughAll (Dir reversed)"
        Exit Function
    End If
    lastErr = Err.Number: lastDesc = Err.Description: Err.Clear

    If Not useScope Then
        Set TryCreateGapCutFeature = swFeatMgr.FeatureCut4( _
            True, False, False, _
            SW_END_THROUGH_ALL, SW_END_BLIND, 0#, 0#, _
            False, False, False, False, _
            0#, 0#, _
            False, False, False, False, _
            False, _
            False, True, _
            False, False, False, _
            SW_START_SKETCH_PLANE, 0#, False, False)
        If Not TryCreateGapCutFeature Is Nothing Then
            methodUsed = "FeatureCut4 ThroughAll (auto-intersect)"
            Exit Function
        End If
        lastErr = Err.Number: lastDesc = Err.Description: Err.Clear
    End If

    If useScope Then
        Set TryCreateGapCutFeature = swFeatMgr.FeatureCut4( _
            True, False, False, _
            SW_END_THROUGH_ALL, SW_END_BLIND, 0#, 0#, _
            False, False, False, False, _
            0#, 0#, _
            False, False, False, False, _
            False, _
            True, True, _
            False, False, False, _
            startCond, startOff, False, False)
        If Not TryCreateGapCutFeature Is Nothing Then
            methodUsed = "FeatureCut4 ThroughAll (scope auto-intersect)"
            Exit Function
        End If
        lastErr = Err.Number: lastDesc = Err.Description: Err.Clear
    End If

    DebugPrint "TryCreateGapCutFeature FeatureCut4 failed: " & lastErr & " " & lastDesc
    NoteGearBuildError "FeatureCut4 failed: " & lastErr & " " & lastDesc
    On Error GoTo 0
End Function

Public Function CutExtrudeToothGap() As Boolean
    On Error GoTo CutFail
    Dim methodUsed As String
    Dim selCount As Long

    DebugPrint "CutExtrudeToothGap: begin depth=" & Format(GearGeom.RimWidthMM, "0.###") & " mm"
    If swToothGapSketchFeat Is Nothing Then Set swToothGapSketchFeat = FindLastSketchFeature()
    If swToothGapSketchFeat Is Nothing Then
        NoteGearBuildError "CutExtrudeToothGap: no gap sketch feature"
        DebugPrint gLastGearBuildError
        GoTo CutFail
    End If

    RebuildSWPart
    ExitActiveSketchIfAny
    If Not SelectGapSketchForCut() Then
        NoteGearBuildError "CutExtrudeToothGap: sketch select failed (" & swToothGapSketchFeat.Name & ")"
        DebugPrint gLastGearBuildError
        GoTo CutFail
    End If

    If PartNeedsCutFeatureScope() Then
        If SelectCurrentGearBodyForCutScope() Then
            If Not SelectGapSketchForCutAppend() Then
                DebugPrint "CutExtrudeToothGap: warning — could not re-select sketch after body scope"
            End If
        Else
            DebugPrint "CutExtrudeToothGap: warning — could not pre-select body for feature scope"
        End If
    End If

    selCount = swSelMgr.GetSelectedObjectCount2(-1)
    DebugPrint "CutExtrudeToothGap: selections before cut, count=" & selCount
    On Error Resume Next
    If Not swToothGapSketchFeat Is Nothing Then
        DebugPrint "CutExtrudeToothGap: gap sketch feat=" & swToothGapSketchFeat.Name & _
            " errorCode=" & swToothGapSketchFeat.GetErrorCode2(True)
    End If
    On Error GoTo CutFail

    Set swToothCutFeat = TryCreateGapCutFeature(methodUsed)
    If swToothCutFeat Is Nothing Then
        If gLastGearBuildError = "" Then NoteGearBuildError "CutExtrudeToothGap: TryCreateGapCutFeature returned Nothing"
        GoTo CutFail
    End If

    If PartNeedsCutFeatureScope() Then
        SetCutFeatureScopeToCurrentGearBody swToothCutFeat
    End If

    DebugPrint "Tooth gap cut feature: " & swToothCutFeat.Name & " (" & methodUsed & _
        IIf(PartNeedsCutFeatureScope(), ", scoped to current gear body)", ")")
    ClearSWSelection
    RebuildSWPart
    DebugPrint "CutExtrudeToothGap OK"
    CutExtrudeToothGap = True
    Exit Function

CutFail:
    NoteGearBuildError FormatVBError("CutExtrudeToothGap")
    If gLastGearBuildError = "" Then NoteGearBuildError "CutExtrudeToothGap failed (sketch select or FeatureCut4)"
    DebugPrint "CutExtrudeToothGap FAIL: " & gLastGearBuildError
    CutExtrudeToothGap = False
End Function
