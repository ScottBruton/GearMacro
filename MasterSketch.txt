Option Explicit

' =============================================================================
' FEATURE: Master layout sketch  ->  SolidWorks feature:  GearN_Layout  (Front Plane)
' =============================================================================
' Creates:
'   - Sketch point   GearN_Center     (gear axis centre on Front plane)
'   - Construction circles (OD, Pitch, Root, Hub, Shaft) concentric to GearN_Center
'   - Dimensions: InvoluteGear_OD, PitchDia, RootDia, HubDia, ShaftDia
'   - Driven only: InvoluteGear_LayoutCenterDist from prior gear's Gear(N-1)_Center
'
' Next step in GenerateGear: CreateGearRefAxisFromLayoutSketchPoint  ->  GearN_Axis (one ref axis)
' =============================================================================

Public Function CreateMasterLayoutSketch() As Boolean
    On Error GoTo LayoutFail

    Dim swCircles(1 To 5) As SldWorks.SketchSegment
    Dim swCenterPt As SldWorks.SketchPoint
    Dim labels(1 To 5) As String
    Dim diams(1 To 5) As Double
    Dim layoutSkName As String
    Dim i As Long
    Dim cx As Double
    Dim cy As Double

    CreateMasterLayoutSketch = False
    DebugPrint "CreateMasterLayoutSketch: --- begin ---"
    layoutSkName = BuildGearLayoutSketchName()          ' e.g. Gear1_Layout
    gLayoutCenterPointName = BuildGearLayoutCenterPointName()  ' e.g. Gear1_Center

    ' Driving gear uses part Origin. Driven gear uses layout offset from SetGearTrainSketchOffsetFromLayout.
    If GearTrainRole = GEAR_ROLE_DRIVING Then
        cx = 0#
        cy = 0#
    Else
        cx = gSketchOffsetXM
        cy = gSketchOffsetYM
    End If
    DebugPrint "CreateMasterLayoutSketch: centre (mm)=(" & Format(cx * MM_PER_M, "0.###") & ", " & Format(cy * MM_PER_M, "0.###") & ")"

    ' --- Open Front Plane sketch -------------------------------------------------
    If Not BeginSketchOnFrontPlane() Then GoTo LayoutFail
    Set sw2DSketch = swSketchManager.ActiveSketch
    Set swLayoutSketchFeat = Nothing
    Set swLayoutCenterPt = Nothing

    ' --- GearN_Center point ------------------------------------------------------
    Set swCenterPt = swSketchManager.CreatePoint(cx, cy, 0)
    If swCenterPt Is Nothing Then GoTo LayoutFail
    Set swLayoutCenterPt = swCenterPt
    ConstrainLayoutCenterPointForRole swCenterPt   ' coincident Origin OR dim from driving centre
    TryRenameSketchPoint swCenterPt, gLayoutCenterPointName

    ' --- Construction circles (all concentric to GearN_Center) -------------------
    labels(1) = GEAR_DIM_LAYOUT_OD: diams(1) = GearGeom.AddendumDiameterMM
    labels(2) = GEAR_DIM_LAYOUT_PITCH: diams(2) = GearGeom.PitchDiameterMM
    labels(3) = GEAR_DIM_LAYOUT_ROOT: diams(3) = GearGeom.DedendumDiameterMM
    labels(4) = GEAR_DIM_LAYOUT_HUB: diams(4) = GearGeom.HubDiameterMM
    labels(5) = GEAR_DIM_LAYOUT_SHAFT: diams(5) = GearGeom.ShaftDiameterMM

    For i = 1 To 5
        Set swCircles(i) = swSketchManager.CreateCircle(cx, cy, 0, cx + diams(i) / MM_PER_M / 2#, cy, 0)
        If swCircles(i) Is Nothing Then GoTo LayoutFail
        swCircles(i).ConstructionGeometry = True
        ConstrainSketchPointConcentricToCircle swCenterPt, swCircles(i)
        AddDiameterDimensionToCircle swCircles(i), labels(i), diams(i)
        If i = 1 Then TryRenameSketchSegment swCircles(i), BuildGearPrefixName() & "_OD"
        If i = 2 Then TryRenameSketchSegment swCircles(i), BuildGearPrefixName() & "_Pitch"
        If i = 3 Then TryRenameSketchSegment swCircles(i), BuildGearPrefixName() & "_Root"
    Next i

    ' --- Close sketch and rename feature to GearN_Layout -------------------------
    swSketchManager.InsertSketch True
    Set swLayoutSketchFeat = FindLastSketchFeature()
    If swLayoutSketchFeat Is Nothing Then GoTo LayoutFail
    RenameSketchFeatureSafe swLayoutSketchFeat, layoutSkName
    gLayoutSketchFeatName = layoutSkName

    ClearSWSelection
    RebuildSWPart
    DebugPrint "CreateMasterLayoutSketch OK feat=" & gLayoutSketchFeatName & " centre=" & gLayoutCenterPointName
    CreateMasterLayoutSketch = True
    Exit Function

LayoutFail:
    NoteGearBuildError FormatVBError("CreateMasterLayoutSketch")
    If gLastGearBuildError = "" Then NoteGearBuildError "CreateMasterLayoutSketch failed (plane/sketch/circles)"
    DebugPrint "CreateMasterLayoutSketch FAIL: " & gLastGearBuildError
    CreateMasterLayoutSketch = False
End Function

Public Function GetLayoutSketchFeatName() As String
    GetLayoutSketchFeatName = gLayoutSketchFeatName
End Function
