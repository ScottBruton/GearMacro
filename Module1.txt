' Read nominal + mfg clearance/backlash from UserForm before driven gear build
Public Sub SyncGearTrainLayoutCenterFromForm()
    Dim z2 As Long
    Dim cNom As Double
    Dim clr As Double
    Dim bl As Double

    On Error Resume Next
    GearTrainLayoutCenterDistanceMM = 0#
    If GearTrainRole <> GEAR_ROLE_DRIVEN Then Exit Sub
    If GearTrainDrivingTeeth <= 0 Or GearTrainDrivingModuleMM <= 0 Then Exit Sub

    z2 = CLng(UserForm1.tbTeeth.Text)
    If z2 <= 0 Then Exit Sub

    cNom = CenterDistanceMM(GearTrainDrivingModuleMM, GearTrainDrivingTeeth, z2, GearTrainDrivingIsInternal, GearGeom.IsInternal)
    GearTrainCenterDistanceMM = cNom

    clr = UserForm1.MfgClearanceMM()
    bl = UserForm1.MfgBacklashMM()

    GearTrainLayoutCenterDistanceMM = cNom + clr + bl
    DebugPrint "Layout center distance: nominal=" & Format(cNom, "0.###") & " clr=" & Format(clr, "0.###") & _
        " backlash=" & Format(bl, "0.###") & " -> C=" & Format(GearTrainLayoutCenterDistanceMM, "0.###") & " mm"
    On Error GoTo 0
End Sub

Private Function GetGearBodyCenterFromBodyBoxM(ByVal swBody As SldWorks.Body2, ByRef xM As Double, ByRef yM As Double) As Boolean
    Dim vBox As Variant
    On Error Resume Next
    GetGearBodyCenterFromBodyBoxM = False
    xM = 0#: yM = 0#
    If swBody Is Nothing Then Exit Function
    vBox = swBody.GetBodyBox()
    If Not IsArray(vBox) Then Exit Function
    If UBound(vBox) < 5 Then Exit Function
    xM = (vBox(0) + vBox(3)) / 2#
    yM = (vBox(1) + vBox(4)) / 2#
    GetGearBodyCenterFromBodyBoxM = True
    On Error GoTo 0
End Function

' Centre of a macro gear on the Front sketch plane — axis feature, body box, then largest cylinder (metres)
Public Function GetGearBodyCenterOnFrontPlaneM(ByVal swPartDoc As SldWorks.ModelDoc2, ByVal bodyLabel As String, _
    ByRef xM As Double, ByRef yM As Double) As Boolean
    Dim swBody As SldWorks.Body2
    Dim swAxisFeat As SldWorks.Feature
    Dim axisName As String

    GetGearBodyCenterOnFrontPlaneM = False
    xM = 0#: yM = 0#
    If swPartDoc Is Nothing Or bodyLabel = "" Then Exit Function

    axisName = BuildGearAxisFeatureNameFromBodyLabel(bodyLabel)
    If axisName <> "" Then
        Set swAxisFeat = FindRefAxisFeatureByName(axisName, swPartDoc)
        If GetRefAxisPointOnFrontPlaneM(swAxisFeat, xM, yM) Then
            GetGearBodyCenterOnFrontPlaneM = True
            Exit Function
        End If
    End If

    Set swBody = FindSolidBodyByName(swPartDoc, bodyLabel)
    If swBody Is Nothing Then Exit Function

    If GetGearBodyCenterFromBodyBoxM(swBody, xM, yM) Then
        GetGearBodyCenterOnFrontPlaneM = True
        Exit Function
    End If

    GetGearBodyCenterOnFrontPlaneM = GetGearBodyCylindricalCenterOnFrontPlaneM(swPartDoc, bodyLabel, xM, yM)
End Function

' Cylindrical face centre of a macro gear body on the Front sketch plane (metres)
Private Function GetGearBodyCylindricalCenterOnFrontPlaneM(ByVal swPartDoc As SldWorks.ModelDoc2, ByVal bodyLabel As String, _
    ByRef xM As Double, ByRef yM As Double) As Boolean
    Dim swBody As SldWorks.Body2
    Dim vFaces As Variant
    Dim swFace As SldWorks.Face2
    Dim swSurf As SldWorks.Surface
    Dim cylParams As Variant
    Dim bestR As Double
    Dim r As Double
    Dim i As Long

    GetGearBodyCylindricalCenterOnFrontPlaneM = False
    xM = 0#: yM = 0#
    If swPartDoc Is Nothing Or bodyLabel = "" Then Exit Function

    Set swBody = FindSolidBodyByName(swPartDoc, bodyLabel)
    If swBody Is Nothing Then Exit Function

    On Error Resume Next
    vFaces = swBody.GetFaces
    On Error GoTo 0
    If Not IsArray(vFaces) Then Exit Function

    bestR = 0#
    For i = LBound(vFaces) To UBound(vFaces)
        Set swFace = vFaces(i)
        If swFace Is Nothing Then GoTo NextFace
        Set swSurf = swFace.GetSurface
        If swSurf Is Nothing Then GoTo NextFace
        If swSurf.IsCylinder() Then
            cylParams = swSurf.CylinderParams
            If IsArray(cylParams) Then
                If UBound(cylParams) >= 6 Then
                    r = cylParams(6)
                    If r > bestR Then
                        bestR = r
                        xM = cylParams(0)
                        yM = cylParams(1)
                        GetGearBodyCylindricalCenterOnFrontPlaneM = True
                    End If
                End If
            End If
        End If
NextFace:
    Next i
End Function

Private Function GetDrivingGearCenterOnFrontPlaneM(ByRef xM As Double, ByRef yM As Double) As Boolean
    Dim swPartDoc As SldWorks.ModelDoc2

    GetDrivingGearCenterOnFrontPlaneM = False
    xM = 0#: yM = 0#
    If GearTrainDrivingBodyLabel = "" Then Exit Function

    Set swPartDoc = FindOpenPartDocumentByTitle(GearTrainDrivingPartTitle)
    If swPartDoc Is Nothing Then Set swPartDoc = swDoc

    GetDrivingGearCenterOnFrontPlaneM = GetGearBodyCenterOnFrontPlaneM(swPartDoc, GearTrainDrivingBodyLabel, xM, yM)
End Function

' Bearing (rad) of ray from part origin through the driving gear centre; mesh angle 0 = along this ray
Private Function GetDrivingGearInlineBearingRad(ByVal drvXM As Double, ByVal drvYM As Double) As Double
    If Abs(drvXM) < 0.0000000001 And Abs(drvYM) < 0.0000000001 Then
        GetDrivingGearInlineBearingRad = 0#
    Else
        GetDrivingGearInlineBearingRad = Atn2(drvYM, drvXM)
    End If
End Function

' Point on ref axis at Front sketch plane (Z ~ 0); gear axes are normal to Front
Private Function GetRefAxisPointOnFrontPlaneM(ByVal swAxisFeat As SldWorks.Feature, ByRef xM As Double, ByRef yM As Double) As Boolean
    Dim swAxis As Object
    Dim params As Variant
    Dim dirZ As Double
    On Error Resume Next
    GetRefAxisPointOnFrontPlaneM = False
    xM = 0#: yM = 0#
    If swAxisFeat Is Nothing Then Exit Function
    Set swAxis = swAxisFeat.GetSpecificFeature2
    If swAxis Is Nothing Then Exit Function
    params = swAxis.GetRefAxisParams
    If IsArray(params) Then
        If UBound(params) >= 5 Then
            dirZ = Abs(params(5))
            If dirZ > 0.85 Then
                xM = params(0)
                yM = params(1)
                GetRefAxisPointOnFrontPlaneM = True
            End If
        ElseIf UBound(params) >= 2 Then
            xM = params(0)
            yM = params(1)
            GetRefAxisPointOnFrontPlaneM = True
        End If
    End If
    On Error GoTo 0
End Function

Public Sub SetGearTrainSketchOffsetFromLayout()
    Dim cM As Double
    Dim aRad As Double
    Dim cMM As Double
    Dim drvXM As Double
    Dim drvYM As Double
    Dim vecXM As Double
    Dim vecYM As Double
    Dim inlineRad As Double
    Dim layoutRad As Double

    If GearTrainRole <> GEAR_ROLE_DRIVEN Then
        ResetGearSketchOffset
        Exit Sub
    End If
    cMM = GetGearTrainLayoutCenterDistanceMM()
    cM = cMM / MM_PER_M
    aRad = GearTrainMeshAngleDeg * Pi / 180#

    If GetDrivingGearCenterOnFrontPlaneM(drvXM, drvYM) Then
        ' Mesh angle 0 = inline: continue outward along line origin -> driving gear centre
        inlineRad = GetDrivingGearInlineBearingRad(drvXM, drvYM)
        layoutRad = inlineRad + aRad
        vecXM = cM * Cos(layoutRad)
        vecYM = cM * Sin(layoutRad)
        mSketchOffsetXM = drvXM + vecXM
        mSketchOffsetYM = drvYM + vecYM
        DebugPrint "Driven layout: driving """ & GearTrainDrivingBodyLabel & """ centre (mm)=" & Format(drvXM * MM_PER_M, "0.###") & ", " & Format(drvYM * MM_PER_M, "0.###") & _
            " inline=" & Format(inlineRad * 180# / Pi, "0.###") & " deg + mesh=" & Format(GearTrainMeshAngleDeg, "0.###") & _
            " deg C=" & Format(cMM, "0.###") & " -> (" & Format(mSketchOffsetXM * MM_PER_M, "0.###") & ", " & Format(mSketchOffsetYM * MM_PER_M, "0.###") & ") mm"
    Else
        layoutRad = aRad
        vecXM = cM * Cos(layoutRad)
        vecYM = cM * Sin(layoutRad)
        mSketchOffsetXM = vecXM
        mSketchOffsetYM = vecYM
        DebugPrint "Driven layout: driving centre unknown — offset from origin only (mm)=" & Format(mSketchOffsetXM * MM_PER_M, "0.###") & ", " & Format(mSketchOffsetYM * MM_PER_M, "0.###") & _
            " (C=" & Format(cMM, "0.###") & " mm, body=" & GearTrainDrivingBodyLabel & ")"
    End If
End Sub

Public Sub CaptureSolidBodyCountBeforeGearBuild()
    Dim vBodies As Variant
    On Error Resume Next
    mGearSolidBodyCountBefore = 0
    If swPart Is Nothing Then Exit Sub
    vBodies = swPart.GetBodies2(swSolidBody, False)
    If IsEmpty(vBodies) Then Exit Sub
    If IsArray(vBodies) Then mGearSolidBodyCountBefore = UBound(vBodies) - LBound(vBodies) + 1
    On Error GoTo 0
End Sub

Public Function FindRefAxisFeatureByName(ByVal axisName As String, Optional ByVal swPartDoc As SldWorks.ModelDoc2) As SldWorks.Feature
    Dim swP As SldWorks.PartDoc
    On Error Resume Next
    Set FindRefAxisFeatureByName = Nothing
    If axisName = "" Then Exit Function
    If swPartDoc Is Nothing Then
        Set swP = swPart
    ElseIf swPartDoc.GetType = swDocumentTypes_e.swDocPART Then
        Set swP = swPartDoc
    Else
        Exit Function
    End If
    If swP Is Nothing Then Exit Function
    Set FindRefAxisFeatureByName = swP.FeatureByName(axisName)
    If Not FindRefAxisFeatureByName Is Nothing Then
        If Not FeatureIsRefAxis(FindRefAxisFeatureByName) Then Set FindRefAxisFeatureByName = Nothing
    End If
    On Error GoTo 0
End Function

Private Function FormatPitchDiameterForName(ByVal pitchDiaMM As Double) As String
    Dim s As String
    s = CStr(Round(pitchDiaMM, 4))
    If Right$(s, 1) = "." Then s = Left$(s, Len(s) - 1)
    FormatPitchDiameterForName = s
End Function

' ---------------------------------------------------------------------------
' 1. ReadGearInputsFromForm
'    Reads and validates UserForm1 values; populates GearGeom inputs (mm).
' ---------------------------------------------------------------------------
Public Function ReadGearInputsFromForm() As Boolean
    On Error GoTo ReadFail

    GearGeom.PressureAngleDeg = CDbl(UserForm1.tbPressureAngle.Text)
    GearGeom.ModuleMM = CDbl(UserForm1.tbModule.Text)
    GearGeom.NumTeeth = CLng(UserForm1.tbTeeth.Text)
    GearGeom.WidthMM = CDbl(UserForm1.tbWidth.Text)

    ReadGearInputsFromForm = ValidateGearInputsPublic(GearGeom.PressureAngleDeg, GearGeom.ModuleMM, GearGeom.NumTeeth, GearGeom.WidthMM)
    If ReadGearInputsFromForm Then
        CalculateGearGeometry
        DebugPrint "ReadGearInputsFromForm OK"
    End If
    Exit Function

ReadFail:
    MsgBox "Could not read form inputs: " & Err.Description, vbCritical
    ReadGearInputsFromForm = False
End Function

Public Function ValidateGearInputsPublic(ByVal pressureAngleDeg As Double, ByVal moduleMM As Double, ByVal numTeeth As Long, ByVal widthMM As Double) As Boolean
    Dim errMsg As String
    Dim minTeeth As Double
    Dim alphaRad As Double
    errMsg = ""

    If pressureAngleDeg <= 0# Or pressureAngleDeg >= 90# Then errMsg = errMsg & "Invalid pressure angle." & vbCrLf
    If moduleMM <= 0# Then errMsg = errMsg & "Invalid module." & vbCrLf
    If numTeeth < 3 Then errMsg = errMsg & "Tooth count must be >= 3." & vbCrLf
    If widthMM <= 0# Then errMsg = errMsg & "Invalid width." & vbCrLf

    If GearGeom.IsInternal Then
        Dim tipDia As Double
        Dim rootDia As Double
        alphaRad = pressureAngleDeg * 3.14159265358979 / 180#
        minTeeth = 2# / (Sin(alphaRad) ^ 2)
        If numTeeth < minTeeth + 1 Then
            errMsg = errMsg & "Internal gear needs more teeth (min ~" & Format(minTeeth + 1, "0") & " at this pressure angle)." & vbCrLf
        End If
        tipDia = moduleMM * numTeeth - 2# * moduleMM
        rootDia = moduleMM * numTeeth + 2.5 * moduleMM
        If tipDia <= 0# Or rootDia <= tipDia Then
            errMsg = errMsg & "Internal gear: tip diameter must be less than root diameter (increase teeth or module)." & vbCrLf
        End If
    End If

    If errMsg <> "" Then
        MsgBox errMsg, vbExclamation, "Invalid Gear Inputs"
        ValidateGearInputsPublic = False
    Else
        ValidateGearInputsPublic = True
    End If
End Function

' ---------------------------------------------------------------------------
' 2. CalculateGearGeometry
'    All dimensions in mm stored in GearGeom; meters only for API radii/width.
' ---------------------------------------------------------------------------
Public Sub CalculateGearGeometry()
    Dim alphaRad As Double
    Dim z As Double

    Pi = 3.14159265358979
    z = GearGeom.NumTeeth
    alphaRad = GearGeom.PressureAngleDeg * Pi / 180#

    GearGeom.PitchDiameterMM = GearGeom.ModuleMM * z
    GearGeom.PitchRadiusMM = GearGeom.PitchDiameterMM / 2#
    GearGeom.BaseDiameterMM = GearGeom.PitchDiameterMM * Cos(alphaRad)
    GearGeom.BaseRadiusMM = GearGeom.BaseDiameterMM / 2#

    If GearGeom.IsInternal Then
        ' Ring gear: tip (addendum) toward bore, root (dedendum) at outer rim
        GearGeom.AddendumDiameterMM = GearGeom.PitchDiameterMM - 2# * GearGeom.ModuleMM
        GearGeom.DedendumDiameterMM = GearGeom.PitchDiameterMM + 2.5 * GearGeom.ModuleMM
    Else
        GearGeom.AddendumDiameterMM = GearGeom.PitchDiameterMM + 2# * GearGeom.ModuleMM
        GearGeom.DedendumDiameterMM = GearGeom.PitchDiameterMM - 2.5 * GearGeom.ModuleMM
    End If
    GearGeom.AddendumRadiusMM = GearGeom.AddendumDiameterMM / 2#
    GearGeom.DedendumRadiusMM = GearGeom.DedendumDiameterMM / 2#

    GearGeom.CircularPitchMM = Pi * GearGeom.ModuleMM
    GearGeom.ToothThicknessPitchMM = GearGeom.CircularPitchMM / 2#
    GearGeom.AngularPitchDeg = 360# / z
    GearGeom.AngularPitchRad = 2# * Pi / z
    ' Half tooth at pitch circle: (pi*m/2) / (m*z/2) = pi/z rad total width -> pi/(2z) each side of centerline
    GearGeom.HalfToothAngleRad = Pi / (2# * z)
    GearGeom.HalfToothAngleDeg = 90# / z

    GearGeom.BaseLargerThanRoot = (GearGeom.BaseRadiusMM > GearGeom.DedendumRadiusMM)

    ' Involute parameter t: radius r = rb * Sqr(1 + t^2)
    GearGeom.InvoluteTPitch = InvoluteTFromRadius(GearGeom.BaseRadiusMM, GearGeom.PitchRadiusMM)
    GearGeom.InvoluteTEnd = InvoluteTFromRadius(GearGeom.BaseRadiusMM, GearGeom.AddendumRadiusMM)

    ' Start involute at dedendum when root circle is outside base (normal for moderate/high tooth counts)
    GearGeom.InvoluteTStart = InvoluteTFromRadius(GearGeom.BaseRadiusMM, GearGeom.DedendumRadiusMM)
    If GearGeom.DedendumRadiusMM <= GearGeom.BaseRadiusMM Then
        GearGeom.InvoluteTStart = 0#
    End If

    If GearGeom.InvoluteTEnd < GearGeom.InvoluteTStart Then
        GearGeom.InvoluteTEnd = GearGeom.InvoluteTStart + 0.001
    End If

    GearGeom.PitchRadiusM = GearGeom.PitchRadiusMM / MM_PER_M
    GearGeom.BaseRadiusM = GearGeom.BaseRadiusMM / MM_PER_M
    GearGeom.AddendumRadiusM = GearGeom.AddendumRadiusMM / MM_PER_M
    GearGeom.DedendumRadiusM = GearGeom.DedendumRadiusMM / MM_PER_M
    GearGeom.WidthM = GearGeom.WidthMM / MM_PER_M

    DebugPrint "CalculateGearGeometry: " & IIf(GearGeom.IsInternal, "INTERNAL", "EXTERNAL") & _
        " Dp=" & Format(GearGeom.PitchDiameterMM, "0.###") & _
        " Db=" & Format(GearGeom.BaseDiameterMM, "0.###") & _
        " Dd=" & Format(GearGeom.DedendumDiameterMM, "0.###") & _
        " Da=" & Format(GearGeom.AddendumDiameterMM, "0.###") & _
        " rb<rr=" & (Not GearGeom.BaseLargerThanRoot) & " t=[" & Format(GearGeom.InvoluteTStart, "0.####") & _
        "," & Format(GearGeom.InvoluteTEnd, "0.####") & "]"
End Sub

Private Function InvoluteTFromRadius(ByVal rbMM As Double, ByVal rMM As Double) As Double
    If rbMM <= 0# Or rMM <= rbMM Then
        InvoluteTFromRadius = 0#
    Else
        InvoluteTFromRadius = Sqr((rMM / rbMM) ^ 2 - 1#)
    End If
End Function

Private Function MMToM(ByVal valueMM As Double) As Double
    MMToM = valueMM / MM_PER_M
End Function

' Datum plane/axis helpers - use Feature references from tree (SelectByID2 alone is unreliable on NewPart)
Private Function GetRefPlaneByIndex(ByVal planeIndex As Long) As SldWorks.Feature
    Dim ft As SldWorks.Feature
    Dim n As Long
    n = 0
    Set ft = swPart.FirstFeature
    Do While Not ft Is Nothing
        If ft.GetTypeName2 = "RefPlane" Then
            n = n + 1
            If n = planeIndex Then
                Set GetRefPlaneByIndex = ft
                Exit Function
            End If
        End If
        Set ft = ft.GetNextFeature
    Loop
End Function

Private Function SelectSWPlaneByName(ByVal planeName As String, Optional ByVal append As Boolean = False) As Boolean
    SelectSWPlaneByName = swModelDocExt.SelectByID2(planeName, "PLANE", 0, 0, 0, append, 0, Nothing, 0)
End Function

' Resolve front plane from feature tree (1st RefPlane = Front in default templates; same as original macro)
Private Function ResolveAndStoreFrontPlane() As Boolean
    Dim desc As String

    Set swFrontPlaneFeat = GetRefPlaneByIndex(1)
    If swFrontPlaneFeat Is Nothing Then
        DebugPrint "ResolveAndStoreFrontPlane: no RefPlane in part"
        ResolveAndStoreFrontPlane = False
        Exit Function
    End If

    mFrontPlaneName = swFrontPlaneFeat.Name
    On Error Resume Next
    desc = swFrontPlaneFeat.Description
    On Error GoTo 0
    If desc <> "" Then mFrontPlaneName = desc

    ResolveAndStoreFrontPlane = True
    DebugPrint "Front plane resolved: " & mFrontPlaneName & " (feature " & swFrontPlaneFeat.Name & ")"
End Function

' Call Feature.Select (Sub in VBA - no reliable return value)
Private Sub SelectPlaneFeature(ByVal planeFeat As SldWorks.Feature, Optional ByVal append As Boolean = True)
    If planeFeat Is Nothing Then Exit Sub
    On Error Resume Next
    planeFeat.Select append
    On Error GoTo 0
End Sub

Private Function SelectFrontPlaneFeature(Optional ByVal append As Boolean = False) As Boolean
    If swFrontPlaneFeat Is Nothing Then
        If Not ResolveAndStoreFrontPlane() Then
            SelectFrontPlaneFeature = False
            Exit Function
        End If
    End If
    If Not append Then ClearSWSelection
    SelectPlaneFeature swFrontPlaneFeat, append
    If swSelMgr.GetSelectedObjectCount2(-1) < 1 Then
        SelectFrontPlaneFeature = SelectSWPlaneByName(mFrontPlaneName, append)
        If Not SelectFrontPlaneFeature Then SelectFrontPlaneFeature = SelectSWPlaneByName("Front Plane", append)
    Else
        SelectFrontPlaneFeature = True
    End If
End Function

' Start sketch on front plane - original: RefPlaneFt(1).Select True, InsertSketch True
' NOTE: Feature.Select is a Sub in SW VBA - do NOT test its return value (always False wrongly)
Private Function BeginSketchOnFrontPlane() As Boolean
    Dim selCount As Long

    BeginSketchOnFrontPlane = False
    ExitActiveSketchIfAny
    Set swFrontPlaneFeat = GetRefPlaneByIndex(1)
    If swFrontPlaneFeat Is Nothing Then
        DebugPrint "BeginSketchOnFrontPlane: no front RefPlane"
        Exit Function
    End If
    mFrontPlaneName = swFrontPlaneFeat.Name

    On Error Resume Next
    swFrontPlaneFeat.Select True
    selCount = swSelMgr.GetSelectedObjectCount2(-1)
    If selCount < 1 Then
        swModelDocExt.SelectByID2 mFrontPlaneName, "PLANE", 0, 0, 0, False, 0, Nothing, 0
    End If
    If swSelMgr.GetSelectedObjectCount2(-1) < 1 Then
        swModelDocExt.SelectByID2 "Front Plane", "PLANE", 0, 0, 0, False, 0, Nothing, 0
    End If
    On Error GoTo 0

    swSketchManager.InsertSketch True
    If swSketchManager.ActiveSketch Is Nothing Then
        DebugPrint "BeginSketchOnFrontPlane: InsertSketch did not open a sketch"
        Exit Function
    End If

    BeginSketchOnFrontPlane = True
    DebugPrint "BeginSketchOnFrontPlane OK"
End Function

Private Function CreateGearAxis() As Boolean
    ' Driven gear: one ref axis normal to Front plane at layout centre (same direction as driving Axis1)
    If Abs(mSketchOffsetXM) > 0.0000000001 Or Abs(mSketchOffsetYM) > 0.0000000001 Then
        CreateGearAxis = CreateDrivenGearRefAxisAtLayoutCenter()
        Exit Function
    End If

    Dim swTopPlaneFeat As SldWorks.Feature
    Dim swRightPlaneFeat As SldWorks.Feature

    If swFrontPlaneFeat Is Nothing Then
        If Not ResolveAndStoreFrontPlane() Then GoTo AxisFail
    End If

    ClearSWSelection

    ' Method A: exact original sequence - plane.Select True + Point1@Origin
    SelectPlaneFeature swFrontPlaneFeat, True
    boolstatus = swModelDocExt.SelectByID2("Point1@Origin", "EXTSKETCHPOINT", 0, 0, 0, True, 0, Nothing, 0)
    If boolstatus Then
        boolstatus = swDoc.InsertAxis2(True)
        If boolstatus Then GoTo AxisOk
    End If

    ' Method B: SelectByID2 plane + origin
    ClearSWSelection
    If SelectSWPlaneByName(mFrontPlaneName, False) Or SelectSWPlaneByName("Front Plane", False) Then
        boolstatus = swModelDocExt.SelectByID2("Point1@Origin", "EXTSKETCHPOINT", 0, 0, 0, True, 0, Nothing, 0)
        If Not boolstatus Then boolstatus = swModelDocExt.SelectByID2("Origin", "EXTSKETCHPOINT", 0, 0, 0, True, 0, Nothing, 0)
        If boolstatus Then
            boolstatus = swDoc.InsertAxis2(True)
            If boolstatus Then GoTo AxisOk
        End If
    End If

    ' Method C: Top + Right planes -> Z axis (normal to Front sketch plane)
    ClearSWSelection
    Set swTopPlaneFeat = GetRefPlaneByIndex(2)
    Set swRightPlaneFeat = GetRefPlaneByIndex(3)
    If Not swTopPlaneFeat Is Nothing And Not swRightPlaneFeat Is Nothing Then
        SelectPlaneFeature swTopPlaneFeat, True
        SelectPlaneFeature swRightPlaneFeat, True
        boolstatus = swDoc.InsertAxis2(True)
        If boolstatus Then GoTo AxisOk
    End If

    ClearSWSelection
    If SelectSWPlaneByName("Top Plane", False) Then
        If SelectSWPlaneByName("Right Plane", True) Then
            boolstatus = swDoc.InsertAxis2(True)
            If boolstatus Then GoTo AxisOk
        End If
    End If

    DebugPrint "CreateGearAxis: all methods failed"
    GoTo AxisFail

AxisOk:
    Set swGearAxisFeat = FindLastRefAxisFeature()
    If swGearAxisFeat Is Nothing Then Set swGearAxisFeat = swSelMgr.GetSelectedObject5(1)
    If swGearAxisFeat Is Nothing Then GoTo AxisFail
    NameGearAxisForCurrentBuild
    ClearSWSelection
    CreateGearAxis = True
    DebugPrint "CreateGearAxis OK: " & mAxisName
    Exit Function

AxisFail:
    CreateGearAxis = False
End Function

Private Sub ExitActiveSketchIfAny()
    On Error Resume Next
    If Not swSketchManager.ActiveSketch Is Nothing Then
        swSketchManager.InsertSketch True
    End If
    On Error GoTo 0
End Sub

Private Function FeatureIsRefAxis(ByVal ft As SldWorks.Feature) As Boolean
    Dim tn As String
    On Error Resume Next
    FeatureIsRefAxis = False
    If ft Is Nothing Then Exit Function
    tn = ft.GetTypeName2
    FeatureIsRefAxis = (tn = "RefAxis") Or (tn = "Axis") Or (InStr(1, tn, "Axis", vbTextCompare) > 0)
    On Error GoTo 0
End Function

' Driven gear: RefAxis through layout point, normal to Front plane (parallel to driving Axis1, not in-plane line axis)
Private Function CreateDrivenGearRefAxisAtLayoutCenter() As Boolean
    mUseSketchCenterlineAsAxis = False
    mPatternAxisLineId = ""
    mPatternAxisSketchName = ""
    Set mPatternAxisSegment = Nothing

    ' Do not reuse FindSecondRefAxisFeature — after the first driven gear that is the previous
    ' driven layout axis, not the driving gear or the new driven centre.
    Set swGearAxisFeat = Nothing
    If CreateGearAxisAtSketchCenter() Then
        CreateDrivenGearRefAxisAtLayoutCenter = True
        Exit Function
    End If

    ' InsertAxis2 at offset point often does not add Axis2 when Axis1 exists — create axis from blank cylinder after extrude
    mDrivenPatternUsesBossFace = True
    mAxisName = ""
    CreateDrivenGearRefAxisAtLayoutCenter = True
    DebugPrint "CreateDrivenGearRefAxisAtLayoutCenter: defer axis until addendum boss (cylindrical face)"
End Function

Private Function CreateGearAxisAtSketchCenter() As Boolean
    Dim swSkPt As SldWorks.SketchPoint
    Dim swSkFeat As SldWorks.Feature
    Dim skName As String
    Dim docTitle As String
    Dim ptName As String
    Dim axisCountBefore As Long
    Dim axisCountAfter As Long

    CreateGearAxisAtSketchCenter = False
    If swFrontPlaneFeat Is Nothing Then
        If Not ResolveAndStoreFrontPlane() Then
            DebugPrint "CreateGearAxisAtSketchCenter: front plane failed"
            Exit Function
        End If
    End If

    axisCountBefore = CountRefAxisFeatures()
    ExitActiveSketchIfAny
    ClearSWSelection

    ' Method 1: same as driving gear — Front plane + point, axis normal to face (Z), through layout centre
    SelectPlaneFeature swFrontPlaneFeat, True
    boolstatus = swModelDocExt.SelectByID2("", "EXTSKETCHPOINT", mSketchOffsetXM, mSketchOffsetYM, 0#, True, 0, Nothing, 0)
    If boolstatus Then boolstatus = swDoc.InsertAxis2(True)

    ' Method 1b: point then plane (append)
    If Not boolstatus Then
        ClearSWSelection
        boolstatus = swModelDocExt.SelectByID2("", "EXTSKETCHPOINT", mSketchOffsetXM, mSketchOffsetYM, 0#, False, 0, Nothing, 0)
        If boolstatus Then
            SelectPlaneFeature swFrontPlaneFeat, True
            boolstatus = swDoc.InsertAxis2(True)
        End If
    End If

    ' Method 2: sketch point at layout centre, then axis
    If Not boolstatus Then
        If BeginSketchOnFrontPlane() Then
            Set swSkPt = swSketchManager.CreatePoint(mSketchOffsetXM, mSketchOffsetYM, 0#)
            If Not swSkPt Is Nothing Then
                Set swSkFeat = Nothing
                On Error Resume Next
                Set swSkFeat = swSketchManager.ActiveSketch.GetFeature
                On Error GoTo 0
                If swSkFeat Is Nothing Then Set swSkFeat = FindLastSketchFeature()
                If Not swSkFeat Is Nothing Then skName = swSkFeat.Name

                ClearSWSelection
                SelectPlaneFeature swFrontPlaneFeat, True
                swSkPt.Select4 True, Nothing
                boolstatus = swDoc.InsertAxis2(True)
            End If
            swSketchManager.InsertSketch True
        End If
    End If

    ' Method 3: named sketch point after closing sketch
    If Not boolstatus And skName <> "" Then
        ClearSWSelection
        docTitle = swDoc.GetTitle
        ptName = "Point1@" & skName
        If docTitle <> "" Then ptName = ptName & "@" & docTitle
        SelectPlaneFeature swFrontPlaneFeat, True
        boolstatus = swModelDocExt.SelectByID2(ptName, "EXTSKETCHPOINT", mSketchOffsetXM, mSketchOffsetYM, 0#, True, 0, Nothing, 0)
        If Not boolstatus Then boolstatus = swModelDocExt.SelectByID2("Point1@" & skName, "EXTSKETCHPOINT", mSketchOffsetXM, mSketchOffsetYM, 0#, True, 0, Nothing, 0)
        If boolstatus Then boolstatus = swDoc.InsertAxis2(True)
    End If

    ExitActiveSketchIfAny
    ClearSWSelection
    RebuildSWPart

    axisCountAfter = CountRefAxisFeatures()
    If axisCountAfter <= axisCountBefore Then
        DebugPrint "CreateGearAxisAtSketchCenter: no new axis (before=" & axisCountBefore & " after=" & axisCountAfter & ")"
        Exit Function
    End If

    Set swGearAxisFeat = FindLastRefAxisFeature()
    If swGearAxisFeat Is Nothing Then
        DebugPrint "CreateGearAxisAtSketchCenter: axis count increased but could not get axis feature"
        Exit Function
    End If

    NameGearAxisForCurrentBuild
    mUseSketchCenterlineAsAxis = False
    CreateGearAxisAtSketchCenter = True
    DebugPrint "CreateGearAxisAtSketchCenter OK: " & mAxisName & " (normal to Front) at " & Format(mSketchOffsetXM * MM_PER_M, "0.###") & " mm"
End Function

Private Function FindAddendumCylindricalFace() As SldWorks.Face2
    Dim vFaces As Variant
    Dim i As Long
    Dim swFace As SldWorks.Face2
    Dim swSurf As SldWorks.Surface

    Set FindAddendumCylindricalFace = Nothing
    If swAddendumExtrudeFeat Is Nothing Then Exit Function
    On Error Resume Next
    vFaces = swAddendumExtrudeFeat.GetFaces
    On Error GoTo 0
    If Not IsArray(vFaces) Then Exit Function
    For i = LBound(vFaces) To UBound(vFaces)
        Set swFace = vFaces(i)
        If Not swFace Is Nothing Then
            Set swSurf = swFace.GetSurface
            If Not swSurf Is Nothing Then
                If swSurf.IsCylinder() Then
                    Set FindAddendumCylindricalFace = swFace
                    Exit Function
                End If
            End If
        End If
    Next i
End Function

' Ref axis normal to gear face through driven centre — from cylindrical face of addendum blank (reliable when Axis1 exists)
Public Function TryCreateDrivenRefAxisFromBossFace() As Boolean
    Dim swFace As SldWorks.Face2
    Dim axisCountBefore As Long
    Dim axisCountAfter As Long

    TryCreateDrivenRefAxisFromBossFace = False
    If Not mDrivenPatternUsesBossFace And Not swGearAxisFeat Is Nothing Then Exit Function

    Set swFace = FindAddendumCylindricalFace()
    If swFace Is Nothing Then
        DebugPrint "TryCreateDrivenRefAxisFromBossFace: no cylindrical face on blank boss"
        Exit Function
    End If

    axisCountBefore = CountRefAxisFeatures()
    ExitActiveSketchIfAny
    ClearSWSelection
    RebuildSWPart

    On Error Resume Next
    If Not SelectForCirPatternMark(swFace, 0, False) Then swFace.Select2 False, 0
    boolstatus = swDoc.InsertAxis2(True)
    If Err.Number <> 0 Then
        DebugPrint "TryCreateDrivenRefAxisFromBossFace InsertAxis2 err: " & Err.Number & " " & Err.Description
        Err.Clear
        boolstatus = False
    End If
    On Error GoTo 0

    ClearSWSelection
    RebuildSWPart
    axisCountAfter = CountRefAxisFeatures()

    If axisCountAfter > axisCountBefore Then
        Set swGearAxisFeat = FindLastRefAxisFeature()
        If Not swGearAxisFeat Is Nothing Then
            NameGearAxisForCurrentBuild
            mDrivenPatternUsesBossFace = False
            TryCreateDrivenRefAxisFromBossFace = True
            DebugPrint "TryCreateDrivenRefAxisFromBossFace OK: " & mAxisName
            Exit Function
        End If
    End If

    DebugPrint "TryCreateDrivenRefAxisFromBossFace: using cylindrical face directly (before=" & axisCountBefore & " after=" & axisCountAfter & ")"
    mDrivenPatternUsesBossFace = True
    TryCreateDrivenRefAxisFromBossFace = True
End Function

Private Function FindSecondRefAxisFeature() As SldWorks.Feature
    Dim ft As SldWorks.Feature
    Dim n As Long

    n = 0
    Set ft = swPart.FirstFeature
    Do While Not ft Is Nothing
        If FeatureIsRefAxis(ft) Then
            n = n + 1
            If n = 2 Then
                Set FindSecondRefAxisFeature = ft
                Exit Function
            End If
        End If
        Set ft = ft.GetNextFeature
    Loop
End Function

Private Function CountRefAxisFeatures() As Long
    Dim ft As SldWorks.Feature
    Dim n As Long
    n = 0
    Set ft = swPart.FirstFeature
    Do While Not ft Is Nothing
        If FeatureIsRefAxis(ft) Then n = n + 1
        Set ft = ft.GetNextFeature
    Loop
    CountRefAxisFeatures = n
End Function

' ---------------------------------------------------------------------------
' 4. CreateAddendumBlank - full disk at outer (addendum) diameter
' ---------------------------------------------------------------------------
Public Function CreateAddendumBlank() As Boolean
    On Error GoTo BlankFail
    Dim swCircle As SketchSegment
    Dim rM As Double
    Dim mergeBoss As Boolean

    ExitActiveSketchIfAny
    If GearTrainRole = GEAR_ROLE_DRIVEN Then RebuildSWPart

    If GearGeom.IsInternal Then
        rM = GearGeom.DedendumRadiusM
    Else
        rM = GearGeom.AddendumRadiusM
    End If

    If Not BeginSketchOnFrontPlane() Then
        DebugPrint "CreateAddendumBlank: BeginSketchOnFrontPlane failed"
        GoTo BlankFail
    End If
    Set sw2DSketch = swSketchManager.ActiveSketch

    Set swCircle = swSketchManager.CreateCircleByRadius(mSketchOffsetXM, mSketchOffsetYM, 0, rM)
    If swCircle Is Nothing Then GoTo BlankFail

    If GearGeom.IsInternal Then
        AddDiameterDimensionToCircle swCircle, GEAR_DIM_DEDENDUM_DIAMETER, GearGeom.DedendumDiameterMM
    Else
        AddDiameterDimensionToCircle swCircle, GEAR_DIM_ADDENDUM_DIAMETER, GearGeom.AddendumDiameterMM
    End If
    ConstrainGearCenterToOrigin swCircle

    swSketchManager.InsertSketch True
    mergeBoss = Not PartNeedsCutFeatureScope()

    Set swAddendumExtrudeFeat = swFeatMgr.FeatureExtrusion3(True, False, False, 0, 0, GearGeom.WidthM, 0#, _
        False, False, False, False, 0, 0, False, False, False, False, mergeBoss, False, False, 0, 0, False)

    If swAddendumExtrudeFeat Is Nothing Then GoTo BlankFail
    DebugPrint "Addendum blank feature: " & swAddendumExtrudeFeat.Name

    If PartNeedsCutFeatureScope() Then
        SetBossExtrudeNoMerge swAddendumExtrudeFeat
    End If

    ClearSWSelection
    RebuildSWPart

    If GearGeom.IsInternal Then
        If Not CutInternalGearBore() Then GoTo BlankFail
    End If

    DebugPrint "CreateAddendumBlank OK (" & IIf(GearGeom.IsInternal, "ring Dd/2=", "Da/2=") & _
        Format(IIf(GearGeom.IsInternal, GearGeom.DedendumRadiusMM, GearGeom.AddendumRadiusMM), "0.###") & " mm" & _
        IIf(PartNeedsCutFeatureScope(), ", separate body)", ")")

    If GearTrainRole = GEAR_ROLE_DRIVEN Then
        If mDrivenPatternUsesBossFace Or swGearAxisFeat Is Nothing Then
            TryCreateDrivenRefAxisFromBossFace
        End If
    End If

    CreateAddendumBlank = True
    Exit Function

BlankFail:
    DebugPrint "CreateAddendumBlank FAIL: " & Err.Description
    CreateAddendumBlank = False
End Function

' ---------------------------------------------------------------------------
' 5. GenerateInvolutePoints - one flank, meters, parametric involute
' ---------------------------------------------------------------------------
Public Function GenerateInvolutePoints(ByRef outPoints() As Double, ByRef pointCount As Long) As Boolean
    On Error GoTo GenPtsFail

    Dim i As Long
    Dim t As Double
    Dim rb As Double
    Dim x As Double
    Dim y As Double
    Dim rotAngle As Double
    Dim tPitch As Double
    Dim xPitch As Double
    Dim yPitch As Double
    Dim phiPitch As Double

    rb = GearGeom.BaseRadiusM
    pointCount = INVOLUTE_POINT_COUNT
    ReDim outPoints(0 To pointCount * 3 - 1)

    InvoluteXY GearGeom.InvoluteTPitch, rb, xPitch, yPitch
    phiPitch = Atn2(yPitch, xPitch)
    rotAngle = GearGeom.HalfToothAngleRad - phiPitch

    For i = 0 To pointCount - 1
        If pointCount = 1 Then
            t = GearGeom.InvoluteTStart
        Else
            t = GearGeom.InvoluteTStart + (GearGeom.InvoluteTEnd - GearGeom.InvoluteTStart) * CDbl(i) / CDbl(pointCount - 1)
        End If
        InvoluteXY t, rb, x, y
        RotatePoint x, y, rotAngle, x, y
        If GearGeom.IsInternal Then RotatePoint x, y, Pi, x, y
        outPoints(i * 3) = x
        outPoints(i * 3 + 1) = y
        outPoints(i * 3 + 2) = 0#
    Next i

    DebugPrint "GenerateInvolutePoints OK (" & pointCount & " points, rot=" & Format(rotAngle * 180# / Pi, "0.###") & _
        " deg" & IIf(GearGeom.IsInternal, ", internal", "") & ")"
    GenerateInvolutePoints = True
    Exit Function

GenPtsFail:
    DebugPrint "GenerateInvolutePoints FAIL: " & Err.Description
    GenerateInvolutePoints = False
End Function

Private Sub InvoluteXY(ByVal t As Double, ByVal rb As Double, ByRef x As Double, ByRef y As Double)
    x = rb * (Cos(t) + t * Sin(t))
    y = rb * (Sin(t) - t * Cos(t))
End Sub

' VBA has no built-in Atn2 (unlike VB.NET); required for involute arc angles
Public Function Atn2(ByVal y As Double, ByVal x As Double) As Double
    If x > 0# Then
        Atn2 = Atn(y / x)
    ElseIf x < 0# Then
        Atn2 = Atn(y / x) + Pi
    ElseIf y > 0# Then
        Atn2 = Pi / 2#
    ElseIf y < 0# Then
        Atn2 = -Pi / 2#
    Else
        Atn2 = 0#
    End If
End Function

' ---------------------------------------------------------------------------
' 6. RotatePoint
' ---------------------------------------------------------------------------
Public Sub RotatePoint(ByVal xIn As Double, ByVal yIn As Double, ByVal angleRad As Double, ByRef xOut As Double, ByRef yOut As Double)
    Dim c As Double
    Dim s As Double
    c = Cos(angleRad)
    s = Sin(angleRad)
    xOut = xIn * c - yIn * s
    yOut = xIn * s + yIn * c
End Sub

' Force point onto circle (keeps angle); ties spline ends to addendum / dedendum
Private Sub SnapPointToRadius(ByVal rTarget As Double, ByRef x As Double, ByRef y As Double)
    Dim r As Double
    r = Sqr(x * x + y * y)
    If r > 0.0000000001 Then
        x = rTarget * x / r
        y = rTarget * y / r
    End If
End Sub

' Minor arc between two angles (short way on circle — between teeth, not around the blank)
Private Function ArcMidAngleShort(ByVal angStart As Double, ByVal angEnd As Double) As Double
    Dim span As Double
    span = angEnd - angStart
    Do While span > Pi
        span = span - 2# * Pi
    Loop
    Do While span <= -Pi
        span = span + 2# * Pi
    Loop
    ArcMidAngleShort = angStart + span / 2#
End Function

Private Function CreateCircleArcBetween(ByVal r As Double, _
    ByVal xStart As Double, ByVal yStart As Double, _
    ByVal xEnd As Double, ByVal yEnd As Double) As SldWorks.SketchSegment
    Dim angS As Double, angE As Double, angM As Double
    angS = Atn2(yStart - mSketchOffsetYM, xStart - mSketchOffsetXM)
    angE = Atn2(yEnd - mSketchOffsetYM, xEnd - mSketchOffsetXM)
    angM = ArcMidAngleShort(angS, angE)
    Set CreateCircleArcBetween = swSketchManager.Create3PointArc( _
        xStart, yStart, 0, xEnd, yEnd, 0, _
        mSketchOffsetXM + r * Cos(angM), mSketchOffsetYM + r * Sin(angM), 0)
End Function

Private Sub OffsetSketchPointArray(ByRef pts() As Double, ByVal n As Long)
    Dim i As Long
    For i = 0 To n - 1
        pts(i * 3) = pts(i * 3) + mSketchOffsetXM
        pts(i * 3 + 1) = pts(i * 3 + 1) + mSketchOffsetYM
    Next i
End Sub

Private Sub SnapGapSplineEndsToCircles(ByRef ptsFlank() As Double, ByVal n As Long, ByVal ra As Double, ByVal rr As Double)
    SnapPointToRadius rr, ptsFlank(0), ptsFlank(1)
    SnapPointToRadius ra, ptsFlank((n - 1) * 3), ptsFlank((n - 1) * 3 + 1)
End Sub

' Coincident between spline end and arc end so the cut contour is closed
Private Sub CoincidentSketchEnds(ByVal seg1 As SldWorks.SketchSegment, ByVal useEnd1 As Boolean, _
    ByVal seg2 As SldWorks.SketchSegment, ByVal useEnd2 As Boolean)
    Dim sp1 As SldWorks.SketchPoint
    Dim sp2 As SldWorks.SketchPoint
    On Error Resume Next
    If useEnd1 Then Set sp1 = seg1.GetEndPoint2 Else Set sp1 = seg1.GetStartPoint2
    If useEnd2 Then Set sp2 = seg2.GetEndPoint2 Else Set sp2 = seg2.GetStartPoint2
    If sp1 Is Nothing Or sp2 Is Nothing Then Exit Sub
    sp1.Select4 False, Nothing
    sp2.Select4 True, Nothing
    swDoc.SketchAddConstraints "sgCOINCIDENT"
    ClearSWSelection
    On Error GoTo 0
End Sub

Private Sub ConstrainGapSketchClosed()
    On Error Resume Next
    swSketchManager.AddToDB = False
    swSketchManager.AutoSolve = True
    ' Flank A: root=start, tip=end. Tip arc: start=high Y (A tip), end=B tip. Root arc: start=high Y root, end=B root
    CoincidentSketchEnds swInvoluteSpline, False, swRootArc, False      ' A root -> root arc start
    CoincidentSketchEnds swInvoluteSpline, True, swTipLandArc, False     ' A tip -> tip arc start
    CoincidentSketchEnds swMirroredInvoluteSpline, True, swTipLandArc, True   ' B tip -> tip arc end
    CoincidentSketchEnds swMirroredInvoluteSpline, False, swRootArc, True     ' B root -> root arc end
    swSketchManager.AddToDB = True
    On Error GoTo 0
End Sub

' Root arc on Dd; tip arc on Da (minor arc between flank ends, not around circumference)
Private Function AddGapTipRootArcs(ByVal ra As Double, ByVal rr As Double, _
    ByVal xRootA As Double, ByVal yRootA As Double, ByVal xRootB As Double, ByVal yRootB As Double, _
    ByVal xTipA As Double, ByVal yTipA As Double, ByVal xTipB As Double, ByVal yTipB As Double) As Boolean

    AddGapTipRootArcs = False

    Set swRootArc = CreateCircleArcBetween(rr, xRootA, yRootA, xRootB, yRootB)
    If swRootArc Is Nothing Then Exit Function

    Set swTipLandArc = CreateCircleArcBetween(ra, xTipA, yTipA, xTipB, yTipB)
    If swTipLandArc Is Nothing Then Exit Function

    AddGapTipRootArcs = True
End Function

' ---------------------------------------------------------------------------
' 7. CreateToothGapSketch - closed gap between teeth (material removed by cut)
'    Flank A = right flank of reference tooth; Flank B = left flank of next tooth (+pitch)
' ---------------------------------------------------------------------------
Public Function CreateToothGapSketch() As Boolean
    On Error GoTo SketchFail

    Dim pts() As Double
    Dim ptsRight() As Double
    Dim ptsNextLeft() As Double
    Dim n As Long
    Dim i As Long
    Dim ra As Double
    Dim rr As Double
    Dim pitch As Double
    Dim x As Double
    Dim y As Double
    Dim xTipA As Double, yTipA As Double
    Dim xTipB As Double, yTipB As Double
    Dim xRootA As Double, yRootA As Double
    Dim xRootB As Double, yRootB As Double
    Dim sketchStep As String
    Dim halfPitch As Double

    sketchStep = "init"
    If Not GenerateInvolutePoints(pts, n) Then GoTo SketchFail

    ra = GearGeom.AddendumRadiusM
    rr = GearGeom.DedendumRadiusM
    pitch = GearGeom.AngularPitchRad
    halfPitch = pitch / 2#

    sketchStep = "begin sketch"
    If Not BeginSketchOnFrontPlane() Then
        DebugPrint "CreateToothGapSketch: BeginSketchOnFrontPlane failed"
        GoTo SketchFail
    End If
    Set sw2DSketch = swSketchManager.ActiveSketch

    ' Center gap on +X (reference tooth is at 0; gap center is at half pitch)
    For i = 0 To n - 1
        RotatePoint pts(i * 3), pts(i * 3 + 1), -halfPitch, pts(i * 3), pts(i * 3 + 1)
    Next i

    ' Right flank: root at index 0, tip at n-1 (dedendum -> addendum, same order as flank B)
    sketchStep = "gap flank A"
    ReDim ptsRight(0 To n * 3 - 1)
    For i = 0 To n - 1
        ptsRight(i * 3) = pts(i * 3)
        ptsRight(i * 3 + 1) = -pts(i * 3 + 1)
        ptsRight(i * 3 + 2) = 0#
    Next i
    SnapGapSplineEndsToCircles ptsRight, n, ra, rr

    ' Left flank of next tooth (+one pitch); root index 0, tip index n-1
    sketchStep = "gap flank B"
    ReDim ptsNextLeft(0 To n * 3 - 1)
    For i = 0 To n - 1
        RotatePoint pts(i * 3), pts(i * 3 + 1), pitch, x, y
        ptsNextLeft(i * 3) = x
        ptsNextLeft(i * 3 + 1) = y
        ptsNextLeft(i * 3 + 2) = 0#
    Next i
    SnapGapSplineEndsToCircles ptsNextLeft, n, ra, rr

    OffsetSketchPointArray ptsRight, n
    OffsetSketchPointArray ptsNextLeft, n

    xRootA = ptsRight(0): yRootA = ptsRight(1)
    xTipA = ptsRight((n - 1) * 3): yTipA = ptsRight((n - 1) * 3 + 1)
    xRootB = ptsNextLeft(0): yRootB = ptsNextLeft(1)
    xTipB = ptsNextLeft((n - 1) * 3): yTipB = ptsNextLeft((n - 1) * 3 + 1)

    Set swInvoluteSpline = swSketchManager.CreateSpline(ptsRight)
    If swInvoluteSpline Is Nothing Then GoTo SketchFail
    Set swMirroredInvoluteSpline = swSketchManager.CreateSpline(ptsNextLeft)
    If swMirroredInvoluteSpline Is Nothing Then GoTo SketchFail

    ' Arcs use exact spline corner coords; CW on Da/Dd; high-Y end on flank A
    sketchStep = "gap tip and root arcs"
    If Not AddGapTipRootArcs(ra, rr, xRootA, yRootA, xRootB, yRootB, xTipA, yTipA, xTipB, yTipB) Then GoTo SketchFail

    AddDiameterDimensionOnConstructionCircle xTipA, yTipA, GEAR_DIM_ADDENDUM_DIAMETER, GearGeom.AddendumDiameterMM
    AddDiameterDimensionOnConstructionCircle xRootA, yRootA, GEAR_DIM_DEDENDUM_DIAMETER, GearGeom.DedendumDiameterMM

    sketchStep = "coincident corners"
    ConstrainGapSketchClosed

    sketchStep = "capture sketch"
    Set swToothGapSketchFeat = FindLastSketchFeature()
    If swToothGapSketchFeat Is Nothing Then
        DebugPrint "CreateToothGapSketch: could not get sketch feature"
        GoTo SketchFail
    End If

    sketchStep = "exit sketch"
    swSketchManager.InsertSketch True
    ClearSWSelection

    DebugPrint "CreateToothGapSketch OK (gap pitch=" & Format(pitch * 180# / Pi, "0.###") & " deg)"
    CreateToothGapSketch = True
    Exit Function

SketchFail:
    DebugPrint "CreateToothGapSketch FAIL (" & sketchStep & "): " & Err.Number & " " & Err.Description
    CreateToothGapSketch = False
End Function

' FeatureCut4: through cut; multibody parts use feature scope on current gear body only
Private Function TryCreateGapCutFeature(ByRef methodUsed As String) As SldWorks.Feature
    Dim lastErr As Long
    Dim lastDesc As String
    Dim useScope As Boolean
    Dim useFeatScope As Boolean
    Dim useAutoSelect As Boolean

    Set TryCreateGapCutFeature = Nothing
    methodUsed = ""
    useScope = PartNeedsCutFeatureScope()
    useFeatScope = useScope
    useAutoSelect = Not useScope
    On Error Resume Next

    Err.Clear
    Set TryCreateGapCutFeature = swFeatMgr.FeatureCut4( _
        True, False, False, _
        SW_END_THROUGH_ALL, SW_END_BLIND, 0#, 0#, _
        False, False, False, False, _
        0#, 0#, _
        False, False, False, False, _
        False, _
        useFeatScope, useAutoSelect, _
        False, False, False, _
        SW_START_SKETCH_PLANE, 0#, False, False)
    If Not TryCreateGapCutFeature Is Nothing Then
        methodUsed = "FeatureCut4 ThroughAll"
        Exit Function
    End If
    lastErr = Err.Number: lastDesc = Err.Description: Err.Clear

    Set TryCreateGapCutFeature = swFeatMgr.FeatureCut4( _
        True, False, False, _
        SW_END_BLIND, SW_END_BLIND, GearGeom.WidthM, 0#, _
        False, False, False, False, _
        0#, 0#, _
        False, False, False, False, _
        False, _
        useFeatScope, useAutoSelect, _
        False, False, False, _
        SW_START_SKETCH_PLANE, 0#, False, False)
    If Not TryCreateGapCutFeature Is Nothing Then
        methodUsed = "FeatureCut4 Blind depth"
        Exit Function
    End If
    lastErr = Err.Number: lastDesc = Err.Description: Err.Clear

    ' Reverse cut direction (same 27-arg FeatureCut4 API)
    Set TryCreateGapCutFeature = swFeatMgr.FeatureCut4( _
        True, False, True, _
        SW_END_THROUGH_ALL, SW_END_BLIND, 0#, 0#, _
        False, False, False, False, _
        0#, 0#, _
        False, False, False, False, _
        False, _
        useFeatScope, useAutoSelect, _
        False, False, False, _
        SW_START_SKETCH_PLANE, 0#, False, False)
    If Not TryCreateGapCutFeature Is Nothing Then
        methodUsed = "FeatureCut4 ThroughAll (Dir reversed)"
        Exit Function
    End If
    lastErr = Err.Number: lastDesc = Err.Description: Err.Clear

    ' Multibody fallback: sketch-only + feature scope auto-intersect (tooth sketch is on driven blank only)
    If useScope Then
        Set TryCreateGapCutFeature = swFeatMgr.FeatureCut4( _
            True, False, False, _
            SW_END_THROUGH_ALL, SW_END_BLIND, 0#, 0#, _
            False, False, False, False, _
            0#, 0#, _
            False, False, False, False, _
            False, _
            True, True, _
            False, False, False, _
            SW_START_SKETCH_PLANE, 0#, False, False)
        If Not TryCreateGapCutFeature Is Nothing Then
            methodUsed = "FeatureCut4 ThroughAll (scope auto-intersect)"
            Exit Function
        End If
        lastErr = Err.Number: lastDesc = Err.Description: Err.Clear
    End If

    DebugPrint "TryCreateGapCutFeature FeatureCut4 failed: " & lastErr & " " & lastDesc
    On Error GoTo 0
End Function

' ---------------------------------------------------------------------------
' 8. CutExtrudeToothGap - through-all cut in addendum blank
' ---------------------------------------------------------------------------
Public Function CutExtrudeToothGap() As Boolean
    On Error GoTo CutFail
    Dim methodUsed As String
    Dim selCount As Long

    If swToothGapSketchFeat Is Nothing Then Set swToothGapSketchFeat = FindLastSketchFeature()
    If swToothGapSketchFeat Is Nothing Then
        DebugPrint "CutExtrudeToothGap: no gap sketch feature"
        GoTo CutFail
    End If

    RebuildSWPart
    If Not SelectGapSketchForCut() Then
        DebugPrint "CutExtrudeToothGap: sketch select failed (" & swToothGapSketchFeat.Name & ")"
        GoTo CutFail
    End If

    If PartNeedsCutFeatureScope() Then
        If SelectCurrentGearBodyForCutScope() Then
            If Not SelectGapSketchForCutAppend() Then
                DebugPrint "CutExtrudeToothGap: warning — could not re-select sketch after body scope"
            End If
        Else
            DebugPrint "CutExtrudeToothGap: warning — could not pre-select body for feature scope"
        End If
    End If

    selCount = swSelMgr.GetSelectedObjectCount2(-1)
    DebugPrint "CutExtrudeToothGap: selections before cut, count=" & selCount

    Set swToothCutFeat = TryCreateGapCutFeature(methodUsed)
    If swToothCutFeat Is Nothing Then GoTo CutFail

    If PartNeedsCutFeatureScope() Then
        SetCutFeatureScopeToCurrentGearBody swToothCutFeat
    End If

    DebugPrint "Tooth gap cut feature: " & swToothCutFeat.Name & " (" & methodUsed & _
        IIf(PartNeedsCutFeatureScope(), ", scoped to current gear body)", ")")
    ClearSWSelection
    RebuildSWPart
    DebugPrint "CutExtrudeToothGap OK (through all)"
    CutExtrudeToothGap = True
    Exit Function

CutFail:
    DebugPrint "CutExtrudeToothGap FAIL: " & Err.Number & " " & Err.Description
    CutExtrudeToothGap = False
End Function

' SW doc: mark 4 = features to pattern, mark 1 = axis; feature selected FIRST
Private Function PreselectCircularPattern(ByRef stepDetail As String) As Boolean
    Dim toothName As String
    Dim axisName As String
    Dim toothOk As Boolean
    Dim axisOk As Boolean
    Dim swCylFace As SldWorks.Face2

    PreselectCircularPattern = False
    stepDetail = "clear"
    ClearSWSelection

    toothName = swToothCutFeat.Name
    axisName = mAxisName
    If axisName = "" And Not swGearAxisFeat Is Nothing Then axisName = swGearAxisFeat.Name

    stepDetail = "tooth BODYFEATURE mark4"
    toothOk = swModelDocExt.SelectByID2(toothName, "BODYFEATURE", 0, 0, 0, False, 4, Nothing, 0)
    If Not toothOk Then toothOk = SelectForCirPatternMark(swToothCutFeat, 4, False)
    If Not toothOk Then
        DebugPrint "Preselect: tooth failed for " & toothName
        Exit Function
    End If

    stepDetail = "axis mark1"
    axisOk = False
    If axisName <> "" Then
        axisOk = swModelDocExt.SelectByID2(axisName, "AXIS", 0, 0, 0, True, 1, Nothing, 0)
        If Not axisOk Then axisOk = swModelDocExt.SelectByID2(axisName, "REFAXIS", 0, 0, 0, True, 1, Nothing, 0)
        If Not axisOk Then axisOk = swModelDocExt.SelectByID2(axisName, "RefAxis", 0, 0, 0, True, 1, Nothing, 0)
        If Not axisOk Then axisOk = SelectAxisByNameWithDocSuffix(axisName)
    End If
    If Not axisOk And Not swGearAxisFeat Is Nothing Then
        axisOk = SelectForCirPatternMark(swGearAxisFeat, 1, True)
    End If
    If Not axisOk And (mDrivenPatternUsesBossFace Or GearTrainRole = GEAR_ROLE_DRIVEN) Then
        Set swCylFace = FindAddendumCylindricalFace()
        If Not swCylFace Is Nothing Then
            axisOk = SelectForCirPatternMark(swCylFace, 1, True)
            If axisOk Then DebugPrint "Preselect: pattern axis = addendum cylindrical face"
        End If
    End If
    If Not axisOk Then
        DebugPrint "Preselect: axis failed for " & axisName
        Exit Function
    End If

    PreselectCircularPattern = True
End Function

Private Function CreateCircularPatternFromSelection(ByRef stepDetail As String) As Boolean

End Function

' ---------------------------------------------------------------------------
' 9. CircularPatternTooth - pattern the cut-extrude (tooth gap), not the blank boss
' ---------------------------------------------------------------------------
Public Function CircularPatternTooth() As Boolean
    On Error GoTo PatternFail
    Dim swFeatData As Object
    Dim featArr(0) As Object
    Dim vFeatArr As Variant
    Dim stepDetail As String
    Dim toothName As String
    Dim axisName As String

    stepDetail = "resolve refs"
    mUseSketchCenterlineAsAxis = False
    If swGearAxisFeat Is Nothing Then
        Set swGearAxisFeat = FindRefAxisFeatureByName(BuildGearAxisFeatureName())
    End If
    If swGearAxisFeat Is Nothing And mAxisName <> "" Then
        Set swGearAxisFeat = FindRefAxisFeatureByName(mAxisName)
    End If
    ' Driving gear only: legacy parts may still have Axis1 / unnamed origin axis
    If swGearAxisFeat Is Nothing And GearTrainRole <> GEAR_ROLE_DRIVEN Then
        Set swGearAxisFeat = FindLastRefAxisFeature()
    End If
    ResolveToothCutForPattern
    If swToothCutFeat Is Nothing Then
        DebugPrint "CircularPatternTooth: missing tooth gap cut"
        GoTo PatternFail
    End If
    If GearTrainRole = GEAR_ROLE_DRIVEN Then
        RebuildSWPart
        If swGearAxisFeat Is Nothing Or mDrivenPatternUsesBossFace Then TryCreateDrivenRefAxisFromBossFace
        If swGearAxisFeat Is Nothing Then
            Set swGearAxisFeat = FindRefAxisFeatureByName(BuildGearAxisFeatureName())
        End If
        If swGearAxisFeat Is Nothing And mAxisName <> "" Then
            Set swGearAxisFeat = FindRefAxisFeatureByName(mAxisName)
        End If
    End If

    If swGearAxisFeat Is Nothing And mAxisName = "" And FindAddendumCylindricalFace() Is Nothing Then
        DebugPrint "CircularPatternTooth: missing ref axis and no cylindrical face on blank"
        GoTo PatternFail
    End If

    toothName = swToothCutFeat.Name
    axisName = mAxisName
    If axisName = "" And Not swGearAxisFeat Is Nothing Then axisName = swGearAxisFeat.Name
    DebugPrint "Pattern seed=" & toothName & " axis=" & axisName & _
        IIf(mDrivenPatternUsesBossFace And axisName = "", " (cylindrical face)", "")

    ' Method 1: SelectByID2 BODYFEATURE + AXIS (correct order) then CreateFeature
    stepDetail = "select BODYFEATURE+AXIS"
    If CreateCircularPatternFromSelection(stepDetail) Then GoTo PatternOk

    ' Method 2: FeatureCircularPattern4 (legacy API, selection must be active)
    stepDetail = "FeatureCircularPattern4"
    ClearSWSelection
    If PreselectCircularPattern(stepDetail) Then
        On Error Resume Next
        Set swPatternFeat = swFeatMgr.FeatureCircularPattern4(GearGeom.NumTeeth, 2# * Pi, True, "NULL", False, False, True)
        If swPatternFeat Is Nothing And mAxisName <> "" Then
            Set swPatternFeat = swFeatMgr.FeatureCircularPattern4(GearGeom.NumTeeth, 2# * Pi, True, mAxisName, False, False, True)
        End If
        On Error GoTo PatternFail
        If Not swPatternFeat Is Nothing Then GoTo PatternOk
        DebugPrint "FeatureCircularPattern4 returned Nothing after preselect"
    End If

    ' Method 2b: FeaturesToPattern + ref axis feature (driven Axis2)
    If Not mUseSketchCenterlineAsAxis And Not swGearAxisFeat Is Nothing Then
        stepDetail = "FeaturesToPattern+RefAxis"
        Set swFeatData = swFeatMgr.CreateDefinition(swFeatureNameID_e.swFmCirPattern)
        If Not swFeatData Is Nothing Then
            swFeatData.BodyPattern = False
            swFeatData.GeometryPattern = False
            swFeatData.EqualSpacing = True
            swFeatData.ReverseDirection = False
            swFeatData.Spacing = 2# * Pi
            swFeatData.TotalInstances = GearGeom.NumTeeth
            swFeatData.Direction2 = False
            swFeatData.VarySketch = False
            On Error Resume Next
            Set featArr(0) = swToothCutFeat
            vFeatArr = featArr
            swFeatData.FeaturesToPattern = vFeatArr
            Set swFeatData.Axis = swGearAxisFeat
            Set swPatternFeat = swFeatMgr.CreateFeature(swFeatData)
            If Err.Number <> 0 Then
                DebugPrint "FeaturesToPattern+RefAxis err: " & Err.Number & " " & Err.Description
                Err.Clear
            End If
            On Error GoTo PatternFail
            If Not swPatternFeat Is Nothing Then GoTo PatternOk
        End If
    End If

    ' Method 3: select edge on axis (ref axis only; sketch centerline uses Method 1/2)
    stepDetail = "axis edge"
    ClearSWSelection
    If Not mUseSketchCenterlineAsAxis Then
    If swModelDocExt.SelectByID2(toothName, "BODYFEATURE", 0, 0, 0, False, 4, Nothing, 0) Then
        If swModelDocExt.SelectByID2(axisName, "EDGE", 0, 0, 0, True, 1, Nothing, 0) Then
            Set swFeatData = swFeatMgr.CreateDefinition(swFeatureNameID_e.swFmCirPattern)
            If Not swFeatData Is Nothing Then
                swFeatData.BodyPattern = False
                swFeatData.EqualSpacing = True
                swFeatData.Spacing = 2# * Pi
                swFeatData.TotalInstances = GearGeom.NumTeeth
                Set swPatternFeat = swFeatMgr.CreateFeature(swFeatData)
                If Not swPatternFeat Is Nothing Then GoTo PatternOk
            End If
        End If
    End If
    End If

    ' Method 4: FeaturesToPattern + Axis on feature data (ref axis only)
    If Not mUseSketchCenterlineAsAxis Then
    stepDetail = "FeaturesToPattern property"
    Set swFeatData = swFeatMgr.CreateDefinition(swFeatureNameID_e.swFmCirPattern)
    If Not swFeatData Is Nothing Then
        swFeatData.BodyPattern = False
        swFeatData.GeometryPattern = False
        swFeatData.EqualSpacing = True
        swFeatData.ReverseDirection = False
        swFeatData.Spacing = 2# * Pi
        swFeatData.TotalInstances = GearGeom.NumTeeth
        swFeatData.Direction2 = False
        swFeatData.VarySketch = False
        On Error Resume Next
        Set featArr(0) = swToothCutFeat
        vFeatArr = featArr
        swFeatData.FeaturesToPattern = vFeatArr
        Set swFeatData.Axis = swGearAxisFeat
        Set swPatternFeat = swFeatMgr.CreateFeature(swFeatData)
        On Error GoTo PatternFail
        If Not swPatternFeat Is Nothing Then GoTo PatternOk
        DebugPrint "FeaturesToPattern CreateFeature failed: " & Err.Number & " " & Err.Description
        Err.Clear
    End If
    End If

    GoTo PatternFail

PatternOk:
    ClearSWSelection
    RebuildSWPart
    DebugPrint "CircularPatternTooth OK (instances=" & GearGeom.NumTeeth & ", " & swPatternFeat.Name & ")"
    CircularPatternTooth = True
    Exit Function

PatternFail:
    DebugPrint "CircularPatternTooth FAIL (" & stepDetail & "): " & Err.Number & " " & Err.Description
    CircularPatternTooth = False
End Function

' ---------------------------------------------------------------------------
' 10. CombineGearBodies - union leftover solids (not a Cut-Extrude; skip if already one body)
' ---------------------------------------------------------------------------
Public Function CombineGearBodies() As Boolean
    On Error GoTo CombineDone

    Const swCombineAdd As Long = 0   ' swCombineBodiesOperationType_e.swCombineOperationType_Add

    Dim vBodies As Variant
    Dim swBody As SldWorks.Body2
    Dim i As Long
    Dim bodyCount As Long
    Dim appendSel As Boolean

    CombineGearBodies = True
    RebuildSWPart
    vBodies = swPart.GetBodies2(swSolidBody, False)
    If IsEmpty(vBodies) Then
        DebugPrint "CombineGearBodies: no solid bodies"
        CombineGearBodies = False
        Exit Function
    End If

    If IsArray(vBodies) Then
        bodyCount = UBound(vBodies) - LBound(vBodies) + 1
    Else
        bodyCount = 1
    End If

    DebugPrint "CombineGearBodies: body count=" & bodyCount
    If bodyCount < 2 Then
        DebugPrint "CombineGearBodies: skipped (single solid already merged)"
        Exit Function
    End If

    ClearSWSelection
    For i = LBound(vBodies) To UBound(vBodies)
        Set swBody = vBodies(i)
        appendSel = (i > LBound(vBodies))
        If Not SelectSolidBodyForCombine(swBody, appendSel) Then
            DebugPrint "CombineGearBodies: could not select body index " & i
            GoTo CombineDone
        End If
    Next i

    Set swCombineFeat = swFeatMgr.InsertCombineFeature(swCombineAdd, Nothing, False)
    If swCombineFeat Is Nothing Then
        DebugPrint "CombineGearBodies: InsertCombineFeature returned Nothing"
        GoTo CombineDone
    End If

    ClearSWSelection
    RebuildSWPart
    DebugPrint "CombineGearBodies OK (" & swCombineFeat.Name & ")"
    CombineGearBodies = True
    Exit Function

CombineDone:
    DebugPrint "CombineGearBodies skipped or failed: " & Err.Number & " " & Err.Description
    ClearSWSelection
    CombineGearBodies = False
End Function

' ---------------------------------------------------------------------------
' 11. RebuildAndValidate
' ---------------------------------------------------------------------------
Public Function RebuildAndValidate() As Boolean
    On Error GoTo ValFail
    Dim vBodies As Variant
    Dim bodyCount As Long
    Dim feat As SldWorks.Feature
    Dim failed As Boolean
    Dim errCount As Long

    RebuildSWPart
    failed = False
    errCount = 0

    Set feat = swPart.FirstFeature
    Do While Not feat Is Nothing
        If feat.IsSuppressed = False Then
            If feat.GetErrorCode2(True) <> 0 Then
                errCount = errCount + 1
                DebugPrint "Feature error: " & feat.Name & " code=" & feat.GetErrorCode2(True)
                failed = True
            End If
        End If
        Set feat = feat.GetNextFeature
    Loop

    vBodies = swPart.GetBodies2(swSolidBody, False)
    If IsEmpty(vBodies) Then
        bodyCount = 0
        failed = True
    ElseIf IsArray(vBodies) Then
        bodyCount = UBound(vBodies) - LBound(vBodies) + 1
    Else
        bodyCount = 1
    End If

    DebugPrint "RebuildAndValidate: bodies=" & bodyCount & ", feature errors=" & errCount

    If bodyCount < 1 Then failed = True
    If failed Then
        RebuildAndValidate = False
        MsgBox "Gear rebuild reported problems. Bodies=" & bodyCount & ", feature errors=" & errCount & vbCrLf & _
            "See Immediate Window (Ctrl+G) for debug log.", vbExclamation, "Gear Validation"
    Else
        RebuildAndValidate = True
    End If
    Exit Function

ValFail:
    DebugPrint "RebuildAndValidate FAIL: " & Err.Description
    RebuildAndValidate = False
End Function

Public Function CenterDistanceMM(ByVal moduleMM As Double, ByVal z1 As Long, ByVal z2 As Long, _
    Optional ByVal gear1Internal As Boolean = False, Optional ByVal gear2Internal As Boolean = False) As Double
    If gear1Internal And Not gear2Internal Then
        CenterDistanceMM = moduleMM * (CDbl(z1) - CDbl(z2)) / 2#
    ElseIf gear2Internal And Not gear1Internal Then
        CenterDistanceMM = moduleMM * (CDbl(z2) - CDbl(z1)) / 2#
    Else
        CenterDistanceMM = moduleMM * (CDbl(z1) + CDbl(z2)) / 2#
    End If
End Function

End Function

Private Function TryParseGearBodyName(ByVal bodyName As String, ByRef pitchDiameterMM As Double, ByRef numTeeth As Long) As Boolean
    Dim s As String
    Dim rest As String
    Dim zPart As String
    Dim teethPart As String
    Dim diaPart As String
    Dim i As Long
    Dim ch As String
    Dim tPos As Long

    TryParseGearBodyName = False
    pitchDiameterMM = 0#
    numTeeth = 0
    s = Trim$(bodyName)
    If Len(s) < 3 Then Exit Function

    ' Current format: {teeth}T {pitch dia}  e.g. 100T 125
    i = 1
    Do While i <= Len(s)
        ch = Mid$(s, i, 1)
        If ch >= "0" And ch <= "9" Then
            i = i + 1
        Else
            Exit Do
        End If
    Loop
    If i >= 2 Then
        teethPart = Left$(s, i - 1)
        tPos = i
        If UCase$(Mid$(s, tPos, 1)) = "T" Then
            rest = Trim$(Mid$(s, tPos + 1))
            If rest <> "" Then
                i = 1
                Do While i <= Len(rest)
                    ch = Mid$(rest, i, 1)
                    If (ch >= "0" And ch <= "9") Or ch = "." Then
                        i = i + 1
                    Else
                        Exit Do
                    End If
                Loop
                If i >= 2 Then
                    diaPart = Left$(rest, i - 1)
                    numTeeth = CLng(teethPart)
                    pitchDiameterMM = CDbl(diaPart)
                    If numTeeth >= 3 And pitchDiameterMM > 0# Then
                        TryParseGearBodyName = True
                        Exit Function
                    End If
                End If
            End If
        End If
    End If

    ' Legacy: XT {pitch} z{teeth}  e.g. XT 125 z100
    If Len(s) < 4 Then Exit Function
    If UCase$(Left$(s, 2)) <> "XT" Then Exit Function
    If Mid$(s, 3, 1) <> " " And Mid$(s, 3, 1) <> vbTab Then Exit Function
    rest = Trim$(Mid$(s, 4))
    If rest = "" Then Exit Function

    i = 1
    Do While i <= Len(rest)
        ch = Mid$(rest, i, 1)
        If (ch >= "0" And ch <= "9") Or ch = "." Then
            i = i + 1
        Else
            Exit Do
        End If
    Loop
    If i < 2 Then Exit Function
    pitchDiameterMM = CDbl(Left$(rest, i - 1))

    zPart = Trim$(Mid$(rest, i))
    If zPart = "" Then Exit Function
    If UCase$(Left$(zPart, 1)) = "Z" Then zPart = Trim$(Mid$(zPart, 2))
    If zPart = "" Then Exit Function
    numTeeth = CLng(zPart)
    If numTeeth < 3 Then Exit Function
    TryParseGearBodyName = True
End Function

Public Function ReadGearMetadataFromBody(ByVal swBody As SldWorks.Body2, ByVal swPartDoc As SldWorks.ModelDoc2, _
    ByRef moduleMM As Double, ByRef numTeeth As Long, ByRef pressureDeg As Double, ByRef widthMM As Double) As Boolean
    Dim pitchDia As Double
    Dim swExt As SldWorks.ModelDocExtension

    ReadGearMetadataFromBody = False
    pressureDeg = 20#
    widthMM = GearGeom.WidthMM
    If widthMM <= 0# Then widthMM = 10#

    If Not swBody Is Nothing Then
        If TryParseGearBodyName(swBody.Name, pitchDia, numTeeth) Then
            moduleMM = pitchDia / CDbl(numTeeth)
            If swPartDoc Is Nothing Then GoTo BodyMetaOk
            Set swExt = swPartDoc.Extension
            If Not ReadPartCustomPropertyDouble(swExt, GEAR_PROP_PRESSURE, pressureDeg) Then pressureDeg = 20#
            If Not ReadPartCustomPropertyDouble(swExt, GEAR_PROP_WIDTH, widthMM) Then widthMM = 10#
BodyMetaOk:
            ReadGearMetadataFromBody = True
            Exit Function
        End If
    End If

    If Not swPartDoc Is Nothing Then
        ReadGearMetadataFromBody = ReadGearMetadataFromDoc(swPartDoc, moduleMM, numTeeth, pressureDeg, widthMM)
    End If
End Function

            Set FindSolidBodyByName = swBody
            Exit Function
        End If
    Next i
End Function

Public Function FindOpenPartDocumentByTitle(ByVal partTitle As String) As SldWorks.ModelDoc2
    Dim swDocWalk As SldWorks.ModelDoc2
    On Error Resume Next
    Set FindOpenPartDocumentByTitle = Nothing
    If partTitle = "" Then Exit Function
    Set FindOpenPartDocumentByTitle = swApp.GetOpenDocumentByName(partTitle)
    If Not FindOpenPartDocumentByTitle Is Nothing Then Exit Function
    Set swDocWalk = swApp.GetFirstDocument
    Do While Not swDocWalk Is Nothing
        If swDocWalk.GetTitle = partTitle And swDocWalk.GetType = swDocumentTypes_e.swDocPART Then

        Set swDocWalk = swDocWalk.GetNext
    Loop
    On Error GoTo 0
End Function

Private Function DrivingGearComboDisplayText(ByVal partTitle As String, ByVal bodyName As String, _
    ByVal hasGearMeta As Boolean, ByVal z1 As Long, ByVal modMM As Double) As String
    If hasGearMeta Then
        DrivingGearComboDisplayText = bodyName & "  (" & partTitle & ", z=" & z1 & ", m=" & Format(modMM, "0.###") & ")"
    Else
        DrivingGearComboDisplayText = bodyName & "  (" & partTitle & " — not a macro gear)"
    End If
End Function

Private Sub AddMacroGearBodiesFromPartToCombo(ByVal swPartDoc As SldWorks.ModelDoc2)
    Dim swPart As SldWorks.PartDoc
    Dim vBodies As Variant
    Dim swBody As SldWorks.Body2
    Dim i As Long
    Dim partTitle As String
    Dim bodyName As String

    If swPartDoc Is Nothing Then Exit Sub
    If swPartDoc.GetType <> swDocumentTypes_e.swDocPART Then Exit Sub
    Set swPart = swPartDoc
    partTitle = swPartDoc.GetTitle

    vBodies = swPart.GetBodies2(swBodyType_e.swSolidBody, True)
    If IsEmpty(vBodies) Then Exit Sub

    For i = LBound(vBodies) To UBound(vBodies)
        Set swBody = vBodies(i)
        bodyName = BodyDisplayName(swBody, i - LBound(vBodies) + 1)
        If IsMacroGearBody(swBody, swPartDoc) Then
            AppendDrivingGearComboEntry partTitle, bodyName
        End If
    Next i
End Sub

Private Sub AddMacroGearsFromAssemblyToCombo(ByVal swAssyDoc As SldWorks.ModelDoc2)
    Dim swAssy As SldWorks.AssemblyDoc
    Dim vComps As Variant
    Dim swComp As SldWorks.Component2
    Dim swPartDoc As SldWorks.ModelDoc2
    Dim i As Long
    Dim compName As String
    Dim partTitle As String
    Dim modMM As Double
    Dim z1 As Long
    Dim pa As Double
    Dim w As Double

    Set swAssy = swAssyDoc
    vComps = swAssy.GetComponents(False)
    If IsEmpty(vComps) Then
        AppendDrivingGearComboEntry swAssyDoc.GetTitle, "(no components)"
        Exit Sub
    End If

    For i = LBound(vComps) To UBound(vComps)
        Set swComp = vComps(i)
        If swComp Is Nothing Then GoTo NextComp
        Set swPartDoc = swComp.GetModelDoc2
        If swPartDoc Is Nothing Then GoTo NextComp
        If swPartDoc.GetType <> swDocumentTypes_e.swDocPART Then GoTo NextComp
        If Not ReadGearMetadataFromDoc(swPartDoc, modMM, z1, pa, w) Then GoTo NextComp
        compName = swComp.Name2
        partTitle = swPartDoc.GetTitle
        AppendDrivingGearComboEntry partTitle, compName
NextComp:
    Next i
End Sub

' Rebuild list from all open parts (and assembly macro gear components)
Public Function RefreshDrivingGearBodyComboList() As Long
    Dim swWalk As SldWorks.ModelDoc2

    ClearDrivingGearComboList
    Set swApp = Application.SldWorks
    RefreshDrivingGearBodyComboList = 0

    If swApp Is Nothing Then
        AppendDrivingGearComboEntry "", "(SolidWorks not running)"
        RefreshDrivingGearBodyComboList = GearTrainDrivingComboCount
        Exit Function
    End If

    Set swWalk = swApp.GetFirstDocument
    Do While Not swWalk Is Nothing
        Select Case swWalk.GetType
            Case swDocumentTypes_e.swDocPART
                AddMacroGearBodiesFromPartToCombo swWalk
            Case swDocumentTypes_e.swDocASSEMBLY
                AddMacroGearsFromAssemblyToCombo swWalk
        End Select
        Set swWalk = swWalk.GetNext
    Loop

    If GearTrainDrivingComboCount = 0 Then
        AppendDrivingGearComboEntry "", "(no macro gears — name bodies ""teethT pitchDia"" e.g. 100T 125, or generate a driving gear)"
    End If

    RefreshDrivingGearBodyComboList = GearTrainDrivingComboCount
End Function

Public Function GetPartDocFromSelectedBody(ByVal swBody As SldWorks.Body2) As SldWorks.ModelDoc2
    On Error Resume Next
    Set GetPartDocFromSelectedBody = Nothing
    If swBody Is Nothing Then Exit Function
    If Not swDoc Is Nothing Then
        If swDoc.GetType = swDocumentTypes_e.swDocPART Then
            Set GetPartDocFromSelectedBody = swDoc
        End If
    End If
    On Error GoTo 0
End Function

Public Function PickDrivingGearBodyFromSelection() As Boolean
    Dim swBody As SldWorks.Body2
    Dim swFace As SldWorks.Face2
    Dim swComp As SldWorks.Component2
    Dim swPartDoc As SldWorks.ModelDoc2
    Dim modMM As Double
    Dim z1 As Long
    Dim pa As Double
    Dim w As Double
    Dim selCount As Long
    Dim selType As Long

    PickDrivingGearBodyFromSelection = False
    Set swApp = Application.SldWorks
    Set swDoc = swApp.ActiveDoc
    If swDoc Is Nothing Then Exit Function
    Set swSelMgr = swDoc.SelectionManager
    selCount = swSelMgr.GetSelectedObjectCount2(-1)
    If selCount < 1 Then Exit Function

    selType = swSelMgr.GetSelectedObjectType3(1, -1)
    Set swPartDoc = Nothing
    Set swBody = Nothing

    Select Case selType
        Case swSelectType_e.swSelSOLIDBODIES
            Set swBody = swSelMgr.GetSelectedObject6(1, -1)
            If Not swBody Is Nothing Then Set swPartDoc = swBody.GetParent
        Case swSelectType_e.swSelFACES
            Set swFace = swSelMgr.GetSelectedObject6(1, -1)
            If Not swFace Is Nothing Then
                Set swBody = swFace.GetBody
                If Not swBody Is Nothing Then Set swPartDoc = swBody.GetParent
            End If
        Case swSelectType_e.swSelCOMPONENTS
            Set swComp = swSelMgr.GetSelectedObject6(1, -1)
            If Not swComp Is Nothing Then Set swPartDoc = swComp.GetModelDoc2
        Case Else
            Exit Function
    End Select

    If swPartDoc Is Nothing Then Exit Function
    If swPartDoc.GetType <> swDocumentTypes_e.swDocPART Then Exit Function
    If Not ReadGearMetadataFromBody(swBody, swPartDoc, modMM, z1, pa, w) Then Exit Function

    GearTrainDrivingPartTitle = swPartDoc.GetTitle
    If Not swBody Is Nothing Then
        GearTrainDrivingBodyLabel = Trim$(swBody.Name)
    Else
        GearTrainDrivingBodyLabel = "Solid Body"
    End If
    GearTrainDrivingModuleMM = modMM
    GearTrainDrivingTeeth = z1
    GearTrainDrivingPressureDeg = pa
    GearTrainDrivingWidthMM = w
    GearTrainDrivingIsInternal = ReadGearIsInternalFromDoc(swPartDoc)
    SyncDrivingGearAxisReference swPartDoc
    PickDrivingGearBodyFromSelection = True
    DebugPrint "Picked driving gear: " & GearTrainDrivingPartTitle & " / " & GearTrainDrivingBodyLabel & _
        " z=" & z1 & " m=" & modMM & " axis=" & GearTrainDrivingAxisName
End Function

Public Function ApplyDrivenGearInputsFromDrivingGear() As Boolean
    ApplyDrivenGearInputsFromDrivingGear = False
    If GearTrainDrivingTeeth <= 0 Or GearTrainDrivingModuleMM <= 0 Then Exit Function

    GearGeom.PressureAngleDeg = GearTrainDrivingPressureDeg
    GearGeom.ModuleMM = GearTrainDrivingModuleMM
    If GearTrainDrivingWidthMM > 0 Then GearGeom.WidthMM = GearTrainDrivingWidthMM

    GearTrainCenterDistanceMM = CenterDistanceMM(GearTrainDrivingModuleMM, GearTrainDrivingTeeth, GearGeom.NumTeeth, _
        GearTrainDrivingIsInternal, GearGeom.IsInternal)
    ApplyDrivenGearInputsFromDrivingGear = True
    DebugPrint "Driven gear inherits m=" & GearGeom.ModuleMM & " pa=" & GearGeom.PressureAngleDeg & _
        " C=" & Format(GearTrainCenterDistanceMM, "0.###") & " mm"
End Function

' Sketch dimension names (InvoluteGear_*) complement custom properties on the part.
Public Sub AddDiameterDimensionToCircle(ByVal swCircle As SldWorks.SketchSegment, ByVal dimName As String, ByVal diameterMM As Double)
    Dim diameterM As Double
    Dim radiusM As Double
    Dim dimX As Double
    Dim dimY As Double
    Dim swDispDim As Object
    Dim swDim As SldWorks.Dimension
    Dim showInput As Boolean
    Dim dispType As Long

    On Error Resume Next
    If swCircle Is Nothing Or swDoc Is Nothing Then Exit Sub
    diameterM = diameterMM / MM_PER_M
    radiusM = diameterM / 2#
    dimX = mSketchOffsetXM + radiusM * 0.85
    dimY = mSketchOffsetYM

    swCircle.Select4 False, Nothing
    showInput = swApp.GetUserPreferenceToggle(swUserPreferenceToggle_e.swInputDimValOnCreate)
    swApp.SetUserPreferenceToggle swUserPreferenceToggle_e.swInputDimValOnCreate, False

    Set swDispDim = swModelDocExt.AddDimension(dimX, dimY, 0#, 2) ' swSmartDimensionDirection_Right
    If swDispDim Is Nothing Then Set swDispDim = swDoc.AddDimension2(dimX, dimY, 0#)

    swApp.SetUserPreferenceToggle swUserPreferenceToggle_e.swInputDimValOnCreate, showInput

    If Not swDispDim Is Nothing Then
        Set swDim = swDispDim.GetDimension2(0)
        If Not swDim Is Nothing Then
            dispType = swDispDim.Type2
            If dispType = SW_DIM_TYPE_RADIAL Then
                swDim.SystemValue = radiusM
            Else
                swDim.SystemValue = diameterM
            End If
            swDim.Name = dimName
        End If
    End If
    ClearSWSelection
    On Error GoTo 0
End Sub

' Tooth-gap sketch: full construction circle so SW creates Ø (diameter), not R (radius)
Private Sub AddDiameterDimensionOnConstructionCircle(ByVal ptOnCircleX As Double, ByVal ptOnCircleY As Double, _
    ByVal dimName As String, ByVal diameterMM As Double)
    Dim swConstrCirc As SldWorks.SketchSegment
    On Error Resume Next
    Set swConstrCirc = swSketchManager.CreateCircle(mSketchOffsetXM, mSketchOffsetYM, 0, ptOnCircleX, ptOnCircleY, 0)
    If swConstrCirc Is Nothing Then Exit Sub
    swConstrCirc.ConstructionGeometry = True
    AddDiameterDimensionToCircle swConstrCirc, dimName, diameterMM
    On Error GoTo 0
End Sub
